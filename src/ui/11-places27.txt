  const TEMPLE_LV = { 0:0, 1:1, 2:2, 3:3, 5:4, 6:3 };
  function goTemple() {
    if (!canTemple(S) || phase !== 'port') return;
    const ci = S.city; S.temple[ci] = true; save();
    pzTemple({ title:TEMPLES[ci], sub:CITIES[ci].name + ' · tapınak', level:TEMPLE_LV[ci] || 0, seed:(S.rs || 7) * 31 + ci * 977 + S.day, cb:q => templeDone(ci, q) });
  }
  function templeDone(ci, q) {
    S.temple[ci] = true;
    let txt;
    if (q === 'miss') txt = 'Işık yolunu bulamadı. Rahipler başlarını salladı: "Belki bir dahaki sefere."';
    else {
      const k = pick(Object.keys(BLESS)), cash = q === 'perfect' ? 300 : 120; S.bless = k; S.cash += cash; addTrust(S, ci, q === 'perfect' ? 180 : 80); S.stats.temples = (S.stats.temples || 0) + 1;
      txt = (q === 'perfect' ? 'Kusursuz! ' : '') + 'Heykel parladı, rahipler adağını kabul etti: +' + fmt(cash) + ' deben ve ' + BLESS[k].name + ' (' + BLESS[k].desc + ')';
    }
    S.log.push({ d:S.day, t:TEMPLES[ci] + ': ' + txt });
    save(); renderAll(); play(q === 'miss' ? 'no' : 'fanfare');
    openSheet('<div class="out icons">' + outHtml({ kind:q === 'miss' ? 'info' : 'good', title:TEMPLES[ci], text:txt }) + '</div><div class="row"><button class="btn go1" type="button" data-close data-first>Tamam</button></div>', false, 'tdone');
  }
