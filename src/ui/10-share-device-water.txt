  /* ================= water with real light: a WebGL band under the harbour painting ================= */
  const GLW = (() => {
    let cv = null, gl = null, prog = null, tex = null, raf = 0, on = false, t0 = performance.now(), U = {}, cur = null, pal = null, host = null, ready = false;
    const VS = 'attribute vec2 p;varying vec2 v;void main(){v=p*0.5+0.5;gl_Position=vec4(p,0.0,1.0);}';
    const FS = 'precision mediump float;varying vec2 v;uniform sampler2D s;uniform float t;uniform vec3 c0;uniform vec3 c1;uniform vec3 c2;uniform float sx;uniform float nt;uniform float wt;uniform float wh;' +
      'void main(){float d=1.0-v.y;' +
      'float wave=sin(v.x*38.0+t*1.5+d*11.0)*0.0055+sin(v.x*15.0-t*1.05)*0.0035;' +
      'float wy=sin(d*28.0+t*1.9+v.x*6.0)*0.009*(1.0-d);' +
      'vec2 src=vec2(v.x+wave,wt-d*wh+wy);' +
      'vec3 refl=texture2D(s,src).rgb;' +
      'vec3 water=mix(c0,mix(c1,c2,smoothstep(0.15,1.0,d)),smoothstep(0.0,0.22,d));' +
      'float fres=pow(1.0-d,2.0);' +
      'vec3 col=mix(water,refl,0.42*fres+0.10);' +
      'float gx=exp(-pow((v.x-sx)*6.5,2.0));' +
      'float sp=pow(max(0.0,sin(v.x*110.0+t*2.8+d*36.0)*sin(d*80.0-t*2.1+v.x*9.0)),7.0);' +
      'col+=vec3(1.0,0.95,0.78)*gx*sp*(0.95-d*0.75)*(0.5+0.5*(1.0-nt));' +
      'col+=vec3(pow(max(0.0,sin(d*70.0+v.x*10.0+t*1.2)),16.0)*0.07*(1.0-d));' +
      'col=mix(col,col*vec3(0.75,0.72,0.9),nt*0.35);' +
      'gl_FragColor=vec4(col,1.0);}';
    const hex3 = h => { const a = hex2(h); return [a[0] / 255, a[1] / 255, a[2] / 255]; };
    function init(container) {
      host = container;
      if (cv) return true;
      try {
        cv = document.createElement('canvas'); cv.className = 'glw'; cv.setAttribute('aria-hidden', 'true');
        gl = cv.getContext('webgl', { antialias:false, alpha:false, premultipliedAlpha:false, powerPreference:'low-power' });
        if (!gl) { cv = null; return false; }
        const sh = (ty, src) => { const o = gl.createShader(ty); gl.shaderSource(o, src); gl.compileShader(o); if (!gl.getShaderParameter(o, gl.COMPILE_STATUS)) throw new Error(gl.getShaderInfoLog(o)); return o; };
        prog = gl.createProgram(); gl.attachShader(prog, sh(gl.VERTEX_SHADER, VS)); gl.attachShader(prog, sh(gl.FRAGMENT_SHADER, FS)); gl.linkProgram(prog);
        if (!gl.getProgramParameter(prog, gl.LINK_STATUS)) throw new Error('link');
        gl.useProgram(prog);
        const buf = gl.createBuffer(); gl.bindBuffer(gl.ARRAY_BUFFER, buf); gl.bufferData(gl.ARRAY_BUFFER, new Float32Array([-1, -1, 1, -1, -1, 1, 1, 1]), gl.STATIC_DRAW);
        const loc = gl.getAttribLocation(prog, 'p'); gl.enableVertexAttribArray(loc); gl.vertexAttribPointer(loc, 2, gl.FLOAT, false, 0, 0);
        ['s', 't', 'c0', 'c1', 'c2', 'sx', 'nt', 'wt', 'wh'].forEach(n => { U[n] = gl.getUniformLocation(prog, n); });
        tex = gl.createTexture(); gl.bindTexture(gl.TEXTURE_2D, tex);
        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE); gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR); gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
        return true;
      } catch (e) { cv = null; gl = null; return false; }
    }
    /* the painting is rasterised once per scene and used as the mirror */
    /* the painting's frame and its water band, in its own units (old paintings: 900x400, water 250-400) */
    let VB = [0, 0, 900, 400], WB = [250, 400];
    async function set(svgStr, P) {
      if (!init(host || $('#port')) || !svgStr) return;
      const key = svgStr.length + ':' + svgStr.slice(0, 200);
      pal = P; cur = key; if (!ready) host.classList.remove('gl');
      const vm = svgStr.match(/viewBox="([-\d. ]+)"/), wm = svgStr.match(/data-water="([-\d.]+)[ ,]+([-\d.]+)"/);
      VB = vm ? vm[1].trim().split(/\s+/).map(Number) : [0, 0, 900, 400]; WB = wm ? [+wm[1], +wm[2]] : [250, 400];
      const ri = svgStr.indexOf('<!--r-->'); if (ri > 0) svgStr = svgStr.slice(0, ri) + '</svg>';
      let img; try { img = await sceneToImage(svgStr.replace(/ class="(bob|walk|walk2|sway|sway2|row|pennant|cloud|glint|flow|felucca|bird|sunr|ripple|bob2|smoke|flag|splash)"/g, '').replace(/ class="(bob|bob spot)" data-spot="ship"/g, ''), 1200, Math.round(1200 * VB[3] / VB[2])); } catch (e) { return; }
      if (cur !== key || !gl) return;
      gl.bindTexture(gl.TEXTURE_2D, tex);
      try { gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGB, gl.RGB, gl.UNSIGNED_BYTE, img); } catch (e) { return; }
      if (!cv.parentNode) host.insertBefore(cv, host.firstChild);
      ready = true; host.classList.add('gl'); layout(); start();
    }
    function layout() {
      if (!cv || !host) return;
      const W = host.clientWidth, H = host.clientHeight; if (!W || !H) return;
      const sv = host.querySelector('.pscene svg') || host.querySelector('svg'), hr = host.getBoundingClientRect(), dpr = Math.min(1.5, window.devicePixelRatio || 1);
      let a = null, b = null; if (sv) { a = vbPt(sv, VB[0], WB[0]); b = vbPt(sv, VB[0] + VB[2], WB[1]); }
      if (!a || !b) { const sc = Math.max(W / VB[2], H / VB[3]); a = [hr.left + (W - VB[2] * sc) / 2, hr.top + H - (VB[3] - WB[0]) * sc]; b = [a[0] + VB[2] * sc, a[1] + (WB[1] - WB[0]) * sc]; }
      cv.style.left = (a[0] - hr.left) + 'px'; cv.style.top = (a[1] - hr.top) + 'px'; cv.style.width = (b[0] - a[0]) + 'px'; cv.style.height = (b[1] - a[1]) + 'px';
      const w = Math.round((b[0] - a[0]) * dpr), h = Math.round((b[1] - a[1]) * dpr);
      if (cv.width !== w || cv.height !== h) { cv.width = w; cv.height = h; }
    }
    function frame() {
      raf = 0;
      if (!on || !gl || !ready || host.hidden || document.hidden) return;
      if (!cv.parentNode) return;
      gl.viewport(0, 0, cv.width, cv.height);
      gl.uniform1f(U.t, (performance.now() - t0) / 1000);
      const c = pal.water.map(hex3); gl.uniform3fv(U.c0, c[0]); gl.uniform3fv(U.c1, c[1]); gl.uniform3fv(U.c2, c[2]);
      gl.uniform1f(U.sx, ((pal.sunX || 450) - VB[0]) / VB[2]); gl.uniform1f(U.nt, pal.stars || 0);
      gl.uniform1f(U.wt, (WB[0] - VB[1]) / VB[3]); gl.uniform1f(U.wh, (WB[1] - WB[0]) / VB[3] * 0.9);
      gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4);
      if (!reduce) raf = requestAnimationFrame(frame);
    }
    function start() { on = true; if (!raf) raf = requestAnimationFrame(frame); }
    function stop() { on = false; }
    function hide() { on = false; ready = false; cur = null; if (host) host.classList.remove('gl'); if (cv && cv.parentNode) cv.parentNode.removeChild(cv); }
    document.addEventListener('visibilitychange', () => { if (!document.hidden && on) start(); });
    return { set, layout, start, stop, hide, active:() => ready && on };
  })();
