  function sumAdvice(r, st) {
    if ((st.interest || 0) > Math.max(800, Math.abs(r.score) * 0.3)) return 'En büyük gider faiz oldu. Bir dahaki sefere Mennefer\'e uğrayıp borcu erken kapatmayı dene.';
    if ((st.lostVal || 0) > 600) return 'Nehirde çok yük kaybettin. Sakin su ve tapınak duraklarıyla gövdeyi koru; tersanede gövdeyi güçlendir.';
    if (st.trips <= 3) return 'Az yolculuk yaptın. Kısa rotalarla daha sık al-sat yapmak çoğu zaman daha çok kazandırır.';
    if (!st.haggles) return 'Hiç pazarlık kazanmadın. Her şehirde günde bir kez dene: iyi bir pazarlık bütün alış satışı iyileştirir.';
    if (!st.orders) return 'Hiç sipariş teslim etmedin. Siparişler para ve tüccar güveni getirir.';
    return r.beat ? 'Sethi\'yi geçtin. Bir sonraki hedef: Nil\'in Efendisi için 16.000 deben.' : 'Sethi\'yi geçmek için değerli malları kıt oldukları şehirlerden topla.';
  }
