  /* ================= start ================= */
  function start(data) {
    bootDone();
    let saved = data && restoreSave(data.S), fromHot = !!saved, unreadable = false, hadQ = !!(data && saveObject(data.S) && 'quest' in data.S);
    if (!saved) {
      const raw = lsGet(KEY);
      if (raw != null) {
        try { const p = JSON.parse(raw); saved = restoreSave(p); if (saved) hadQ = 'quest' in p; } catch (e) {}
        if (!saved) {
          unreadable = true; recoveryRaw = raw; recoveryKey = KEY + '-recovery-' + Date.now();
          while (lsGet(recoveryKey) != null) recoveryKey += '-1';
        }
      }
    }
    if (saved) {
      S = saved; RS = S.rs == null ? null : S.rs;
      /* saves from before the story arcs get one offered, if the run is young enough */
      if (!hadQ && !S.over && S.day <= 6 && has(S, 'quest')) S.quest = { id:ARCS[meta.runs % ARCS.length], step:0, st:'offer', by:0, cargo:0, f:{}, log:[] };
      sel = data && data.sel != null ? data.sel : null; qty = data && data.qty ? data.qty : 1;
      line = pickLine(); renderAll();
      if (fromHot) { if (S.voyage) resumeVoyage(); else if (S.pendingEvent && eventDef(S, S.pendingEvent)) showEvent(); return; }
      if (S.over) { showEnd(); return; }
      showTitle();
    } else {
      S = newGame({ perks:Object.assign({}, meta.lv), feat:featFor(meta.runs), arc:featFor(meta.runs).quest ? ARCS[meta.runs % ARCS.length] : null }); line = pickLine(); const stored = save(); renderAll(); showTitle();
      if (unreadable) setTimeout(() => toast(stored ? 'Önceki kayıt açılamadı. Eski kayıt verisi korunuyor; yeni bir sefer hazırladık.' : 'Önceki kayıt açılamadı ve yeni sefer kaydedilemedi. Eski kayıt verisi korunuyor; bu sekmeyi kapatma.'), 0);
    }
  }
  try { window.claude && window.claude.hot && window.claude.hot.snapshot && window.claude.hot.snapshot(() => ({ S, sel, qty })); } catch (e) {}
  setTimeout(() => { if (Snd.pref.music) Snd.prefetch(); }, 60);
