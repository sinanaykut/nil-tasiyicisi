  function crisisList() {
    if (!S || !has(S, 'crisis')) return [];
    return (S.crises || []).filter(c => S.day >= c.from - 1).map(c => Object.assign({ state:S.day < c.from ? 'soon' : S.day > c.until ? 'past' : 'now' }, c));
  }
  function crisisLine(c) {
    const d = CRISES[c.k];
    if (d.close) return S.day <= c.closeUntil ? c.from + '–' + c.closeUntil + '. gün kapalı, sonra ' + c.until + '. güne kadar her mal %30 pahalı' : c.until + '. güne kadar her mal %30 pahalı';
    return gName(d.g) + ' ' + (d.mult > 1 ? '×' + d.mult.toFixed(1).replace('.', ',') + ' pahalı' : 'yarı fiyatına') + ' · ' + c.from + '–' + c.until + '. gün';
  }
