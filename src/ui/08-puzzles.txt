  function liarPuzzle() {
    const names = ['Neb', 'Huy', 'Sitre'];
    const kinds = [
      (j) => ({ txt:names[j] + ' masum.', ok:c => c !== j }),
      (j) => ({ txt:'Kalpazan ' + names[j] + '.', ok:c => c === j }),
      (j) => ({ txt:names[j] + ' doğruyu söylüyor.', ok:c => c !== j }),
      (j) => ({ txt:names[j] + ' yalan söylüyor.', ok:c => c === j })
    ];
    for (let tries = 0; tries < 400; tries++) {
      const st = [0, 1, 2].map(i => { const others = [0, 1, 2].filter(x => x !== i), j = others[Math.floor(Math.random() * 2)]; return kinds[Math.floor(Math.random() * kinds.length)](j); });

      if (sols.length === 1) return { names, st, c:sols[0] };
    }
    return { names, st:[kinds[1](1), kinds[1](0), kinds[1](1)], c:1 };
  }

  function balSolve(T) { const d = []; let n = T; for (let k = 0; k < 4; k++) { let r = ((n % 3) + 3) % 3; if (r === 2) r = -1; d.push(r); n = (n - r) / 3; } return d.map(x => x === 1 ? 1 : x === -1 ? 2 : 0); }
  function pzBalance(o) {

  function balStart() {
    if (!PZ || PZ.kind !== 'bal' || PZ.started) return;
    PZ.started = true; PZ.t0 = performance.now();
    const intro = $('#pzIntro'); if (intro) intro.hidden = true;
    const how = document.querySelector('.balhow'); if (how) how.classList.add('mini');
    document.querySelectorAll('[data-pzw]').forEach(b => { b.disabled = false; });
    const go = document.querySelector('[data-act="pzBalGo"]'); if (go) go.remove();
    const hb = document.querySelector('[data-act="pzHint"]'); if (hb) hb.disabled = false;
    const f = document.querySelector('[data-pzw="3"][data-pos="1"]'); if (f) f.focus({ preventScroll:true });
    const tok = PZ.tok;
    const tick = () => {
      if (!PZ || PZ.tok !== tok || PZ.done) return;
      const el = (performance.now() - PZ.t0) / 1000, bar = $('#pzTime');
      if (bar) { bar.style.width = Math.max(0, 100 - el / PZ.lim * 100) + '%'; bar.classList.toggle('late', el > PZ.lim * 0.7); }
      if (el >= PZ.lim) { PZ.done = true; const cb = PZ.cb; meta.balN = (meta.balN || 0) + 1; saveMeta(); toast('Süre doldu: tüccar teraziyi kaldırdı.'); pzClear(); play('bad'); cb('miss'); return; }
      PZ.timers.push(setTimeout(tick, 100));
    };
    tick();
  }
  const balSums = () => ({ L:PZ.w.reduce((a, p, k) => a + (p === 2 ? BALV[k] : 0), PZ.T), R:PZ.w.reduce((a, p, k) => a + (p === 1 ? BALV[k] : 0), 0) });
  function balDraw() {
    if (!PZ || PZ.kind !== 'bal') return;
    const { L, R } = balSums(), left = [{ k:'g', v:PZ.T }].concat(PZ.w.map((p, k) => p === 2 ? { k:'w', v:BALV[k] } : null).filter(Boolean)), right = PZ.w.map((p, k) => p === 1 ? { k:'w', v:BALV[k] } : null).filter(Boolean);
    tiltTo($('#pzBal'), left, right, Math.max(-1, Math.min(1, (R - L) / 10)));

    if (rd) rd.innerHTML = '<span class="side">Sol (mal tarafı)<b>' + L + '</b></span><span class="diff' + (L === R ? ' ok' : '') + '">' + (L === R ? '⚖ Dengede!' : R < L ?   }
  function balPress(k, pos) {
