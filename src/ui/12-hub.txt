  function placeHold(r, sr) {
    /* v31: the hold sits in the picture's corner by the boat, never over the town */
    const el = $('#hold'); if (!el) return;
    el.style.left = ''; el.style.top = ''; el.style.bottom = ''; el.dataset.side = ''; el.classList.add('corner'); return;
    if (!r) { el.style.left = '10px'; el.style.top = ''; el.style.bottom = '10px'; return; }
    el.style.bottom = '';
    const w = el.offsetWidth, h = el.offsetHeight, W = sr.width, H = sr.height, cx = (r.left + r.right) / 2;
    /* other things in the picture the box should not hide: places, their tags, the cartouche */
    const avoid = [];
    document.querySelectorAll((town3On() ? '#port .v3t-l:not([hidden]):not(.ship)' : '#pscene [data-spot]:not([data-spot="ship"]), #spots [data-spot]') + ', #cart, #goal:not([hidden])').forEach(e => { const b = e.getBoundingClientRect(); if (b.width) avoid.push([b.left - sr.left, b.top - sr.top, b.right - sr.left, b.bottom - sr.top, e.closest('#port') ? 0.35 : 1]); });
    const cost = (x, y) => avoid.reduce((a, [l, t, rr, bb, k]) => a + k * Math.max(0, Math.min(x + w, rr) - Math.max(x, l)) * Math.max(0, Math.min(y + h, bb) - Math.max(y, t)), 0);
    const cands = [[cx - w / 2, r.bottom + 2, 'b'], [r.left - w - 4, r.bottom - h - 4, 'l'], [r.right + 4, r.bottom - h - 4, 'r'], [cx - w / 2, r.top - h - 4, 't'], [r.left - w - 4, (r.top + r.bottom) / 2 - h / 2, 'l'], [r.right + 4, (r.top + r.bottom) / 2 - h / 2, 'r']];
    let best = null;
    cands.forEach(([x0, y0, sd], k) => {
      const x1 = Math.max(6, Math.min(W - w - 6, x0)), y1 = Math.max(56, Math.min(H - h - 6, y0));
      /* moved far from where it wanted to be = it covers the ship: punish */
      const drift = Math.abs(x1 - x0) + Math.abs(y1 - y0), c = cost(x1, y1) + drift * h * 1.5 + k * 40;
      if (!best || c < best.c) best = { x:x1, y:y1, side:sd === 't' ? 'tp' : sd, c };
    });
    const x = best.x, y = best.y, side = best.side;
    el.style.left = Math.round(x) + 'px'; el.style.top = Math.round(y) + 'px'; el.dataset.side = side;
    el.style.setProperty('--ax', Math.round(Math.max(14, Math.min(w - 14, cx - x))) + 'px');
  }
