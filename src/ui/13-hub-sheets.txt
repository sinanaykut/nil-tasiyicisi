      if (st / full < 0.4 || want > 1.25) return ['no', 'İsteksiz: ' + (st / full < 0.4 ? 'tezgâhı boşalıyor' : 'şehir bu malı çok arıyor')];
      return ['mid', 'Ne istekli ne isteksiz: yüzüne bak'];

  }

  function hagGo(gid, side, n) {
    /* the honest price of that many: the average of the normal deal, which already moves with every unit */
    const q = side === 'buy' ? quoteBuy(S, gid, n) : quoteSell(S, gid, n); n = q.n; if (!n) { play('no'); renderPanel(); return; }
    const g = gById(gid), unit = (side === 'buy' ? q.cost : q.rev) / n;
    const se = SEASONS[seasonIdx(S)].mult[gid] || 1, crisis = (S.mods || []).filter(m => m.city === S.city && (!m.g || m.g === gid) && S.day >= m.from && S.day <= m.until).reduce((a, m) => a * m.mult, 1);
    const o = { ci:S.city, g:gid, side, qty:n, unit, stock:S.stock ? S.stock[S.city][gid] : 0, stockFull:STOCK[gid] || 0, want:+((CITIES[S.city].mult[gid] || 1) * se * crisis).toFixed(3), trust:trustLv(S), mood:Math.max(-1, Math.min(1, moodBoost * 0.4 + (trustLv(S) - 1) * 0.2)), seed:((S.rs || 7) * 131 + S.city * 97 + S.day * 13 + GOODS.indexOf(g)) | 0, who:S.city };
    let done = false, st0 = null;
    try {
      haggleDeal(o, r => {
        if (done) return; done = true; r = r || {};
        /* walking away before a single offer costs nothing: the good can be haggled again today */
        const none = !r.done && !r.angry && st0 && !st0.round && !st0.praised && !st0.walked;
        if (!none) hgMark(S, gid, r);
        let res = null;
        if (r.done && r.mult > 0) res = side === 'buy' ? dealBuy(S, gid, n, unit * r.mult) : dealSell(S, gid, n, unit * r.mult);
        moodBoost = r.angry ? -1 : res && res.n ? 1 : 0;
        line = r.angry ? (FACE_LINES[S.city] || {}).walk || 'Yeter! Bugün başka pazarlık yok.' : res && res.n ? (FACE_LINES[S.city] || {}).ok || 'Anlaştık.' : pickLine();
        save(); closeSheet(); renderAll();
        if (res && res.n) { play(side === 'buy' ? 'buy' : 'sell'); toast((side === 'buy' ? res.n + ' ' + gName(gid) + ' aldın · −' + fmt(res.cost) : res.n + ' ' + gName(gid) + ' sattın · +' + fmt(res.rev)) + ' (birimi ' + fmt(res.unit) + ', olağan ' + fmt(Math.round(unit)) + ')'); trustToast(res.lvUp, S.city); pulseCargo(); }
        else if (r.angry) { play('bad'); toast(MERCHANT_NAMES[S.city] + ' küstü: bugün başka pazarlık yok.'); }
        else play('click');
      });
      st0 = PZ && PZ.kind === 'haggle' ? PZ.st : null;
    } catch (e) { done = true; showHaggle(); }
  }
