  /* ================= v45 quick tavern bets (owner: "every gambling game takes a while; there should be short ones too") =================
     Three 20–60 s bets next to the long board games, QK_MAX a day at each tavern (S.tq[city] = { d:day, n:bets }):
       asik    knucklebones duel: the host throws four bones (faces 1, 3, 4, 6) and throws his lowest again while he has less
               than 16; you throw four and, when you are not ahead, may throw one bone of your choice again. The higher total
               wins, a tie is the host's, four different faces (the "Hathor throw") beats everything. Player ≈ 47 % (qkSim).
       bakla   the shell game: the bean goes under the middle cup, the cups swap, tap the cup with the bean. The swaps get
               faster and more numerous while you keep winning and slower after losses (meta.qShell.r), so it settles
               near one win in two for any pair of eyes.
       tekcift odd or even with the four throw sticks: call it, throw, count the light sides; all dark is a "blind throw"
               and the host takes it. Odd 50 %, even 43.75 %.
     The money moves the moment the outcome is fixed (an interrupted animation never loses or doubles a bet); the HUD,
     the banner and the host's face follow when the animation shows it. Menu: quickMenuHtml(ci) in the tavern dialogue
     (ui/13 showTavern), the game: showQuick(id); debug fn.quickAuto({ fast }), fn.quickSim(id, n), fn.showQuick. */

  /* ---------- the rules as pure functions (the game and qkSim use the same ones) ---------- */
  const QK_BONE = [[1, 0.1], [3, 0.4], [4, 0.4], [6, 0.1]];
  function qkBoneRoll(r) { r = r == null ? qkR() : r; let a = 0; for (const [v, p] of QK_BONE) { a += p; if (r < a) return v; } return 6; }
  const qkHathor = b => b.length === 4 && b.slice().sort((x, y) => x - y).join() === '1,3,4,6';
  const qkScore = b => qkHathor(b) ? 100 : b.reduce((a, x) => a + x, 0);
  /* the host's throw: four bones, the lowest thrown again while the total is under 16 (once) */
  function qkHostThrow() {
    const b = [0, 1, 2, 3].map(() => qkBoneRoll()), first = b.slice();
    let re = -1;
    if (!qkHathor(b) && qkScore(b) < 16) { re = b.indexOf(Math.min(...b)); b[re] = qkBoneRoll(); }
    return { first, b, re };
  }
  /* which bone a careful player throws again: the one with the best chance to pass the host */
  function qkBestRe(mine, host) {
    const H = qkScore(host); let best = -1, bp = -1;
    mine.forEach((_, i) => { let p = 0; QK_BONE.forEach(([v, q]) => { const nb = mine.slice(); nb[i] = v; if (qkScore(nb) > H) p += q; }); if (p > bp + 1e-9) { bp = p; best = i; } });
    return { i:best, p:bp };
  }
  const qkSticks = () => [0, 1, 2, 3].map(() => qkR() < 0.5 ? 1 : 0);
  /* tek/çift: light sides counted; all dark = blind throw (the host's) */
  function qkTekWin(call, f) { const n = f.reduce((a, x) => a + x, 0); if (!n) return false; return (n % 2 === 1) === (call === 'tek'); }
  /* the simulated win rate (debug / QA): asik with careful play, tekcift for one call */
  function qkSim(id, n, call) {
    n = n || 20000; let w = 0;
    for (let k = 0; k < n; k++) {
      if (id === 'asik') { const h = qkHostThrow(), m = [0, 1, 2, 3].map(() => qkBoneRoll()); if (qkScore(m) > qkScore(h.b)) { w++; continue; } const r = qkBestRe(m, h.b); m[r.i] = qkBoneRoll(); if (qkScore(m) > qkScore(h.b)) w++; }
      else if (id === 'tekcift') { if (qkTekWin(call || (k % 2 ? 'tek' : 'cift'), qkSticks())) w++; }
      else if (id === 'bakla') { if (Math.floor(qkR() * 3) === 0) w++; }   /* a blind guess; the eyes do the rest */
    }
    return w / n;
  }

  function qkAsikThrow() {
    if (!qkMine() || PZ.phase !== 'play') return;

      qkSum('p', PZ.m);

      if (M > H) { qkSettle(true); qkLater(() => qkShow(true, qkHathor(PZ.m) ? '<b>Hathor atışı!</b> Dört farklı yüz her şeyi yener.' : '<b>' + M + '</b> &gt; ' + H + ': geçtin!'), 450); return; }
      /* not ahead: one bone may be thrown again; the most hopeful one glows */

      if (best.p <= 0) { qkSettle(false); qkLater(() => qkShow(false, 'Hiçbir kemik seni öne geçiremezdi. <b>' + M + '</b> ≤ ' + H + '.'), 500); return; }

      els.forEach((e, k) => { e.disabled = false; e.classList.toggle('pick', k === best.i); });
      qkMsg((M === H ? '<b>Berabere</b>: beraberlik ev sahibinin. ' : '<b>' + M + '</b>, geride kaldın. ') + 'Bir kemiğe dokun, <b>bir kez</b> daha at.');
      qkBtns(qkBig('data-qk="re" data-first', 'Parlayanı at') + qkBig('data-qk="keep"', 'Atmadan bitir', 'plain'));
      qkFace('thinking');
      if (PZ.auto) qkLater(() => qkAsikRe(best.i), 250);
    });

  function qkSwap(n) {
    if (!qkMine() || PZ.phase !== 'play') return;
    if (n >= PZ.swaps) { qkPickTime(); return; }
    /* two slots trade places: one cup arcs over the front, the other slides behind */
    const a = Math.floor(qkR() * 3); let b = Math.floor(qkR() * 2); if (b >= a) b++;
    const ca = PZ.slot.indexOf(a), cb = PZ.slot.indexOf(b);
    const box = $('#qkCups'), w = box ? box.clientWidth || 320 : 320, d = qkT(PZ.dur);
    [[ca, a, b, -1], [cb, b, a, 1]].forEach(([c, from, to, dir]) => {
      const e = $('#qkC' + c); if (!e) return;
      const x0 = qkCupX(w, from), x1 = qkCupX(w, to), lift = 16 + Math.abs(to - from) * 10;
      e.style.zIndex = dir < 0 ? 3 : 1;
      if (e.animate && d > 0) {
        const an = e.animate([{ transform:'translate(' + x0 + 'px,0) scale(1)' }, { transform:'translate(' + ((x0 + x1) / 2) + 'px,' + (dir * lift * 0.35) + 'px) scale(' + (dir < 0 ? 1.06 : 0.94) + ')', offset:0.5 }, { transform:'translate(' + x1 + 'px,0) scale(1)' }], { duration:d, easing:'ease-in-out', fill:'forwards' });
        an.onfinish = () => { e.style.transform = 'translate(' + x1 + 'px,0)'; try { an.cancel(); } catch (er) {} };
      } else e.style.transform = 'translate(' + x1 + 'px,0)';
    });
    PZ.slot[ca] = b; PZ.slot[cb] = a;
    play('click');
    qkLater(() => qkSwap(n + 1), PZ.dur + 40);
  }

    const sh = $('#sheet');

      sh.addEventListener('click', e => {
        const tv = e.target.closest && e.target.closest('[data-tvq]');
        if (tv) { play('click'); showQuick(tv.getAttribute('data-tvq')); return; }
        if (!qkMine()) return;
        const b = e.target.closest && e.target.closest('button');
        if (!b || b.disabled) return;
        const q = b.getAttribute('data-qk'), stv = b.getAttribute('data-qkst'), bn = b.getAttribute('data-qkb'), cup = b.getAttribute('data-qkc');
        if (stv && PZ.phase === 'bet') { PZ.pick = +stv; play('click'); qkPaintChips(); return; }
        if (q === 'menu') { pzClear(); play('click'); if (typeof showTavern === 'function') showTavern(); return; }
        if (q === 'again') { play('click'); qkAgain(); return; }
        if (PZ.id === 'asik') {
          if (q === 'go' && PZ.phase === 'bet') qkAsikGo();
          else if (q === 'throw') qkAsikThrow();
          else if (q === 're') qkAsikRe();
          else if (q === 'keep') qkAsikRe(-1);
          else if (bn != null && bn !== '' && PZ.phase === 'play' && !PZ.re && PZ.reIdx != null) {
            PZ.reIdx = +bn; play('click');

          }
        } else if (PZ.id === 'bakla') {
          if (q === 'go' && PZ.phase === 'bet') qkBaklaGo();
          else if (cup != null) qkPick(+cup);
        } else if ((q === 'tek' || q === 'cift') && PZ.phase === 'bet') qkTekGo(q);
      });
    }
  }
