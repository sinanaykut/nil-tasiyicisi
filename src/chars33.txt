    function faceNormals(pos, nrm, idx) {
      const n = pos.length / 3, nb = new Map(), out = Float32Array.from(nrm);
      for (let j = 0; j < idx.length; j += 3) for (let e = 0; e < 3; e++) { const a = idx[j + e], b = idx[j + (e + 1) % 3]; (nb.get(a) || nb.set(a, []).get(a)).push(b); }
      for (let it = 0; it < 3; it++) {
        const t = Float32Array.from(out);
        for (let i = 0; i < n; i++) { const L = nb.get(i); if (!L) continue; let x = 0, y = 0, z = 0; for (const k of L) { x += t[k * 3]; y += t[k * 3 + 1]; z += t[k * 3 + 2]; } out[i * 3] = t[i * 3] * 0.5 + x / L.length * 0.5; out[i * 3 + 1] = t[i * 3 + 1] * 0.5 + y / L.length * 0.5; out[i * 3 + 2] = t[i * 3 + 2] * 0.5 + z / L.length * 0.5; }
      }
      for (let i = 0; i < n; i++) {
        const x = pos[i * 3], y = pos[i * 3 + 1] - 1.628, z = pos[i * 3 + 2] + 0.012;
        if (y < -0.13) continue;
        let ex = x / 0.075, ey = y / 0.1, ez = z / 0.09; const l = Math.hypot(ex / 0.075, ey / 0.1, ez / 0.09) || 1;
        const k = 0.45 * Math.min(1, (y + 0.13) / 0.05);
        let nx = out[i * 3] * (1 - k) + ex / 0.075 / l * k, ny = out[i * 3 + 1] * (1 - k) + ey / 0.1 / l * k, nz = out[i * 3 + 2] * (1 - k) + ez / 0.09 / l * k;
        const m = Math.hypot(nx, ny, nz) || 1; out[i * 3] = nx / m; out[i * 3 + 1] = ny / m; out[i * 3 + 2] = nz / m;
      }
      return out;
    }
    function vnormals(pos, idx) {
      const n = new Float32Array(pos.length);
      for (let j = 0; j < idx.length; j += 3) {
        const a = idx[j] * 3, b = idx[j + 1] * 3, c = idx[j + 2] * 3;

        n[a] += nx; n[a + 1] += ny; n[a + 2] += nz; n[b] += nx; n[b + 1] += ny; n[b + 2] += nz; n[c] += nx; n[c + 1] += ny; n[c + 2] += nz;
      }
      for (let i = 0; i < n.length; i += 3) { const l = Math.hypot(n[i], n[i + 1], n[i + 2]) || 1; n[i] /= l; n[i + 1] /= l; n[i + 2] /= l; }

    function shell(hd, s) {

      });

    /* rest skeleton for a body type: bone heads moved by the trait deltas stored with the kit */
    function restPose(bw) {
      const T0 = getT(), K = KIT, D = (K.meta && K.meta.bd) || {}, nb = K.bones.length, out = K.bones.map(b => b.pos.slice());
      const dw = K.bones.map(() => [0, 0, 0]); let any = false;
      for (const k in bw) { const w = bw[k], d = D[k]; if (!w || !d) continue; for (const bn in d) { const i = K.bi[bn]; if (i == null) continue; any = true; dw[i][0] += d[bn][0] * w; dw[i][1] += d[bn][1] * w; dw[i][2] += d[bn][2] * w; } }
      if (!any) return out;
      /* local delta = parent rest rotation^-1 (child world delta - parent world delta) */
      const q = new T0.Quaternion(), v = new T0.Vector3(), s3 = new T0.Vector3(), p3 = new T0.Vector3();
      for (let i = 0; i < nb; i++) {
        const pi = K.bones[i].parent; if (pi < 0) { out[i][0] += dw[i][0]; out[i][1] += dw[i][1]; out[i][2] += dw[i][2]; continue; }
        K.bones[pi].world.decompose(p3, q, s3);
        v.set(dw[i][0] - dw[pi][0], dw[i][1] - dw[pi][1], dw[i][2] - dw[pi][2]).applyQuaternion(q.invert()).divide(s3);
        out[i][0] += v.x; out[i][1] += v.y; out[i][2] += v.z;
      }
      return out;
    }

    function sampleInto(st, name, t, w, first) {
      const C = KIT.clips[name] || KIT.clips.idle; if (!C) return;

      for (let b = 0; b < nb; b++) {
        if (FACEB[b]) continue;

        let x = Q[o0], y = Q[o0 + 1], z = Q[o0 + 2], q = Q[o0 + 3]; const x1 = Q[o1], y1 = Q[o1 + 1], z1 = Q[o1 + 2], q1 = Q[o1 + 3];
        const s = (x * x1 + y * y1 + z * z1 + q * q1) < 0 ? -1 : 1;
        x += (x1 * s - x) * a; y += (y1 * s - y) * a; z += (z1 * s - z) * a; q += (q1 * s - q) * a;
        const bq = B[b].quaternion;
        if (first) bq.set(x, y, z, q);
        else { const s2 = (bq.x * x + bq.y * y + bq.z * z + bq.w * q) < 0 ? -1 : 1; bq.set(bq.x + (x * s2 - bq.x) * w, bq.y + (y * s2 - bq.y) * w, bq.z + (z * s2 - bq.z) * w, bq.w + (q * s2 - bq.w) * w); }
        bq.normalize();
      }
      const hi = bi.hips, P = C.P, rp = KIT.bones[hi].pos, rr = st.rest[hi];
      const px = P[f0 * 3] + (P[f1 * 3] - P[f0 * 3]) * a - rp[0], py = P[f0 * 3 + 1] + (P[f1 * 3 + 1] - P[f0 * 3 + 1]) * a - rp[1], pz = P[f0 * 3 + 2] + (P[f1 * 3 + 2] - P[f0 * 3 + 2]) * a - rp[2];

      if (first) hp.set(rr[0] + px, rr[1] + py, rr[2] + pz);
      else hp.set(hp.x + (rr[0] + px - hp.x) * w, hp.y + (rr[1] + py - hp.y) * w, hp.z + (rr[2] + pz - hp.z) * w);
    }
