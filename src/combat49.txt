  /* Armed encounters: readable turns, with the same rules on the Nile and in the desert. */
  const AMBUSH_TELLS = [
    { id:'lunge', name:'Mızrağını doğrulttu', help:'Kalkanını kaldır. Darbeyi karşılayınca karşı saldırı fırsatı doğar.' },
    { id:'sweep', name:'Silahını yana açtı', help:'Yana çekil. Geniş savuruşu kalkanla durdurmak zordur.' },
    { id:'open', name:'Dengesini kaybetti', help:'Şimdi vurabilirsin. Hazır değilsen kalkanını kaldırıp nefeslen.' }
  ];
  function ambushNew(o) {
    const max = Math.max(1, Math.min(4, Math.floor(o.maxHp || 3)));
    const hp = Math.max(0, Math.min(max, Math.floor(o.hp == null ? max : o.hp)));
    return { hp, maxHp:max,
      enemy:3, maxEnemy:3, stamina:3, opening:false, turns:0, hits:0, done:hp <= 0, won:false,
      seed:(o.seed || 0) >>> 0, tell:0, last:'', text:'Silahını hazırla. Önce rakibinin hareketine bak.', hero:o.hero || '' };
  }
  function ambushTell(s) { return AMBUSH_TELLS[(s.seed + s.turns) % AMBUSH_TELLS.length]; }
  function ambushTurn(s, action) {
    if (s.done || !['strike', 'guard', 'dodge', 'retreat'].includes(action)) return false;
    if ((action === 'strike' || action === 'dodge') && s.stamina < 1) return false;
    const tell = ambushTell(s).id; s.last = action;
    let hurt = 0, dealt = 0;
    if (action === 'retreat') { s.done = true; s.text = 'Yükün için daha fazla canını tehlikeye atmadın.'; return true; }
    if (action === 'guard') {
      s.stamina = Math.min(3, s.stamina + 1);
      if (tell === 'lunge') { s.opening = true; s.text = 'Mızrağı kalkanınla saptırdın. Bir sonraki açıkta güçlü vurabilirsin.'; }
      else if (tell === 'sweep') { hurt = 1; s.opening = false; s.text = 'Geniş savuruş kalkanının yanından geldi. Bu harekette yana çekilmelisin.'; }
      else s.text = 'Kalkanının ardında nefeslendin. Gücün toparlandı.';
    } else if (action === 'dodge') {
      s.stamina--; s.opening = true;
      s.text = tell === 'open' ? 'Mesafeni korudun. Rakibinin bir sonraki açığını bekliyorsun.' : 'Silah yanından geçti. Dengen yerinde; karşı saldırıya hazırsın.';
    } else {
      s.stamina--; dealt = s.opening && tell === 'open' ? 2 : 1;
      if (tell !== 'open') hurt = 1;
      s.opening = false;
      s.text = hurt ? 'Vuruşun tuttu ama rakibinin saldırısına da açık kaldın.' : dealt === 2 ? 'Hazırladığın karşı saldırı tuttu! Rakip iki adım geri çekildi.' : 'Açığı gördün ve rakibi geri püskürttün.';
    }
    s.hits += hurt; s.hp = Math.max(0, s.hp - hurt); s.enemy = Math.max(0, s.enemy - dealt); s.turns++;
    if (s.hp <= 0 || s.enemy <= 0 || s.turns >= 12) {
      s.done = true; s.won = s.hp > 0 && s.enemy <= 0;
      s.text = s.won ? 'Yol açıldı. Saldırganlar geri çekiliyor!' : s.hp <= 0 ? 'Canın tükendi. Tayfan seni güvenli yere çekti.' : 'Mücadele uzadı. Tayfanı korumak için geri çekildin.';
    }
    return true;
  }

  let activeAmbushCombat = null;
  function stopAmbushCombat() { if (activeAmbushCombat) activeAmbushCombat.dispose(); }
  function startAmbushCombat(o, callback) {
    stopAmbushCombat();
    const host = o.host; if (!host) return null;
    const s = ambushNew(o), root = document.createElement('section');
    root.className = 'ambush49 t-' + (o.terrain === 'desert' ? 'desert' : 'river');
    root.setAttribute('aria-label', 'Silahlı mücadele'); host.appendChild(root);
    let dead = false;
    const hearts = (n, max) => '<span aria-label="' + n + ' / ' + max + ' can">' + '♥'.repeat(n) + '<i>' + '♡'.repeat(max - n) + '</i></span>';
    function render() {
      const tell = ambushTell(s);
      root.innerHTML = '<header><span class="ac-kicker">' + (o.terrain === 'desert' ? 'Kervanın savunması' : 'Güvertede karşılaşma') + '</span><h2>Yolunu koru</h2><p>Mızrak ve kalkan · ' + s.turns + '/12 hamle</p></header>' +
        '<div class="ac-arena a-' + s.last + '"><div class="ac-horizon"></div><div class="ac-fighter ac-you">' + ambushFigure(o.hero, false) + '<b>Sen</b>' + hearts(s.hp, s.maxHp) + '</div><div class="ac-fighter ac-enemy">' + ambushFigure('', true) + '<b>Saldırgan</b>' + hearts(s.enemy, s.maxEnemy) + '</div><div class="ac-dust"></div></div>' +
        '<div class="ac-info" aria-live="polite"><b>' + (s.done ? s.won ? 'Yol açıldı' : 'Geri çekildin' : tell.name) + '</b><p>' + (s.done ? s.text : tell.help) + '</p></div>' +
        '<p class="ac-response" role="status">' + (s.done ? '' : s.text) + '</p>' +
        (s.done ? '<button type="button" class="btn gold wide" data-ac="finish">Yola devam et</button>' :
          '<div class="ac-strength">Güç <span aria-label="' + s.stamina + ' / 3 güç">' + '●'.repeat(s.stamina) + '○'.repeat(3 - s.stamina) + '</span>' + (s.opening ? '<b>Karşı saldırı hazır</b>' : '<small>Kalkanını kaldırınca güç +1</small>') + '</div><div class="ac-actions">' +
          '<button type="button" data-ac="strike"' + (!s.stamina ? ' disabled' : '') + '><b>1 · Vur</b><small>Mızrak · 1 güç</small></button><button type="button" data-ac="guard"><b>2 · Kalkan</b><small>Doğrultulan mızrağa karşı</small></button><button type="button" data-ac="dodge"' + (!s.stamina ? ' disabled' : '') + '><b>3 · Yana çekil</b><small>Savuruşa karşı · 1 güç</small></button></div><button type="button" class="ac-retreat" data-ac="retreat">Geri çekil · yükün tehlikeye girer</button>');
    }
    function dispose() {
      if (dead) return; dead = true;
      root.removeEventListener('click', click); window.removeEventListener('keydown', key);
      root.remove(); if (host.classList.contains('ac49-overlay')) host.remove();
      document.body.classList.remove('armed49');
      if (activeAmbushCombat === ctl) activeAmbushCombat = null;
    }
    function act(action) {
      if (dead) return;
      if (action === 'finish') {
        if (!s.done) return;
        const result = { won:s.won, hits:s.hits, hp:s.hp, turns:s.turns };
        dispose(); callback(result); return;
      }
      if (!ambushTurn(s, action)) return;
      if (o.onChange) o.onChange({ hp:s.hp, hits:s.hits });
      play(s.done ? s.won ? 'good' : 'bad' : s.last === 'strike' ? 'hit' : 'tick'); render();
      const f = root.querySelector(s.done ? '[data-ac="finish"]' : '[data-ac="' + action + '"]:not(:disabled)');
      if (f) f.focus({ preventScroll:true });
    }
    function click(e) { const b = e.target.closest && e.target.closest('[data-ac]'); if (b) { e.preventDefault(); act(b.dataset.ac); } }
    function key(e) {
      if (e.repeat || !$('#veil').hidden || !['1', '2', '3'].includes(e.key) || /^(INPUT|TEXTAREA|SELECT)$/.test(e.target.tagName)) return;
      e.preventDefault(); act(['strike', 'guard', 'dodge'][+e.key - 1]);
    }
    const ctl = { dispose, peek:() => s, bot() { if (s.done) act('finish'); else act(ambushTell(s).id === 'lunge' ? 'guard' : ambushTell(s).id === 'sweep' ? s.stamina ? 'dodge' : 'guard' : s.stamina ? 'strike' : 'guard'); } };
    activeAmbushCombat = ctl; document.body.classList.add('armed49'); root.addEventListener('click', click); window.addEventListener('keydown', key);
    if (s.hp <= 0) { s.done = true; s.text = 'Tayfan seni güvenli yere taşıyor.'; }
    render(); const f = root.querySelector('button:not(:disabled)'); if (f) f.focus({ preventScroll:true });
    return ctl;
  }
  function riverAmbush() {
    const voyage = VG; if (!voyage) return;
    voyage.scene = 'ambush'; Sea.stop(); $('#voy').hidden = true; $('#sea').hidden = true;
    openSheet('<div class="sh-h"><h2 id="shTitle">Yolunu kestiler<small>Dar geçitte bir saldırgan teknesi</small></h2></div>' +
      '<div class="ac-choice"><div>' + ambushFigure('', true) + '</div><p>Güverteye yaklaşmaya çalışıyorlar. Tayfan küreklerin başında; mızrağın ve kalkanın elinin altında.</p></div>' +
      '<div class="row"><button type="button" class="btn gold" data-ambush-action="fight">Silahlı mücadele</button><button type="button" class="btn" data-ambush-action="run">Kürekle uzaklaş</button></div><p class="muted">Mücadelede rakibin hareketini okuyarak hamle yaparsın. Yaralanırsan canın azalır; teknenin gövdesi etkilenmez.</p>', true, 'ambush');
    const choose = action => {
      if (VG !== voyage || voyage.scene !== 'ambush') return;
      voyage.scene = 'combat'; closeSheet();
      if (action === 'run') { playScene('chase'); return; }
      const host = document.createElement('div'); host.className = 'ac49-overlay'; $('#stage').appendChild(host);
      Snd.mode('chase');
      startAmbushCombat({ terrain:'river', host, hp:voyage.hh, maxHp:voyage.hhMax, hero:S.hero, seed:voyage.plan.seed + voyage.layer * 97, onChange:res => { voyage.hh = res.hp; renderPlate(); } }, res => {
        if (VG !== voyage) return;
        voyage.hh = res.hp; voyage.acc.fought = true; voyage.acc.caught = !res.won; voyage.acc.escaped = res.won;
        voyage.inChase = false; voyage.scene = null;
        const text = res.won ? 'Mızrak ve kalkanla saldırganları geri püskürttün. Yükün güvende.' : 'Geri çekildin. Saldırganlar yolculuğun sonunda yükünden pay alacak.';
        vnote(text + (res.hits ? ' Mücadelede ' + res.hits + ' can kaybettin.' : ''), res.won ? 'good' : 'bad');
        Snd.mode('sea'); renderPlate();
        if (!faintCheck()) advance();
      });
      renderPanel();
    };
    document.querySelectorAll('#sheet [data-ambush-action]').forEach(b => { b.onclick = () => choose(b.dataset.ambushAction); });
  }

    const N = window.__nil; if (N && N.fn) Object.assign(N.fn, {
      combatState:() => activeAmbushCombat && activeAmbushCombat.peek(),
      combatBot:() => { if (activeAmbushCombat) activeAmbushCombat.bot(); },
      combatAction:action => { const b = document.querySelector('.ambush49 [data-ac="' + action + '"]'); if (b && !b.disabled) b.click(); }
    });
  }, 0);
