
  /* ================= temple room: guide the sunbeam to the golden statue with bronze mirrors (owner: bulmaca) ================= */
  /* level table: grid size, path turns, rotatable decoys, fixed path mirrors, fixed decoys, pillars, prisms, minimum ideal taps */
  const TM_LV = [
    { n:5, k:[3, 3], dec:1, fixP:0, fixD:0, pil:0, pri:0, par:2 },
    { n:5, k:[3, 4], dec:1, fixP:1, fixD:0, pil:1, pri:0, par:2 },
    { n:6, k:[4, 5], dec:2, fixP:1, fixD:1, pil:2, pri:0, par:3 },
    { n:6, k:[5, 6], dec:2, fixP:1, fixD:1, pil:2, pri:1, par:3 },
    { n:7, k:[6, 7], dec:3, fixP:1, fixD:1, pil:3, pri:1, par:4 }
  ];
  const TM_C = 60, TM_W = 26;
  function tmRng(seed) {
    let a = 0;
    if (typeof seed === 'number' && isFinite(seed)) a = Math.floor(seed * 9973) ^ 0x9E3779B9;
    else if (seed != null) { const s = String(seed); a = 2166136261; for (let i = 0; i < s.length; i++) a = Math.imul(a ^ s.charCodeAt(i), 16777619); }
    else a = Math.floor(Math.random() * 4294967296);
    a >>>= 0;
    return () => { a = (a + 0x6D2B79F5) >>> 0; let t = a; t = Math.imul(t ^ (t >>> 15), t | 1); t ^= t + Math.imul(t ^ (t >>> 7), t | 61); return ((t ^ (t >>> 14)) >>> 0) / 4294967296; };
  }
  /* follow the light: returns the drawn segments and whether the statue is reached. o:0 = "\", o:1 = "/" */
  function tmTrace(R) {
    const n = R.n, segs = [], seen = new Set(), E = R.entry, touched = new Set();
    const q = [[E.x - E.dx, E.y - E.dy, E.dx, E.dy, true]];
    let hit = false;
    while (q.length && segs.length < 80) {
      const [sx, sy, dx, dy, first] = q.pop(); let x = sx, y = sy, end = 'wall';
      for (;;) {
        x += dx; y += dy;
        if (x < 0 || y < 0 || x >= n || y >= n) { end = 'wall'; break; }
        const key = (y * n + x) * 4 + (dx === 1 ? 0 : dx === -1 ? 1 : dy === 1 ? 2 : 3);
        if (seen.has(key)) { end = 'loop'; break; }
        seen.add(key);
        if (x === R.target.x && y === R.target.y) { end = 'target'; hit = true; break; }
        const it = R.g[y * n + x];
        if (!it) continue;
        if (it.t === 'p') { end = 'pillar'; break; }
        touched.add(y * n + x);
        if (it.t === 'm') { end = 'turn'; if (it.o === 0) q.push([x, y, dy, dx]); else q.push([x, y, -dy, -dx]); break; }
        if (it.t === 'x') { end = 'turn'; q.push([x, y, -dy, dx], [x, y, dy, -dx]); break; }
      }
      segs.push({ sx, sy, x, y, dx, dy, end, first:!!first });
    }
    return { segs, hit, touched };
  }
  /* brute force every orientation of the rotatable mirrors: the solution nearest to the current state */
  function tmNearest(R) {
    const M = R.rot, cur = M.map(m => m.o);
    let best = null, bd = 99;
    for (let mask = 0; mask < (1 << M.length); mask++) {
      let d = 0; for (let i = 0; i < M.length; i++) { M[i].o = (mask >> i) & 1; if (M[i].o !== cur[i]) d++; }
      if (d >= bd) continue;
      if (tmTrace(R).hit) { bd = d; best = M.map(m => m.o); }
    }
    M.forEach((m, i) => { m.o = cur[i]; });
    return best ? { d:bd, sol:best } : null;
  }
  /* build the solution path first (entry gap → mirrors → statue), then add decoys and scramble */
  function tmBuild(level, seed) {
    const L = TM_LV[Math.max(0, Math.min(4, level | 0))], rnd = tmRng(seed), ri = k => Math.floor(rnd() * k);
    for (let tries = 0; tries < 2000; tries++) {
      const n = L.n, side = ri(4), k = 1 + ri(n - 2);
      const E = [{ x:k, y:0, dx:0, dy:1 }, { x:n - 1, y:k, dx:-1, dy:0 }, { x:k, y:n - 1, dx:0, dy:-1 }, { x:0, y:k, dx:1, dy:0 }][side];
      E.side = side; E.k = k;
      const K = L.k[0] + ri(L.k[1] - L.k[0] + 1), used = new Set(), path = [];
      let x = E.x - E.dx, y = E.y - E.dy, dx = E.dx, dy = E.dy, ok = true;
      const free = (a, b) => a >= 0 && b >= 0 && a < n && b < n && !used.has(b * n + a);
      for (let m = 0; m <= K; m++) {
        let max = 0; while (free(x + dx * (max + 1), y + dy * (max + 1))) max++;
        if (max < 1) { ok = false; break; }
        const len = 1 + ri(Math.min(max, m === 0 ? 3 : 4));
        for (let i = 1; i <= len; i++) used.add((y + dy * i) * n + x + dx * i);
        x += dx * len; y += dy * len;
        if (m < K) {
          const t = rnd() < 0.5, ndx = t ? -dy : dy, ndy = t ? dx : -dx;
          path.push({ x, y, sol:(ndx === dy && ndy === dx) ? 0 : 1 });
          dx = ndx; dy = ndy;
        }
      }
      if (!ok) continue;
      if (x === E.x && y === E.y) continue;
      /* the statue should not sit against the entry wall right beside the gap */
      if (Math.abs(x - E.x) + Math.abs(y - E.y) < 2) continue;
      const R = { n, level, entry:E, target:{ x, y }, g:Array(n * n).fill(null), rot:[], items:[] };
      const put = (cx, cy, it) => { it.x = cx; it.y = cy; R.g[cy * n + cx] = it; R.items.push(it); return it; };
      const pm = path.map(p => put(p.x, p.y, { t:'m', o:p.sol, sol:p.sol, fix:false, path:true }));
      /* a prism on the path splits the light both ways; one branch carries on */
      const mid = pm.slice(1, -1);
      for (let i = 0; i < L.pri && mid.length; i++) { const it = mid.splice(ri(mid.length), 1)[0]; it.t = 'x'; delete it.o; }
      const cand = pm.filter(p => p.t === 'm');
      for (let i = 0; i < L.fixP && cand.length > 3; i++) cand.splice(ri(cand.length), 1)[0].fix = true;
      const empty = []; for (let cy = 0; cy < n; cy++) for (let cx = 0; cx < n; cx++) if (!used.has(cy * n + cx)) empty.push([cx, cy]);
      const take = () => empty.length ? empty.splice(ri(empty.length), 1)[0] : null;
      let c;
      for (let i = 0; i < L.dec; i++) if ((c = take())) put(c[0], c[1], { t:'m', o:ri(2), sol:-1, fix:false, path:false });
      for (let i = 0; i < L.fixD; i++) if ((c = take())) put(c[0], c[1], { t:'m', o:ri(2), sol:-1, fix:true, path:false });
      for (let i = 0; i < L.pil; i++) if ((c = take())) put(c[0], c[1], { t:'p' });
      R.rot = R.items.filter(it => it.t === 'm' && !it.fix);
      const need = Math.min(L.par, cand.filter(p => !p.fix).length);
      for (let s = 0; s < 40; s++) {
        R.rot.forEach(m => { m.o = ri(2); });
        if (tmTrace(R).hit) continue;
        const nb = tmNearest(R);
        if (nb && nb.d >= need) { R.par = nb.d; R.rot.forEach((m, i) => { m.i = i; }); return R; }
      }
    }
    return null;
  }

  }
  /* one tap on a rotatable mirror: the same path for players, hints and the auto solver */
  function tmTap(i, viaHint) {
    if (!PZ || PZ.kind !== 'temple' || PZ.done) return false;
    const R = PZ.R, m = R.rot[i]; if (!m) return false;
    m.o ^= 1; m.ang += 90;
    if (viaHint) PZ.hint = true; else PZ.moves++;
    const g = document.querySelector('#tmI' + R.items.indexOf(m) + ' .tmrot'); if (g) g.style.transform = 'rotate(' + m.ang + 'deg)';
    play('tick');
    const tr = tmDraw(false);
    if (tr.hit) tmWin();
    return true;
  }
  function tmWin() {
    const R = PZ.R, tok = PZ.tok; PZ.done = true;
    const q = !PZ.hint && PZ.moves <= R.par + 2 ? 'perfect' : 'good';
    PZ.q = q;
    document.querySelectorAll('#sheet [data-tm],#sheet [data-tmf],#sheet [data-tmh],#sheet [data-tmq]').forEach(b => { b.disabled = true; });
    PZ.timers.push(setTimeout(() => {
      if (!PZ || PZ.tok !== tok) return;
      const room = $('#tmRoom'); if (room) room.classList.add('win');
      const bar = document.querySelector('#sheet .tmbar'); if (bar) bar.classList.add('won');
      const st = $('#tmSt'); if (st) st.innerHTML = '<b class="tmok">' + (q === 'perfect' ? 'Kusursuz! ' : '') + 'Işık heykele ulaştı!</b>';
      play('fanfare'); buzz('big');
    }, reduce ? 0 : 200));
    PZ.timers.push(setTimeout(() => tmFin(q), 1400));
  }
  function tmFin(q) {
    if (!PZ || PZ.kind !== 'temple' || PZ.fin) return;
    PZ.fin = true; const cb = PZ.cb; pzClear();
    try { if (typeof cb === 'function') cb(q); }
    finally { if (document.querySelector('#sheet .tmroom')) closeSheet(); }
  }
  function tmHint() {
    if (!PZ || PZ.kind !== 'temple' || PZ.done) return;
    const R = PZ.R, nb = tmNearest(R); if (!nb) return;
    const wrong = R.rot.filter((m, i) => m.o !== nb.sol[i]); if (!wrong.length) return;
    const lit = tmTrace(R).touched, m = wrong.find(w => lit.has(w.y * R.n + w.x)) || wrong[0];
    const el = document.querySelector('#tmI' + R.items.indexOf(m));
    if (el && !reduce) { el.classList.remove('tmhint'); void el.getBoundingClientRect(); el.classList.add('tmhint'); }
    toast('Bu ayna yanlış duruyordu; düzelttim.');
    tmTap(m.i, true);
  }
  /* test helper: solve the current room by tapping the nearest solution's mirrors (the real tap path) */
  function templeAutoSolve(gap) {
    if (!PZ || PZ.kind !== 'temple' || PZ.done) return 0;
    const R = PZ.R, nb = tmNearest(R); if (!nb) return 0;
    const todo = R.rot.filter((m, i) => m.o !== nb.sol[i]).map(m => m.i), tok = PZ.tok, ms = gap == null ? 260 : gap;
    todo.forEach((i, n) => {
      if (!ms) { tmTap(i); return; }
      PZ.timers.push(setTimeout(() => { if (!PZ || PZ.tok !== tok) return; const b = document.querySelector('#sheet [data-tm="' + i + '"]'); if (b && !b.disabled) b.click(); else tmTap(i); }, (n + 1) * ms));
    });
    return todo.length;
  }
  function tmBind() {
    const sh = $('#sheet');
    if (sh && !sh._tmBound) {
      sh._tmBound = true;
      sh.addEventListener('click', e => {
        if (!PZ || PZ.kind !== 'temple') return;
        const t = e.target.closest && e.target.closest('button'); if (!t || t.disabled) return;
        if (t.dataset.tm != null) tmTap(+t.dataset.tm);
        else if (t.dataset.tmf != null) { play('no'); const el = document.querySelector('#tmI' + t.dataset.tmf); if (el && !reduce) { el.classList.remove('tmno'); void el.getBoundingClientRect(); el.classList.add('tmno'); } }
        else if (t.hasAttribute('data-tmh')) tmHint();
        else if (t.hasAttribute('data-tmq')) { if (!PZ.done) { PZ.done = true; play('bad'); tmFin('miss'); } }
      });
    }
    if (!document._tmKey) {
      document._tmKey = true;
      document.addEventListener('keydown', e => {
        if (!PZ || PZ.kind !== 'temple' || !/^Arrow/.test(e.key)) return;
        const cur = document.activeElement && document.activeElement.closest && document.activeElement.closest('.tmb');
        const all = Array.from(document.querySelectorAll('#tmRoom .tmb')); if (!all.length) return;
        e.preventDefault();
        if (!cur) { all[0].focus(); return; }
        const cx = +cur.dataset.x, cy = +cur.dataset.y, [dx, dy] = { ArrowLeft:[-1, 0], ArrowRight:[1, 0], ArrowUp:[0, -1], ArrowDown:[0, 1] }[e.key];
        let best = null, bd = 1e9;
        all.forEach(b => { const ox = +b.dataset.x - cx, oy = +b.dataset.y - cy, fw = ox * dx + oy * dy, side = Math.abs(ox * dy) + Math.abs(oy * dx); if (fw <= 0) return; const sc = fw + side * 2.5; if (sc < bd) { bd = sc; best = b; } });
        if (best) best.focus();
      });
    }
