    /* ---------------- v43 arms and hands (owner: "the merchant's scratching does not look nice") ----------------
       A limb is a smooth tube through centre points with a width at each point (Catmull-Rom centre line, round ends);
       it gets the ink edge, a cel shadow strip on the side away from the light (the light comes from the upper left),
       a soft rim light on the lit side and creases at the joints. Hands are built from a palm and real fingers
       (two or three segments, knuckle creases, nails where they face us). Used by the scratch, tap and arms tells
       and the thinking hand under the chin. */
    const crm = (a, b, c, d, t) => { const t2 = t * t, t3 = t2 * t; return 0.5 * (2 * b + (c - a) * t + (2 * a - 5 * b + 4 * c - d) * t2 + (3 * b - a - 3 * c + d) * t3); };
    function limbS(pts, ws, n) {
      n = n || 8; const S = [], m = pts.length;
      for (let i = 0; i < m - 1; i++) {
        const a = pts[Math.max(0, i - 1)], b = pts[i], c = pts[i + 1], d = pts[Math.min(m - 1, i + 2)];
        for (let k = 0; k < n; k++) { const t = k / n, e = t * t * (3 - 2 * t); S.push([crm(a[0], b[0], c[0], d[0], t), crm(a[1], b[1], c[1], d[1], t), ws[i] + (ws[i + 1] - ws[i]) * e]); }
      }
      S.push([pts[m - 1][0], pts[m - 1][1], ws[m - 1]]);
      for (let i = 0; i < S.length; i++) { const p = S[Math.max(0, i - 1)], q = S[Math.min(S.length - 1, i + 1)], dx = q[0] - p[0], dy = q[1] - p[1], l = Math.hypot(dx, dy) || 1; S[i].push(-dy / l, dx / l); }
      return S;
    }
    /* points along one side of a limb: side +1 / -1, k = fraction of the half width (1 = the edge) */
    const lside = (S, side, k, a, b) => S.slice(a || 0, b == null ? S.length : b).map(q => [q[0] + q[3] * q[2] / 2 * side * k, q[1] + q[4] * q[2] / 2 * side * k]);
    const lpoly = A => A.map((q, i) => (i ? 'L' : 'M') + f(q[0]) + ',' + f(q[1])).join(' ');
    const lcap = (e, sgn) => { const tx = e[4] * sgn, ty = -e[3] * sgn, r = e[2] / 2, k = r * 1.33; return [e[0] + e[3] * r + tx * k, e[1] + e[4] * r + ty * k, e[0] - e[3] * r + tx * k, e[1] - e[4] * r + ty * k]; };
    function limbD(S, cap0, cap1) {
      const A = lside(S, 1, 1), B = lside(S, -1, 1).reverse(), e = S[S.length - 1], s0 = S[0];
      let d = lpoly(A);
      if (cap1) { const c = lcap(e, 1); d += ' C' + f(c[0]) + ',' + f(c[1]) + ' ' + f(c[2]) + ',' + f(c[3]) + ' ' + f(B[0][0]) + ',' + f(B[0][1]); }
      d += ' ' + lpoly(B).replace(/^M/, 'L');
      if (cap0) { const c = lcap(s0, -1); d += ' C' + f(c[2]) + ',' + f(c[3]) + ' ' + f(c[0]) + ',' + f(c[1]) + ' ' + f(A[0][0]) + ',' + f(A[0][1]); }
      return d + ' Z';
    }
    /* a shaded limb: o = { fill, shade (side of the shadow, +1 / -1), cap0, cap1, w (ink width), rim (false = none) } */
    function limb(pts, ws, K, o) {
      o = o || {}; const S = limbS(pts, ws, o.n || 8), fill = o.fill || K.sk, sd = o.shade || 1, sh = fill === K.sk ? K.sh : mx(fill, '#6A4A34', 0.3);
      let s = ink(limbD(S, o.cap0, o.cap1), fill, o.w || 2.4);
      const a = o.a || 0, b = o.b == null ? S.length : o.b;
      s += P(lpoly(lside(S, sd, 0.96, a, b)) + ' ' + lpoly(lside(S, sd, o.sk == null ? 0.25 : o.sk, a, b).reverse()).replace(/^M/, 'L') + ' Z', sh, op(o.so || '.7'));
      if (o.rim !== false) s += P(lpoly(lside(S, -sd, 0.55, a + 2, Math.max(a + 3, b - 2))), 'none', st(o.rimw || 3, fill === K.sk ? K.hl : mx(fill, '#FFFFFF', 0.45), op('.45')));
      return s;
    }
    /* a crease across a limb at sample i (elbow, wrist, knuckle) */
    const lcrease = (S, i, k, col, w, o2) => { const q = S[Math.max(0, Math.min(S.length - 1, i))], r = q[2] / 2 * k, tx = q[4], ty = -q[3]; return P('M' + f(q[0] + q[3] * r) + ',' + f(q[1] + q[4] * r) + ' Q' + f(q[0] + tx * r * 0.35) + ',' + f(q[1] + ty * r * 0.35) + ' ' + f(q[0] - q[3] * r * 0.4) + ',' + f(q[1] - q[4] * r * 0.4), 'none', st(w || 1.3, col, op(o2 || '.75'))); };

    function finger(pts, w, K, o) {

      if (o.nail) { const e = S[S.length - 3], tx = e[4], ty = -e[3], r = e[2] * 0.3; s += P('M' + f(e[0] + e[3] * r - tx * r * 0.2) + ',' + f(e[1] + e[4] * r - ty * r * 0.2) + ' Q' + f(e[0] + tx * r * 2.2) + ',' + f(e[1] + ty * r * 2.2) + ' ' + f(e[0] - e[3] * r - tx * r * 0.2) + ',' + f(e[1] - e[4] * r - ty * r * 0.2) + ' Z', mx(K.sk, '#FFE8D8', 0.45), st(0.9, K.sh2, op('.8'))); }

    }
    /* the silhouette of the head with its hair or wig (for a hand that rests on it): an ellipse */

      s += P([-0.36, -0.12, 0.12, 0.36].map(b => 'M' + pt(Lp(L - 3, b * w - 3)) + ' Q' + pt(Lp(L + 1, b * w)) + ' ' + pt(Lp(L - 3, b * w + 3))).join(' '), 'none', st(1.2, K.sh2, op('.5')));

    /* --- scratch (v45, owner: "both arms hang down and another arm comes from behind; the hand looks strange"):
       the character's own right arm (our left) is raised. Its hanging pose is gone (torsoPath with g.raiseL), and one

    function scratchGeo(Rc, g) {

      const xs = C - sw + 58, ex = Math.max(58, C - sw + 34), ey = 200 + (g.fem ? 4 : 0) - (g.fat || 0) * 6;

    /* the scratching hand seen from the back: wrist at (x, y), ang = wrist -> knuckles; four fingers fan out a little and

    /* the forearm and the hand, in front of the hair and the head */
    function scratchArm(Rc, g, K) {

    /* --- tap: the forearm rests on the counter, the fingers drum on the wood --- */

    /* --- arms: the forearms folded across the chest, the top hand gripping the far upper arm, the other hand tucked away --- */
    function crossedArms(Rc, g, K) {
      const sw = g.sw, hs = g.fem ? 0.88 : 1 + (g.fat || 0) * 0.08, y0 = 382, ux = Math.min(sw - 20, 150);

      s += limb(A, Aw, K, { shade:1, w:2.5 });

      const B = [[C + ux + 30, y0 + 62], [C + 40, y0 + 30], [C - ux + 70, y0 + 6]], Bw = [60 * hs, 50 * hs, 38 * hs];

      /* clipped to the body: the forearms run on under the upper arms at the sides */
      return '<g class="c40ar"><g ' + clip('tc') + '>' + s + '</g></g>';
    }

    /* --- thinking: the forearm rises from below, the chin rests on the loose fist, the index finger along the jaw --- */
    function chinHand(g, K, Rc) {
      Rc = Rc || {};
      const hs = g.fem ? 0.9 : 1 + (g.fat || 0) * 0.06, cx = C + g.cw - 2, cy = g.cy + 4;
      let s = '';

      s += limb(F, Fw, K, { fill:sleeveOf(Rc) || K.sk, shade:1, cap1:true, w:2.6 }) + bangle(SF, SF.length - 4, Rc, g);
      /* the fist: wrist below right, the curled fingers under the chin (the knuckle row toward us), the index finger up the jaw */
      s += handBlock(K, { x:cx + 46, y:cy + 62, ang:-2.3, len:30 * hs, wid:34 * hs, fl:20 * hs, curl:-1, thumb:0, shade:1 });
      s += finger([[cx + 22, cy + 24], [cx + 6, cy + 8], [cx - 3, cy - 8], [cx - 5, cy - 24]], 11.5 * hs, K, { nail:true, shade:-1 });
      return '<g class="c40ch">' + s + '</g>';
    }
    function tellParts(tell, g, K, Rc) {
      let s = '';
      if (tell === 'sweat') {
        const drop = (x, y, sc, cls) => '<g class="' + cls + '">' + P('M' + f(x) + ',' + f(y - 9 * sc) + ' C' + f(x + 6 * sc) + ',' + f(y) + ' ' + f(x + 5 * sc) + ',' + f(y + 6 * sc) + ' ' + f(x) + ',' + f(y + 6 * sc) + ' C' + f(x - 5 * sc) + ',' + f(y + 6 * sc) + ' ' + f(x - 6 * sc) + ',' + f(y) + ' ' + f(x) + ',' + f(y - 9 * sc) + ' Z', '#9FDCEA', st(1.8)) + El(x - 1.5 * sc, y + 1 * sc, 1.4 * sc, 2.2 * sc, '#FFFFFF', op('.8')) + '</g>';
        s += drop(R(g.fw - 6), 150, 1.3, 'c40sw a') + drop(L(g.fw - 10), 160, 1, 'c40sw b') + drop(R(g.fw + 8), 196, 0.9, 'c40sw c');
        s += El(C - 10, g.by - 22, 12, 4, '#FFFFFF', op('.5') + ' class="c40shn"');
      }
      if (tell === 'tap') s += tapArm(Rc, g, K);
      if (tell === 'arms') s += crossedArms(Rc, g, K);

    }

    /* ---------------- composition ---------------- */
    const FRAMES = { bust:'38 26 324 389', card:'40 50 320 300', head:'78 60 244 268', face:'100 98 200 200' };
    const cache = new Map();
    function draw(key, mood, o) {
      const Rc = REC[key], tell = o.tell && TELLS[o.tell] ? o.tell : null, X = expr(mood, Rc, tell), g = geo(Rc);
      LW = o.frame === 'face' ? 1.65 : o.frame === 'head' ? 1.3 : 1;
      const K = Object.assign(pal(Rc), { defs:'' });
      g.dyE = 0; g.raiseL = tell === 'scratch';
      /* v45 small round faces (badges 30–44 px): heavy lids and squints read as closed eyes there (owner: "the Senet rivals
         have their eyes closed"), so icons keep the eyes open; a wink or a blink stays */
      if (o.frame === 'face' && !X.close) { X.o = Math.max(X.o, mood === 'angry' ? 0.92 : 1); X.sq = Math.min(X.sq, 0.2); g.hood *= 0.3; g.eh = Math.max(g.eh, 1); g.es *= 1.12; }
      const fd = facePath(g);
      K.defs += '<clipPath id="fc§"><path d="' + fd + '"/></clipPath>';
      const tilt = X.tilt || 0, dy = X.dy || 0, ht = (tilt || dy ? ' transform="rotate(' + tilt + ' 200 300) translate(0 ' + dy + ')"' : '');
      /* back layer: props, hair */
      let back = propsBack(Rc, g, K) + '<g class="c40h"><g' + ht + '>' + hairBack(Rc, g, K) + '</g></g>';
      /* body */
      let body = (g.raiseL ? scratchUpper(Rc, g, K) : '') + bodyParts(Rc, g, K, X) + neck(Rc, g, K) + collar(Rc, g, K) + propsFront(Rc, g, K);
      /* head */
      const eL = eye(-1, g, X, Rc, K), eR = eye(1, g, X, Rc, K);
      const braided = (Rc.hair || {}).s === 'braids';
      let head = braided ? lappets(Rc, g, K) : lappets(Rc, g, K) + ears(Rc, g, K);
      if (g.fat > 0.6) { const dc = 'M' + L(g.jw - 4) + ',' + f(g.jy + 4) + ' C' + L(g.jw - 6) + ',' + f(g.cy + 18) + ' ' + L(g.cw + 4) + ',' + f(g.cy + 20) + ' ' + C + ',' + f(g.cy + 20) + ' C' + R(g.cw + 4) + ',' + f(g.cy + 20) + ' ' + R(g.jw - 6) + ',' + f(g.cy + 18) + ' ' + R(g.jw - 4) + ',' + f(g.jy + 4) + ' Z'; head += ink(dc, K.sk, 2.4) + P(dc, K.sh, op('.5') + ' transform="translate(6 2)"'); }
      head += ink(fd, K.sk, 2.8) + '<g ' + clip('fc') + '>' + faceDetail(Rc, g, K, X) + '</g>';
      head += nose(g, Rc, K) + beard(Rc, g, K, X);
      head += Gp('c40e', eL.open + eR.open) + (o.live && (eL.closed || eR.closed) ? Gp('c40bl', eL.closed + eR.closed) : '');
      head += brow(-1, g, X, Rc, K) + brow(1, g, X, Rc, K);
      /* mouths: m0 the mood, m1/m2 the talking frames (live only) */
      head += '<g class="c40m m0">' + mouth(X.m, X.mo || 1, g, K, 'mA') + '</g>';
      if (o.live && !X.mo) { const tk = TALK[mood] || 'talk'; head += '<g class="c40m m1">' + mouth(tk, 1, g, K, 'mB') + '</g><g class="c40m m2">' + mouth(tk, 0.35, g, K, 'mC') + '</g>'; }
      head += hairFront(Rc, g, K) + penBehindEar(Rc, g, K) + lotus(Rc, g);
      let front = '';
      if (X.hand === 'chin') front += chinHand(g, K, Rc);
      if (tell === 'scratch') front += scratchArm(Rc, g, K);
      if (tell) front += tellParts(tell, g, K, Rc);

      if (tell === 'lean') fig = '<g transform="translate(200 500) scale(1.08) translate(-200 -490)">' + fig + '</g>';
      const bg = o.bg === false ? '' : o.frame === 'face' ? '<rect x="0" y="0" width="400" height="480" fill="' + (o.bg || Rc.bg || '#F3E2B8') + '"/>' : '';
      return '<svg class="c40' + (tell ? ' t-' + tell : '') + ' f-' + (o.frame || 'bust') + '" viewBox="' + (FRAMES[o.frame] || FRAMES.bust) + '"' + (o.frame === 'card' || o.frame === 'head' ? ' preserveAspectRatio="xMidYMid slice"' : '') + ' role="img" aria-label="' + (CAST[key] ? CAST[key].name : key) + '"><defs>' + K.defs + '</defs>' + bg + '<g class="c40b">' + fig + '</g></svg>';

    function key(who) {
      if (who == null || who === '') return null;
      if (typeof who === 'number' || /^\d+$/.test(String(who))) {
        const i = +who; if (i < 8) return String(i);
        const n = typeof MERCHANTS !== 'undefined' && MERCHANTS[i] ? MERCHANTS[i].name : '';
        for (const c in CAST) if (CAST[c].name === n) return REC[c] ? c : null;
        return null;
      }
      const k = String(who); return REC[k] ? k : null;
    }
    const has = who => !!key(who);
    function svg(who, o) {
      o = o || {}; const k = key(who); if (!k) return '';
      const mood = moodOf(o.mood), ck = k + '|' + mood + '|' + (o.tell || '') + '|' + (o.frame || '') + '|' + (o.live ? 1 : 0) + '|' + (o.bg || '');
      let t = cache.get(ck);
      if (!t) { t = draw(k, mood, o); cache.set(ck, t); }
      /* BOLT OPTIMIZATION: replaceAll avoids array allocations and string splits per SVG render (~50%+ faster) */
      return t.replaceAll('§', 'c' + (++UID));
    }
    const icon = (who, mood) => svg(who, { mood:mood || 'neutral', frame:'face' });
    const RED = () => { try { return matchMedia('(prefers-reduced-motion: reduce)').matches; } catch (e) { return false; } };

    /* ---------------- the live stage ---------------- */
    function stage(el, who, o) {
      o = o || {};
      const k = key(who); if (!el || !k) return null;
      const frame = FRAMES[o.frame] ? o.frame : 'bust';
      el.innerHTML = '<div class="c40s ' + frame + '" style="--bk:' + (3.4 + Math.random() * 1.8).toFixed(2) + 's;--bd:-' + (Math.random() * 3).toFixed(2) + 's;--c40bg:' + (REC[k].bg || '#F3E2B8') + '"></div>';
      const box = el.firstElementChild;
      const h = { alive:true, ready:Promise.resolve(true), mood:moodOf(o.mood), tell:o.tell || null, talk:false, once:null, t:[] };
      let cur = null, shownK = '';
      const paint = fade => {
        if (!h.alive) return;
        const m = h.once || h.mood, sk = m + '|' + (h.tell || '');
        if (sk === shownK) return; shownK = sk;
        const lay = document.createElement('div'); lay.className = 'c40l';
        lay.innerHTML = svg(k, { mood:m, tell:h.tell, frame, live:true });
        box.appendChild(lay);
        const old = cur; cur = lay;
        box.className = box.className.replace(/ t-\w+/g, '') + (h.tell ? ' t-' + h.tell : '');
        if (old) { if (fade && !RED()) { lay.classList.add('in'); h.t.push(setTimeout(() => { if (old.parentNode) old.remove(); }, 200)); } else old.remove(); }
      };
      h.update = u => {
        if (!h.alive || !u) return;
        if (u.mood) h.mood = moodOf(u.mood);
        if (u.tell !== undefined) h.tell = u.tell && TELLS[u.tell] ? u.tell : null;
        if (u.talk !== undefined) { h.talk = !!u.talk && !RED(); box.classList.toggle('talk', h.talk); }
        if (u.act) { box.classList.remove('shrug'); void box.offsetWidth; box.classList.add(u.act === 'shrug' ? 'shrug' : 'pop'); }
        if (u.mood || u.tell !== undefined) paint(true);
      };
      h.react = kind => {
        if (!h.alive) return;
        const m = kind === 'angry' ? 'angry' : kind === 'wave' ? 'happy' : kind === 'sad' ? 'sad' : 'laugh';
        h.once = m; paint(true);
        box.classList.remove('hop', 'shake'); void box.offsetWidth; box.classList.add(kind === 'angry' ? 'shake' : 'hop');
        h.t.push(setTimeout(() => { if (!h.alive) return; h.once = null; paint(true); }, kind === 'angry' ? 1100 : 1300));
      };
      h.onTap = e => { if (e.button > 0) return; h.react('laugh'); };
      box.addEventListener('pointerdown', h.onTap);
      h.dispose = () => { if (!h.alive) return; h.alive = false; h.t.forEach(clearTimeout); clearInterval(watch); box.removeEventListener('pointerdown', h.onTap); };
      const watch = setInterval(() => { if (!h.alive) return clearInterval(watch); if (!box.isConnected) h.dispose(); }, 1500);
      paint(false);
      if (o.greet !== false && !RED()) box.classList.add('enter');
      return h;
    }
    return { has, key, svg, icon, stage, MOODS, TELLS:Object.keys(TELLS), keys:Object.keys(REC), name:k => { const q = key(k); return q == null ? '' : /^\d$/.test(q) ? ['Meret', 'Ptahhotep', 'Nefer', 'Kha', 'Tiye', 'Tefnakht', 'Hatnefer', 'Parahu'][+q] : (CAST[q] || {}).name || q; } };
  })();
