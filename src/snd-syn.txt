  /* ================= music synthesiser (v44) =================
     Every tune is played by instruments built from code, sample by sample: plucked oud and bass strings
     (Karplus-Strong with a tuned all-pass and a lute body EQ), harp and kanun (modal string partials), ney and
     a soft reed (breath noise + harmonics + vibrato and glides), darbuka, frame drum, tabl, riq, sistrum and
     finger cymbals (modal membranes / metal), bowed and sung drones (band-limited wavetables).
     Mixing: buses with EQ, a ping-pong echo, a convolution reverb (generated warm room / temple impulse, run
     at half rate with FFT overlap-add), then master EQ, a glue compressor and a soft ceiling at -1 dBFS.
     NilSynth is self-contained on purpose: its source text is also the body of the render Worker, so it must
     not use any name from the surrounding game. The same job can be stepped on the main thread in short slices
     when Workers are not allowed. Output is a seamless loop (the tail after the loop end is folded onto the start). */
  function NilSynth() {
    const SR = 32000, BLK = 256, SEG = 32768, S2 = SEG >> 1, TAU = Math.PI * 2, LN2 = Math.LN2;
    const hz = m => 440 * Math.pow(2, (m - 69) / 12);
    const clamp = (x, a, b) => x < a ? a : x > b ? b : x;
    function rng(seed) { let a = seed >>> 0; return () => { a = (a + 0x6D2B79F5) | 0; let t = Math.imul(a ^ (a >>> 15), 1 | a); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296; }; }
    const NZ = new Float32Array(1 << 16); { const r = rng(11); for (let i = 0; i < NZ.length; i++) NZ[i] = r() * 2 - 1; }
    const STL = 4096, STAB = new Float32Array(STL + 1); for (let i = 0; i <= STL; i++) STAB[i] = Math.sin(TAU * i / STL);
    const panG = p => { const a = (clamp(p, -1, 1) + 1) * Math.PI / 4; return [Math.cos(a), Math.sin(a)]; };

    /* ---------- biquads (RBJ cookbook, transposed direct form II) ---------- */
    function bq(type, f, q, db, sr) {
      sr = sr || SR; f = Math.min(f, sr * 0.45);
      const w = TAU * f / sr, cw = Math.cos(w), sw = Math.sin(w), al = sw / (2 * (q || 0.707)), A = Math.pow(10, (db || 0) / 40);
      let b0, b1, b2, a0, a1, a2;
      if (type === 'lp') { b0 = (1 - cw) / 2; b1 = 1 - cw; b2 = b0; a0 = 1 + al; a1 = -2 * cw; a2 = 1 - al; }
      else if (type === 'hp') { b0 = (1 + cw) / 2; b1 = -(1 + cw); b2 = b0; a0 = 1 + al; a1 = -2 * cw; a2 = 1 - al; }
      else if (type === 'bp') { b0 = al; b1 = 0; b2 = -al; a0 = 1 + al; a1 = -2 * cw; a2 = 1 - al; }
      else if (type === 'pk') { b0 = 1 + al * A; b1 = -2 * cw; b2 = 1 - al * A; a0 = 1 + al / A; a1 = -2 * cw; a2 = 1 - al / A; }
      else {
        const s = 2 * Math.sqrt(A) * al;
        if (type === 'ls') { b0 = A * ((A + 1) - (A - 1) * cw + s); b1 = 2 * A * ((A - 1) - (A + 1) * cw); b2 = A * ((A + 1) - (A - 1) * cw - s); a0 = (A + 1) + (A - 1) * cw + s; a1 = -2 * ((A - 1) + (A + 1) * cw); a2 = (A + 1) + (A - 1) * cw - s; }
        else { b0 = A * ((A + 1) + (A - 1) * cw + s); b1 = -2 * A * ((A - 1) + (A + 1) * cw); b2 = A * ((A + 1) + (A - 1) * cw - s); a0 = (A + 1) - (A - 1) * cw + s; a1 = 2 * ((A - 1) - (A + 1) * cw); a2 = (A + 1) - (A - 1) * cw - s; }
      }
      return { b0:b0 / a0, b1:b1 / a0, b2:b2 / a0, a1:a1 / a0, a2:a2 / a0, z1:0, z2:0 };
    }
    function lpCo(o, f, q) {
      const w = TAU * Math.min(f, SR * 0.45) / SR, cw = Math.cos(w), al = Math.sin(w) / (2 * q), a0 = 1 + al;
      o.b0 = (1 - cw) / 2 / a0; o.b1 = (1 - cw) / a0; o.b2 = o.b0; o.a1 = -2 * cw / a0; o.a2 = (1 - al) / a0;
    }
    function bqBuf(o, d, n) {
      let z1 = o.z1, z2 = o.z2; const b0 = o.b0, b1 = o.b1, b2 = o.b2, a1 = o.a1, a2 = o.a2;
      for (let i = 0; i < n; i++) { const x = d[i], y = b0 * x + z1; z1 = b1 * x - a1 * y + z2; z2 = b2 * x - a2 * y; d[i] = y; }
      o.z1 = Math.abs(z1) < 1e-25 ? 0 : z1; o.z2 = Math.abs(z2) < 1e-25 ? 0 : z2;
    }

    /* ---------- plucked string: Karplus-Strong, all-pass tuned, one-pole loss ---------- */
    function ks(f, t60, damp, hard, pos, r) {
      const P = SR / f, w = TAU * f / SR, a = damp;
      const lpD = Math.atan2(a * Math.sin(w), 1 - a * Math.cos(w)) / w, H = (1 - a) / Math.sqrt(1 - 2 * a * Math.cos(w) + a * a);
      const D = P - lpD; let N = Math.floor(D), fr = D - N; if (fr < 0.2) { N--; fr += 1; }
      let size = 2; while (size < N + 4) size <<= 1;
      const buf = new Float64Array(size), ex = new Float64Array(N);
      let e = 0; for (let i = 0; i < N; i++) { e += hard * ((r() * 2 - 1) - e); ex[i] = e; }
      const pp = Math.max(1, Math.round(pos * N));
      let mean = 0; for (let i = N - 1; i >= pp; i--) ex[i] -= ex[i - pp];
      for (let i = 0; i < N; i++) mean += ex[i]; mean /= N;
      let pk = 1e-9; for (let i = 0; i < N; i++) { ex[i] -= mean; pk = Math.max(pk, Math.abs(ex[i])); }
      for (let i = 0; i < N; i++) buf[i] = ex[i] / pk;
      return { buf, m:size - 1, N, C:(1 - fr) / (1 + fr), a, g:Math.min(0.99993, Math.pow(0.001, 1 / (f * t60)) / H), gm:Math.pow(0.001, 1 / (f * 0.07)) / H, w:N, ax:0, ay:0, lp:0 };
    }
    function ksStep(s) {
      const x = s.buf[(s.w - s.N) & s.m];
      const ap = s.C * x + s.ax - s.C * s.ay; s.ax = x; s.ay = ap;
      s.lp = (1 - s.a) * ap + s.a * s.lp;
      const y = s.g * s.lp; s.buf[s.w & s.m] = y; s.w++;
      return x;
    }
    /* o: t60, damp, hard, pos, det (cents, second string of the course), off (s: finger mute), pan, len */
    function vPluck(f, vel, o, r) {
      const det = o.det || 0, k = Math.pow(2, det / 1200);
      const s1 = ks(f * k, o.t60, o.damp, o.hard, o.pos, r), s2 = det ? ks(f / k, o.t60 * 0.9, o.damp, o.hard, o.pos * 1.1, r) : null;
      const off = o.off ? Math.round(o.off * SR) : 0, len = Math.round(Math.min(o.len || o.t60 * 1.05, o.off ? o.off + 0.18 : 99) * SR), fade = Math.round(0.03 * SR);
      const [gl, gr] = panG(o.pan || 0), v = vel * (s2 ? 0.55 : 0.8);
      let n = 0;
      return (L, R, i0, i1) => {
        for (let i = i0; i < i1; i++, n++) {
          if (n === off && off) { s1.g = Math.min(s1.g, s1.gm); s1.a = Math.max(s1.a, 0.55); if (s2) { s2.g = Math.min(s2.g, s2.gm); s2.a = Math.max(s2.a, 0.55); } }
          let y = ksStep(s1); if (s2) y += 0.85 * ksStep(s2);
          y *= v; if (n > len - fade) y *= (len - n) / fade;
          L[i] += y * gl; R[i] += y * gr;
          if (n >= len) return false;
        }
        return true;
      };
    }
    /* ---------- modal voice: string partials (harp, kanun) or struck membranes / metal ---------- */
    function vModes(modes, vel, o, r) {
      const M = modes.length, c1 = new Float64Array(M), c2 = new Float64Array(M), y1 = new Float64Array(M), y2 = new Float64Array(M), ex = new Float64Array(M);
      modes.forEach(([f, a, t], i) => { const w = TAU * Math.min(f, SR * 0.45) / SR, rr = Math.pow(0.001, 1 / (t * SR)); c1[i] = 2 * rr * Math.cos(w); c2[i] = rr * rr; ex[i] = a * Math.sin(w) * vel; });
      const hits = o.hits || [[0, 1]], hs = hits.map(h => Math.round(h[0] * SR));
      const len = Math.round((o.len || 1) * SR), fade = Math.round(0.02 * SR), [gl, gr] = panG(o.pan || 0);
      /* pitched body: a sine with a falling pitch (membrane tension at the strike) */
      const ob = o.osc, oA = ob ? ob[3] * vel : 0, oK = ob ? Math.pow(0.001, 1 / (ob[4] * SR)) : 0, oT = ob ? Math.exp(-1 / (ob[2] * SR)) : 0;
      let oPh = 0, oE = 0, oDf = ob ? ob[0] - ob[1] : 0;
      /* noise: [amp, t60, type, freq, q] — re-triggered on every hit */
      const nz = (o.noise || []).map(q => ({ a:q[0] * vel, k:Math.pow(0.001, 1 / (q[1] * SR)), f:bq(q[2], q[3], q[4]), e:0 }));
      let n = 0, hi = 0, ni = (r() * 65535) | 0;
      return (L, R, i0, i1) => {
        for (let i = i0; i < i1; i++, n++) {
          let x = 0;
          while (hi < hs.length && n === hs[hi]) { const h = hits[hi][1]; for (let m = 0; m < M; m++) y1[m] += ex[m] * h; for (let q = 0; q < nz.length; q++) nz[q].e = nz[q].a * h; if (n === 0) oE = oA; hi++; }
          let y = 0;
          for (let m = 0; m < M; m++) { const v = c1[m] * y1[m] - c2[m] * y2[m]; y2[m] = y1[m]; y1[m] = v; y += v; }
          if (ob) { oDf *= oT; oPh += (ob[1] + oDf) / SR; if (oPh >= 1) oPh -= 1; y += STAB[(oPh * STL) | 0] * oE * (n < 24 ? n / 24 : 1); oE *= oK; }
          for (let q = 0; q < nz.length; q++) { const z = nz[q]; if (z.e > 1e-5) { const f = z.f, s = NZ[(ni++) & 65535] * z.e, v = f.b0 * s + f.z1; f.z1 = f.b1 * s - f.a1 * v + f.z2; f.z2 = f.b2 * s - f.a2 * v; x += v; z.e *= z.k; } }
          y += x;
          if (n > len - fade) y *= (len - n) / fade;
          L[i] += y * gl; R[i] += y * gr;
          if (n >= len) return false;
        }
        return true;
      };
    }
    function stringModes(f, o) {
      const out = [], K = o.K || 10, B = o.B || 0.00012, p = o.pos || 0.25;
      for (let k = 1; k <= K; k++) {
        const fk = k * f * Math.sqrt(1 + B * k * k); if (fk > (o.top || 6000)) break;
        const a = Math.abs(Math.sin(Math.PI * k * p)) / Math.pow(k, o.roll || 1.15);
        out.push([fk, a, o.t60 / (1 + (o.hd || 0.35) * (k - 1))]);
        if (o.det && k <= 4) out.push([fk * (1 + o.det), a * 0.7, o.t60 / (1 + (o.hd || 0.35) * (k - 1)) * 0.92]);
      }
      return out;
    }
    /* ---------- wind phrase: ney (breathy) or a soft folk reed, legato with glides, scoops and vibrato ---------- */
    const WIND = {
      ney: { h:[1, .36, .17, .09, .045, .025], breath:.2, chiff:.55, vib:.0105, rate:5.1, glide:.024, dip:.3, bp:[1700, .6], top:7000, lp:6500 },
      reed: { h:[1, .5, .62, .36, .4, .22, .2, .12, .09, .06], breath:.05, chiff:.25, vib:.007, rate:5.9, glide:.016, dip:.4, bp:[2600, .9], top:5600, lp:5000 }
    };
    function vWind(notes, BEAT, vel, o, r) {
      const W = WIND[o.timbre || 'ney'], segs = []; let t = 0;
      notes.forEach(([m, b]) => { const d = b * BEAT * SR; if (m != null) segs.push({ s:Math.round(t), e:Math.round(t + d) - (o.sep ? Math.round(o.sep * SR) : 0), lf:Math.log2(hz(m)), K:0 }); t += d; });
      segs.forEach((sg, i) => { sg.K = Math.max(1, Math.min(W.h.length, Math.floor(W.top / Math.pow(2, sg.lf)))); sg.leg = i > 0 && segs[i - 1].e >= sg.s - 40; sg.v = vel * (o.acc && o.acc[i] ? o.acc[i] : 1); });
      const end = Math.round(t) + Math.round(0.6 * SR), [gl, gr] = panG(o.pan || 0);
      const kAtt = 1 - Math.exp(-1 / (0.035 * SR)), kRel = 1 - Math.exp(-1 / (0.09 * SR)), kGl = 1 - Math.exp(-1 / (W.glide * SR)), dipN = Math.round(0.045 * SR);
      const bpf = bq('bp', W.bp[0], W.bp[1]), lpf = bq('lp', W.lp, 0.7), H = W.h, vibD = W.vib * (o.vib || 1), vr = W.rate * (0.95 + r() * 0.1) / SR;
      let n = 0, si = 0, lf = segs.length ? segs[0].lf - 0.03 : 8, ph = 0, vph = r(), amp = 0, ch = 0, ni = (r() * 65535) | 0, wob = 0;
      return (L, R, i0, i1) => {
        for (let i = i0; i < i1; i++, n++) {
          while (si < segs.length && n >= segs[si].e) si++;
          const sg = segs[si]; let tgt = 0, vd = 0, K = 1;
          if (sg && n >= sg.s) {
            const into = n - sg.s;
            if (into === 0) { ch = 1; if (!sg.leg) lf = sg.lf - 0.035; }
            tgt = sg.v * (sg.leg && into < dipN ? 1 - W.dip * (1 - into / dipN) : 1);
            const dur = sg.e - sg.s; if (dur > SR * 0.6) tgt *= 0.9 + 0.1 * Math.sin(Math.PI * into / dur);
            vd = vibD * clamp((into - 0.2 * SR) / (0.4 * SR), 0, 1);
            lf += (sg.lf - lf) * kGl; K = sg.K;
          } else if (si > 0) K = segs[si - 1].K;
          vph += vr; if (vph >= 1) vph -= 1; wob += 0.0004 * ((NZ[(ni + 7919) & 65535]) - wob * 8);
          const f = Math.exp((lf + vd * STAB[(vph * STL) | 0] + wob * 0.003) * LN2);
          ph += f / SR; if (ph >= 1) ph -= 1;
          amp += (tgt - amp) * (tgt > amp ? kAtt : kRel);
          if (amp < 1e-5 && tgt === 0) { if (n >= end) return false; continue; }
          let tone = 0; for (let k = 0; k < K; k++) { let p = ph * (k + 1); p -= p | 0; tone += H[k] * STAB[(p * STL) | 0]; }
          const s = NZ[(ni++) & 65535], b = bpf.b0 * s + bpf.z1; bpf.z1 = bpf.b1 * s - bpf.a1 * b + bpf.z2; bpf.z2 = bpf.b2 * s - bpf.a2 * b;
          const br = b * (0.6 + 0.4 * STAB[(ph * STL) | 0]) * (W.breath + W.chiff * ch); ch *= 0.9993;
          let y = (tone * 0.5 + br) * amp;
          const v2 = lpf.b0 * y + lpf.z1; lpf.z1 = lpf.b1 * y - lpf.a1 * v2 + lpf.z2; lpf.z2 = lpf.b2 * y - lpf.a2 * v2; y = v2;
          L[i] += y * gl; R[i] += y * gr;
          if (n >= end) return false;
        }
        return true;
      };
    }
    /* ---------- sustained drones / pads from band-limited wavetables ---------- */
    const WT = {};
    function table(kind, f) {
      const band = Math.max(0, Math.floor(Math.log2(f / 27.5))), key = kind + band;
      if (WT[key]) return WT[key];
      const top = kind === 'choir' ? 3600 : 5200, H = Math.max(1, Math.min(48, Math.floor(top / (27.5 * Math.pow(2, band + 1))))), T = new Float32Array(2049);
      for (let k = 1; k <= H; k++) {
        const a = kind === 'choir' ? 1 / Math.pow(k, 1.7) : kind === 'reed' ? (k % 2 ? 1 : 0.45) / Math.pow(k, 1.1) : 1 / Math.pow(k, 1.3);
        const p = (k * 2.3) % TAU; for (let i = 0; i < 2048; i++) T[i] += a * Math.sin(TAU * k * i / 2048 + p);
      }
      let pk = 1e-9; for (let i = 0; i < 2048; i++) pk = Math.max(pk, Math.abs(T[i])); for (let i = 0; i < 2048; i++) T[i] /= pk; T[2048] = T[0];
      return (WT[key] = T);
    }
    const VOWEL = { a:[[730, 1, 7], [1100, 1.4, 6], [2450, 2, 3]], o:[[520, 1, 8], [880, 1.4, 6], [2400, 2, 2]], u:[[330, 1, 8], [800, 1.6, 5], [2300, 2, 1]] };
    function vPad(m, dur, vel, o, r) {
      const f = hz(m), T = table(o.timbre || 'str', f), dets = o.voices || [-7, 0, 6], NV = dets.length, inc = dets.map(c => f * Math.pow(2, c / 1200) * 2048 / SR), ph = dets.map(() => r() * 2048);
      const att = Math.round((o.att || 1.2) * SR), rel = (o.rel || 1.6), sus = Math.round(dur * SR), len = sus + Math.round(rel * 3 * SR), kR = Math.pow(0.001, 1 / (rel * SR));
      const filt = [bq('lp', o.cut || 2000, 0.6)]; if (o.timbre === 'choir') VOWEL[o.vowel || 'a'].forEach(([ff, q, db]) => filt.push(bq('pk', ff, q * 2.2, db)));
      const [gl, gr] = panG(o.pan || 0), v = vel / Math.sqrt(NV), lfoR = (0.07 + r() * 0.06) / SR, vibR = 4.6 / SR;
      let n = 0, env = 0, lp = r(), vp = r();
      return (L, R, i0, i1) => {
        for (let i = i0; i < i1; i++, n++) {
          env = n < att ? n / att : n < sus ? 1 : env * kR;
          vp += vibR; if (vp >= 1) vp -= 1; const vb = 1 + 0.0012 * STAB[(vp * STL) | 0];
          let y = 0;
          for (let k = 0; k < NV; k++) { let p = ph[k] + inc[k] * vb; if (p >= 2048) p -= 2048; ph[k] = p; const j = p | 0, fr = p - j; y += T[j] + (T[j + 1] - T[j]) * fr; }
          y *= v * env;
          for (let q = 0; q < filt.length; q++) { const b = filt[q], o2 = b.b0 * y + b.z1; b.z1 = b.b1 * y - b.a1 * o2 + b.z2; b.z2 = b.b2 * y - b.a2 * o2; y = o2; }
          if ((n & 63) === 0) { lp += lfoR * 64; lpCo(filt[0], (o.cut || 2000) * (1 + 0.25 * Math.sin(TAU * lp)), 0.6); }
          L[i] += y * gl; R[i] += y * gr;
          if (n >= len) return false;
        }
        return true;
      };
    }
    /* ---------- percussion kit (modal membranes, rims, metal) ---------- */
    const DR = {
      D:{ osc:[160, 100, 0.028, .9, 0.38], m:[[92 * 1.59, .3, .26], [92 * 2.14, .24, .18], [92 * 2.65, .12, .12], [92 * 3.16, .07, .09]], nz:[[.35, .035, 'lp', 900, .7]], len:.7 },
      T:{ m:[[640, .42, .09], [1010, .38, .07], [1460, .32, .06], [1890, .22, .05], [2620, .14, .04]], nz:[[.75, .022, 'bp', 3100, 1.1]], len:.22 },

    function vDrum(k, vel, pan, r, tune) {

    const FFTP = {};
    function plan(N) {
      if (FFTP[N]) return FFTP[N];
      const lg = Math.round(Math.log2(N)), rev = new Uint32Array(N), cs = new Float64Array(N / 2), sn = new Float64Array(N / 2);
      for (let i = 0; i < N; i++) { let x = i, y = 0; for (let b = 0; b < lg; b++) { y = (y << 1) | (x & 1); x >>= 1; } rev[i] = y; }
      for (let i = 0; i < N / 2; i++) { cs[i] = Math.cos(TAU * i / N); sn[i] = Math.sin(TAU * i / N); }
      return (FFTP[N] = { rev, cs, sn });
    }
    function fft(re, im, N, inv) {
      const P = plan(N), rev = P.rev, cs = P.cs, sn = P.sn, sg = inv ? 1 : -1;
      for (let i = 0; i < N; i++) { const j = rev[i]; if (j > i) { let t = re[i]; re[i] = re[j]; re[j] = t; t = im[i]; im[i] = im[j]; im[j] = t; } }
      for (let size = 2; size <= N; size <<= 1) {
        const half = size >> 1, step = N / size;
        for (let i = 0; i < N; i += size) for (let j = 0, k = 0; j < half; j++, k += step) {
          const a = i + j, b = a + half, wr = cs[k], wi = sg * sn[k], xr = re[b] * wr - im[b] * wi, xi = re[b] * wi + im[b] * wr;
          re[b] = re[a] - xr; im[b] = im[a] - xi; re[a] += xr; im[a] += xi;
        }
      }
    }
    /* a warm room: sparse early reflections, then decorrelated noise whose top end dies away faster than its body */
    function makeIR(sr, o) {
      const len = Math.round(o.len * sr), out = [];
      for (let ch = 0; ch < 2; ch++) {
        const r = rng((o.seed || 5) + ch * 101), d = new Float32Array(len), pre = Math.round((o.pre || 0.018) * sr);
        let lp = 0, lp2 = 0;
        for (let i = pre; i < len; i++) {
          const t = (i - pre) / sr, fc = (o.hi || 5200) * Math.pow((o.lo || 900) / (o.hi || 5200), Math.min(1, t / o.len * 1.6)), a = 1 - Math.exp(-TAU * fc / sr);
          lp += a * ((r() * 2 - 1) - lp); lp2 += a * (lp - lp2);
          d[i] = lp2 * Math.exp(-6.9 * t / (o.t60 || o.len * 0.8)) * Math.min(1, t / 0.012);
        }
        for (let e = 0; e < (o.er || 8); e++) { const at = pre + Math.round((0.004 + r() * 0.07) * sr); if (at < len) d[at] += (r() < 0.5 ? -1 : 1) * (0.5 - e * 0.04) * (o.erg || 1); }
        let en = 0; for (let i = 0; i < len; i++) en += d[i] * d[i]; en = Math.sqrt(en) || 1;
        for (let i = 0; i < len; i++) d[i] /= en;
        out.push(d);
      }
      return out;
    }

          for (let i = 0; i < BLK; i++) {
            const l = L[i], r = R[i];
            oL[o0 + i] += l * g; oR[o0 + i] += r * g; snL[o0 + i] += l * rv; snR[o0 + i] += r * rv; dIn[i] += (l + r) * 0.5 * dl;
            L[i] = 0; R[i] = 0; e2 += (l * l + r * r) * g * g;
          }
          if (meter) meter[k] += e2; e2 = 0;
        }
        for (let i = 0; i < BLK; i++) {
          const ri = (dw - DT) & DM, xl = dL[ri], xr = dR[ri];
          const fx = xr * 0.34, fb = dLP.b0 * fx + dLP.z1; dLP.z1 = dLP.b1 * fx - dLP.a1 * fb + dLP.z2; dLP.z2 = dLP.b2 * fx - dLP.a2 * fb;
          const ix = dIn[i], inp = dHP.b0 * ix + dHP.z1; dHP.z1 = dHP.b1 * ix - dHP.a1 * inp + dHP.z2; dHP.z2 = dHP.b2 * ix - dHP.a2 * inp;
          dL[dw & DM] = inp + fb; dR[dw & DM] = xl; dw++;
          oL[o0 + i] += xl * 0.5; oR[o0 + i] += xr * 0.5; snL[o0 + i] += xl * 0.15; snR[o0 + i] += xr * 0.15;
        }
        pos = b1; sp += BLK;
      }
      function segment() {
        /* reverb: decimate the send, convolve (left and right packed into one complex FFT), overlap-add */
        for (let i = 0; i < SEG; i++) { let l = snL[i], r = snR[i]; for (let q = 0; q < 2; q++) { const a = dec[q], y = a.b0 * l + a.z1; a.z1 = a.b1 * l - a.a1 * y + a.z2; a.z2 = a.b2 * l - a.a2 * y; l = y; const b = dec[q + 2], z = b.b0 * r + b.z1; b.z1 = b.b1 * r - b.a1 * z + b.z2; b.z2 = b.b2 * r - b.a2 * z; r = z; } snL[i] = l; snR[i] = r; }
        fr.fill(0); fi.fill(0); for (let m = 0; m < S2; m++) { fr[m] = snL[m * 2]; fi[m] = snR[m * 2]; }
        fft(fr, fi, N, false);
      }
      function segment1() {
        for (let k = 0; k < N; k++) {
          const kk = (N - k) & (N - 1), xlr = (fr[k] + fr[kk]) / 2, xli = (fi[k] - fi[kk]) / 2, xrr = (fi[k] + fi[kk]) / 2, xri = -(fr[k] - fr[kk]) / 2;
          const ylr = xlr * HLr[k] - xli * HLi[k], yli = xlr * HLi[k] + xli * HLr[k], yrr = xrr * HRr[k] - xri * HRi[k], yri = xrr * HRi[k] + xri * HRr[k];
          zr[k] = ylr - yri; zi[k] = yli + yrr;
        }
        fft(zr, zi, N, true);
        const base = (pos - SEG) >> 1, inv = 1 / N, wet = TK.wet * 2;
        for (let m = 0; m < N; m++) { const j = (base + m) & (WS - 1); wL[j] += zr[m] * inv; wR[j] += zi[m] * inv; }
        for (let m = 0; m < S2; m++) {
          const j = (base + m) & (WS - 1), l = wL[j], r = wR[j]; wL[j] = 0; wR[j] = 0;
          if (meter) meter[BUSES.length + 1] += (l * l + r * r) * wet * wet * 2;
          segL[m * 2] += (wPrevL + l) * 0.5 * wet; segR[m * 2] += (wPrevR + r) * 0.5 * wet; segL[m * 2 + 1] += l * wet; segR[m * 2 + 1] += r * wet;
          wPrevL = l; wPrevR = r;
        }
      }
      function segment2() {
        /* master: EQ, glue compressor, trim, soft ceiling (never above -1 dBFS) */
        for (let q = 0; q < mL.length; q++) { bqBuf(mL[q], segL, SEG); bqBuf(mR[q], segR, SEG); }
        const s0 = pos - SEG, sl = new Float32Array(SEG), sr = new Float32Array(SEG);
        for (let i = 0; i < SEG; i++) {
          let l = segL[i], r = segR[i]; const lv = Math.max(Math.abs(l), Math.abs(r));
          cEnv += (lv - cEnv) * (lv > cEnv ? cAtt : cRel);
          if ((i & 7) === 0) { const db = 20 * Math.log10(cEnv + 1e-9), ov = db - thr; const gr = ov <= -knee2 ? 0 : ov < knee2 ? (ov + knee2) * (ov + knee2) / (2 * knee) * slope : ov * slope; cG = Math.pow(10, -gr / 20) * trim; }
          l *= cG; r *= cG;
          const al = Math.abs(l), ar = Math.abs(r);
          if (al > 0.7) l = Math.sign(l) * (0.7 + 0.19 * Math.tanh((al - 0.7) / 0.19));
          if (ar > 0.7) r = Math.sign(r) * (0.7 + 0.19 * Math.tanh((ar - 0.7) / 0.19));
          sl[i] = l; sr[i] = r;
          const p = s0 + i;
          if (p < LOOP) { outL[p] = l; outR[p] = r; } else if (p - LOOP < LOOP) { outL[p - LOOP] += l; outR[p - LOOP] += r; }
        }
        if (J.onSeg && s0 < LOOP) { const n = Math.min(SEG, LOOP - s0); J.onSeg(s0, n < SEG ? sl.slice(0, n) : sl, n < SEG ? sr.slice(0, n) : sr); }
        sp = 0;
      }
      let fin = 0, s2 = 0;
      /* final pass in slices: ceiling on the folded start, peak and RMS */
      function finish() {
        const e = Math.min(LOOP, fin + 131072);
        for (const d of J.out) for (let i = fin; i < e; i++) {
          let x = d[i]; const a = Math.abs(x);
          if (a > 0.89) { x = Math.sign(x) * (0.7 + 0.19 * Math.tanh((a - 0.7) / 0.19)); d[i] = x; }
          const b = Math.abs(x); if (b > peak) peak = b; s2 += x * x;
        }
        fin = e; if (fin < LOOP) return;
        if (meter) { J.meter = {}; let dry = 0; BUSES.forEach((k, i) => { dry += meter[i]; }); BUSES.forEach((k, i) => { J.meter[k] = +(10 * Math.log10(meter[i] / dry + 1e-12)).toFixed(1); }); J.meter.wet = +(10 * Math.log10(meter[BUSES.length + 1] / dry + 1e-12)).toFixed(1); }
        J.stats = { rms:Math.sqrt(s2 / (LOOP * 2)), peak, sec:LOOP / SR, events:ev.length, voices:J.maxVoices };
        J.done = true; J.prog = 1;
      }
      J.dbg = () => ({ pos, sph, sp, act:act.length });
      J.step = (ms) => {
        const t0 = Date.now();
        while (!J.done) {
          if (sp >= SEG) { if (sph === 0) segment(); else if (sph === 1) segment1(); else segment2(); sph = (sph + 1) % 3; }
          else if (pos < TOTAL) block();
          else finish();
          J.prog = Math.min(0.99, pos / TOTAL);
          if (Date.now() - t0 >= ms) break;
        }
        return J.done;
      };
      return J;
    }
    return { SR, job, makeIR, _t:{ vPluck, vModes, vWind, vPad, vDrum, stringModes, hz, rng, parse }, tracks:Object.keys(TRACKS), meta:n => ({ bpm:TRACKS[n].bpm, bars:TRACKS[n].bars, sec:TRACKS[n].bars * 4 * 60 / TRACKS[n].bpm }) };
  }
