    function loop(now) {
      if (!st) return;
      /* 120 Hz phones: draw at most ~60 times a second */
      if (now - last < 15) { raf = requestAnimationFrame(loop); return; }
      const dt = Math.min(0.05, (now - last) / 1000); last = now;
      if (!paused) { if (st.hurtT > 0) st.hurtT -= dt; if (st.hullT > 0) st.hullT -= dt; if (!st.end) { if (st.cfg) step(dt); else { if (st.count > 0) st.count -= dt; else SC[st.kind].step(st, dt); } } if (!RT.nodraw) draw(); }
      raf = requestAnimationFrame(loop);
