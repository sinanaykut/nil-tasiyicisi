    /* does an obstacle touch a boat at lane-x x, world y (same shapes as our own collisions) */
    function touches(o, x, y, br) {
      if (o.k === 'whirl' || o.k === 'quick') return false;
      if (o.k === 'hippo' && !(Math.sin(o.ph * 2.2) > -0.1)) return false;
      const ox = sx(o.u, o.y), oy = sy(o.y), wide = o.k === 'sand' || o.k === 'dune', rr = wide ? o.w * 0.45 : o.k === 'log' ? o.len * 0.45 : o.r;
      const dx = x - ox, dy = (sy(y) - 8) - oy, d2 = o.k === 'log' || wide ? Math.hypot(dx / (rr / o.r), dy) : Math.hypot(dx, dy);
      return d2 < o.r + br;
    }
    /* Sethi picks the cheapest lane: clear water, a current, our draft; never our wake or our hull */
    function sethiLane(rv) {
      const hw = halfW(), look = RT.look, g = st.dist - rv.y; let best = rv.u, bc = 1e9;
      for (let k = -8; k <= 8; k++) {
        const L = k / 10; let c = Math.abs(L - rv.u) * 0.5 + Math.max(0, Math.abs(L) - 0.74) * 5 + Math.abs(L - rv.lane) * 0.12 + Math.abs(L - rv.home) * 0.45;
        for (const o of st.obs) {
          const a = o.y - rv.y; if (a < -24 || a > look || o.k === 'whirl' || (o.miss && a > 64)) continue;
          let ou = o.u + (o.vu || 0) * Math.max(0, a) / Math.max(120, rv.v); if (Math.abs(ou) > 0.85) ou = Math.sign(ou) * 1.7 - ou;
          const half = ((o.k === 'log' ? o.len * 0.5 : o.k === 'sand' ? o.w * 0.5 : o.r) + 24) / hw, d = Math.abs(L - ou);
          if (d < half) c += 2 + 4 * (1 - d / half) * (1 - Math.max(0, a) / (look + 80));
        }
        for (const sm of st.streams) { const a = sm.y - rv.y; if (a > 320 || a < -sm.len + 40) continue; if (Math.abs(L - sm.u) < 0.12) c -= 0.9; }
        /* v42: side by side he keeps a good boat-width of water between the hulls; tucked in behind us he may still draft */
        if (g > 60 && g < 200) { const d = Math.abs(L - st.u) * hw; if (d < 24) c += 1.2; else if (d < 56) c -= 0.25; }
        if (Math.abs(g) < 110) { const d = Math.abs(L - st.u) * hw; if (d < 84) c += 2.6 * (1 - d / 84) + 0.6; }
        if (c < bc) { bc = c; best = L; }
      }
      return best;
    }
    function raceStep(dt, bx) {
      const r = st.race, rv = st.rival, hw = halfW();
      const bi = Math.floor(st.t / r.per), newBeat = bi !== r.lastBeat;
      if (newBeat) { r.lastBeat = bi; if (r.tP == null) { drumSnd(bi % 4 === 0); r.beatT = 0.22; padFx('beat'); } }
      if (r.beatT > 0) r.beatT -= dt; if (r.hitT > 0) r.hitT -= dt;
      if (r.fbT > 0) r.fbT -= dt; if (r.flashT > 0) r.flashT -= dt; if (r.bump > 0) r.bump -= dt;
      /* steering */
      rv.plan -= dt;
      /* his home side is the half we are not in; it only flips after we have held his side for a while */
      { const want = st.u > 0.12 ? -0.5 : st.u < -0.12 ? 0.5 : rv.home; if (want !== rv.home) { rv.homeT += dt; if (rv.homeT > 1.4) { rv.home = want; rv.homeT = 0; } } else rv.homeT = 0; }
      if (r.tS != null) rv.tu = rv.park; else if (rv.plan <= 0) { rv.plan = 0.12; rv.tu = sethiLane(rv); }
      const tg = Math.max(-1.5, Math.min(1.5, (rv.tu - rv.u) * 4));
      rv.vu += (tg - rv.vu) * Math.min(1, dt * 5); rv.u += rv.vu * dt * 0.62;
      if (Math.abs(rv.u) > 0.92) { rv.u = Math.sign(rv.u) * 0.92; rv.vu *= -0.3; }
      /* Sethi's strokes: a strong start, a push when he is behind, and a sprint at the end, always on the beat */
      if (rv.boostT > 0) rv.boostT -= dt; if (rv.slowT > 0) rv.slowT -= dt; if (rv.inv > 0) rv.inv -= dt;
      rv.rech += dt; if (rv.charges < 3 && rv.rech > RT.rech) { rv.rech = 0; rv.charges++; }
      const prog = rv.y / r.finY;
      const gapP = rv.y - st.dist, left = r.finY - rv.y;
      if (newBeat && r.tS == null && rv.charges > 0 && rv.boostT <= 0 && (bi === 1 || (gapP < -130 && R() < 0.35) || (left < r.finY * 0.25 && gapP < 150 && R() < 0.8) || (rv.charges > 2 && R() < 0.06))) {
        rv.charges--; rv.boostT = 0.9; rv.boosts++;
        if (left < r.finY * 0.28 && !r.toldS) { r.toldS = true; flash('Sethi son gücüyle asılıyor!'); }
      }
      /* wake and draft, both ways */
      const lat = Math.abs(st.u - rv.u) * hw, run = r.tP == null && r.tS == null;
      r.wake = run && gapP > 28 && gapP < 175 && lat < 22;
      r.draft = run && !r.wake && gapP > 20 && gapP < 150 && lat < 58;
      rv.wake = run && -gapP > 28 && -gapP < 175 && lat < 22;
      rv.draft = run && !rv.wake && -gapP > 20 && -gapP < 150 && lat < 58;
      if (r.wake) { r.wakeT += dt; if (!r.toldW) { r.toldW = true; flash('Sethi\'nin dalgası! Biraz yana kay'); } }
      if (r.draft) { r.draftT += dt; if (!r.toldD) { r.toldD = true; flash('Rüzgâr gölgesi: hızlandın!'); } }
      /* Sethi's speed */
      let v = rv.base * (1 + Math.min(0.4, st.t / 40) * 0.25);
      /* he never lets it get boring: a pull when he is behind (fading in the last fifth), an easier oar when far ahead */
      if (RT.rubber) { const f = prog < 0.8 ? 1 : Math.max(0.25, (1 - prog) / 0.2); if (gapP < -100) v *= 1 + Math.min(RT.rb, (-gapP - 100) / 1500) * f; else if (gapP > 130) v *= 1 - Math.min(RT.ra, (gapP - 130) / 900); }
      if (rv.boostT > 0) v *= RT.sb;
      rv.inStream = st.streams.some(sm => rv.y > sm.y && rv.y < sm.y + sm.len && Math.abs(rv.u - sm.u) < 0.16); if (rv.inStream) v *= RT.sm;
      if (rv.slowT > 0) v *= 0.6; if (rv.stall > 0) v *= 0.4; if (rv.wake) v *= 0.84; else if (rv.draft) v *= 1.1;
      if (r.tS != null) v = rv.base * 0.12;
      rv.v += (v - rv.v) * Math.min(1, dt * 4); rv.y += rv.v * dt; rv.row += dt * (rv.boostT > 0 ? 14 : 7);
      rv.trail.push({ x:sx(rv.u, rv.y), y:rv.y }); if (rv.trail.length > 46) rv.trail.shift();
      /* Sethi can misjudge a rock too, and once in a while an oar snaps */
      const rx = sx(rv.u, rv.y), ry = sy(rv.y);
      if (rv.stall > 0) rv.stall -= dt;
      if (rv.blunder && prog >= rv.blunder && r.tS == null) { rv.blunder = 0; rv.stall = 2; rv.boostT = 0; flash('Sethi\'nin küreği kırıldı!'); sfx('lash'); for (let k = 0; k < 12; k++) st.parts.push({ x:rx, y:ry, vx:rnd(-120, 120), vy:rnd(-120, 40), life:.7, max:.7, c:k % 2 ?       if (rv.inv <= 0) for (const o of st.obs) if (touches(o, rx, rv.y, 14)) {
        rv.inv = 1.1; rv.slowT = 0.65; rv.hits++; rv.vu += (rv.u > o.u ? 1 : -1) * 0.8;
        for (let k = 0; k < 14; k++) st.parts.push({ x:rx, y:ry, vx:rnd(-150, 150), vy:rnd(-150, 50), life:.6, max:.6, c:k % 3 ? C.foam : '#BFE3C2', r:rnd(2, 4.5) });
        if (ry > 40 && ry < H - 20) floatText(rx, ry - 50, 'Sethi çarptı!', '#F2C65A'); else flash('Sethi kayaya sürttü!');
        sfx('splash'); break;
      }
      /* hull against hull */
      if (run && Math.abs(gapP) < 54 && lat < 28 && r.bump <= 0) {
        const sd = st.u >= rv.u ? 1 : -1; st.vu += sd * 1.4; rv.vu -= sd * 1.4; st.slowT = Math.max(st.slowT, 0.25); rv.slowT = Math.max(rv.slowT, 0.25);
        r.bump = 1; st.shake = 0.15; sfx('hit'); flash('Bordalar çarpıştı!');
      }
      /* lead changes (with a little hysteresis so it does not flicker) */
      if (Math.abs(gapP) > 20) { const lead = gapP > 0 ? -1 : 1; if (r.lead && lead !== r.lead && run && st.t > 2) { r.swaps++; flash(lead > 0 ? 'Sethi\'yi geçtin!' : 'Sethi öne geçti!'); } r.lead = lead; }
      if (!r.toldEnd && r.finY - st.dist < r.finY * 0.2) { r.toldEnd = true; if (!r.toldS) flash('Son düzlük! Kalabalık bekliyor'); }
      /* the line */
      if (r.tP == null && st.dist >= r.finY) { r.tP = st.t - (st.dist - r.finY) / Math.max(60, st.v); r.park = st.u >= rv.u ? 0.62 : -0.62; st.boostT = 0; if (r.tS == null) r.flashT = 0.3; }
      if (r.tS == null && rv.y >= r.finY) { r.tS = st.t - (rv.y - r.finY) / Math.max(60, rv.v); rv.park = rv.u > st.u ? 0.62 : -0.62; rv.boostT = 0; if (r.tP == null) r.flashT = 0.3; }
      /* a close finish plays in slow motion */
      const near = r.finY - Math.min(st.dist, rv.y);
      if (!r.slowUsed && run && near > 0 && near < 190 && Math.abs(gapP) < 80) { r.slow = true; r.slowUsed = true; }
      if (r.slow && ((r.tP != null && r.tS != null) || Math.abs(gapP) > 150)) r.slow = false;
      if (r.phase === 'run' && (r.tP != null || r.tS != null)) {
        const first = Math.min(r.tP == null ? 1e9 : r.tP, r.tS == null ? 1e9 : r.tS);
        if ((r.tP != null && r.tS != null) || st.t - first > 1.6) raceResult();
      }
      if (r.phase === 'fin') {
        r.fin += dt;
        r.conf.forEach(p => { p.x += p.vx * dt; p.y += p.vy * dt; p.vy += 60 * dt; p.a += p.va * dt; }); r.conf = r.conf.filter(p => p.y < H + 20);
        if (r.fin > 3.0) finish();
      }
    }
