  /* ================= v48: the river's village market and the rock tomb (route-map stops) =================
     stop48(t) is called from nodeCard (06-voyage): 'market' / 'smugglers' open a painted village market with stall cards
     (two stalls a day), 'tomb' opens a short tomb adventure that plays out differently each visit (7 layouts).
     tb48Start('dtomb') also works (sand palette) if the desert stop ever wants it.
     A step that draws the next step in the open sheet returns '@keep' from its run() (nodeChoose keeps the sheet).
     Every step keeps an enabled [data-nchoice="0"] that moves on, so the run bot always gets through.
     Pictures are code-drawn SVG; their life is CSS only (stops48.css); reduced motion stops it. */
  function stop48(t) {
    if (!VG) return false;
    if (t === 'market' || t === 'smugglers') { mk48(t, true); return true; }
    if (t === 'tomb') { tb48Start(t); return true; }
    return false;
  }
