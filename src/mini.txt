  function pzBaboon(o) {
    o = o || {};
    pzClear();
    const n = Math.max(1, (o.n | 0) || 10), dur = Math.max(4, +o.dur || 12);
    PZ = { kind:'baboon', tok:Math.random(), timers:[], cb:o.cb, uid:++uid, n, dur, phase:'ready', caught:0, lost:0, spawned:0, act:{}, t0:0, auto:false, fin:false };
    bbBind();
    if (window.__nil && window.__nil.fn) window.__nil.fn.baboonAuto = baboonAuto;
    const kb = BB_SLOTS.map((sl, k) => { const x = sl[0], top = sl[2] < 1 ? 82 : 144, h = sl[2] < 1 ? 64 : 98, w = sl[2] < 1 ? 82 : 94; return '<button type="button" class="bbb" data-bb="' + k + '" style="left:' + ((x - w / 2) / 4) + '%;top:' + (top / 2.5) + '%;width:' + (w / 4) + '%;height:' + (h / 2.5) + '%" aria-label="' + (k + 1) + '. sandık"><kbd>' + (k + 1) + '</kbd></button>'; }).join('');
    let s = '<div class="sh-h"><h2 id="shTitle">' + esc(o.title || 'Babun baskını') + '<small>' + esc(o.sub || 'Thoth\'un yaramaz babunları rıhtımda') + '</small></h2></div>';
    s += '<p class="bbhow">Babun başını çıkarınca <b>ona dokun!</b></p>';
    s += '<div class="bbhud"><span class="bbsc">Kovulan <b id="bbOk">0</b> / ' + n + '</span><span class="bbsc lost">Kaçan <b id="bbNo">0</b></span></div>';
    s += '<div class="tbar bbbar"><i id="bbT" style="width:100%"></i></div>';
    s += '<div class="bbwrap ready" id="bbWrap">' + bbSceneSVG() + kb + '</div>';
    s += '<div class="row center"><button class="btn go1 bbgo" type="button" data-bbgo data-first>Hazırım</button></div>';
    openSheet(s, true, 'pz');
    /* a demo baboon peeks out so the goal is obvious before starting */
    const d = $('#bbS5'); if (d) d.classList.add('up', 'demo');
  }
  function bbStart() {
    if (!PZ || PZ.kind !== 'baboon' || PZ.phase !== 'ready') return;
    PZ.phase = 'run'; PZ.t0 = performance.now();
    const w = $('#bbWrap'); if (w) w.classList.remove('ready');
    const d = $('#bbS5'); if (d) d.classList.remove('up', 'demo');
    const go = document.querySelector('#sheet [data-bbgo]'); if (go) go.closest('.row').remove();
    const f = document.querySelector('#sheet .bbb'); if (f) f.focus({ preventScroll:true });
    play('click');
    const tok = PZ.tok, total = PZ.dur * 1000, gap = (total - 1700) / PZ.n;
    for (let k = 0; k < PZ.n; k++) {
      const at = 700 + k * gap + (Math.random() - 0.5) * gap * 0.5;
      PZ.timers.push(setTimeout(() => { if (!PZ || PZ.tok !== tok) return; bbSpawn(); }, Math.max(300, at)));
    }
    const tick = () => {
      if (!PZ || PZ.tok !== tok || PZ.phase !== 'run') return;
      const el = performance.now() - PZ.t0, bar = $('#bbT');
      if (bar) bar.style.width = Math.max(0, 100 - el / total * 100) + '%';
      PZ.timers.push(setTimeout(tick, 100));
    };
    tick();
  }
  function bbSpawn() {
    const free = BB_SLOTS.map((_, k) => k).filter(k => !PZ.act[k]);
    if (!free.length) { PZ.timers.push(setTimeout(() => { if (PZ && PZ.kind === 'baboon') bbSpawn(); }, 200)); return; }
    const k = free[Math.floor(Math.random() * free.length)], id = ++PZ.spawned, up = 900 + Math.random() * 400, tok = PZ.tok;
    PZ.act[k] = { id };
    const el = $('#bbS' + k); if (el) { el.classList.remove('hit', 'steal'); el.classList.add('up'); }
    Snd.note(86 + Math.floor(Math.random() * 5));
    PZ.timers.push(setTimeout(() => { if (!PZ || PZ.tok !== tok) return; const a = PZ.act[k]; if (a && a.id === id) bbSteal(k); }, up));
    if (PZ.auto) PZ.timers.push(setTimeout(() => { if (!PZ || PZ.tok !== tok) return; const b = document.querySelector('#sheet [data-bb="' + k + '"]'); if (b) b.click(); else bbTap(k); }, 220 + Math.random() * 200));
  }
  /* the tap path shared by fingers, mouse, keys and the auto helper */
  function bbTap(k) {
    if (!PZ || PZ.kind !== 'baboon') return;
    if (PZ.phase === 'ready') { if (k === 5) bbStart(); return; }
    if (PZ.phase !== 'run') return;
    const a = PZ.act[k]; if (!a || a.done) return;
    a.done = true; PZ.caught++;
    const el = $('#bbS' + k); if (el) { el.classList.remove('up'); el.classList.add('hit'); }
    play('pick'); Snd.note(98); buzz('pick');
    const ok = $('#bbOk'); if (ok) ok.textContent = PZ.caught;
    bbFree(k, 420);
  }
  function bbSteal(k) {
    const a = PZ.act[k]; if (!a || a.done) return;
    a.done = true; PZ.lost++;
    const el = $('#bbS' + k); if (el) { el.classList.remove('up'); el.classList.add('steal', k % 4 < 2 ? 'left' : 'right'); }
    play('no');
    const no = $('#bbNo'); if (no) no.textContent = PZ.lost;
    bbFree(k, 750);
  }
  function bbFree(k, ms) {
    const tok = PZ.tok;
    PZ.timers.push(setTimeout(() => {
      if (!PZ || PZ.tok !== tok) return;
      const el = $('#bbS' + k), ape = el && el.querySelector('.bbape');
      if (ape) ape.style.transition = 'none';
      if (el) { el.classList.remove('hit', 'steal', 'left', 'right'); void el.getBoundingClientRect(); }
      if (ape) ape.style.transition = '';
      delete PZ.act[k];
      if (PZ.spawned >= PZ.n && PZ.phase === 'run' && !Object.keys(PZ.act).length) bbEnd();
    }, ms));
  }
  function bbEnd() {
    PZ.phase = 'end';
    const c = PZ.caught, n = PZ.n, tok = PZ.tok, bar = $('#bbT'); if (bar) bar.style.width = '0%';
    const w = $('#bbWrap');
    const t = c === n ? 'Hepsini kovdun!' : c >= n * 0.7 ? 'Aferin!' : c >= n * 0.4 ? 'Fena değil' : 'Babunlar kazandı';
    if (w) w.insertAdjacentHTML('beforeend', '<div class="bbend ' + (c >= n * 0.7 ? 'good' : 'bad') + '"><b>' + c + ' / ' + n + '</b><span>' + t + '</span></div>');
    play(c >= n * 0.7 ? (c === n ? 'fanfare' : 'good') : 'bad');
    PZ.timers.push(setTimeout(() => {
      if (!PZ || PZ.tok !== tok || PZ.fin) return;
      PZ.fin = true; const cb = PZ.cb; pzClear();
      try { if (typeof cb === 'function') cb(c, n); }
      finally { if (document.querySelector('#sheet .bbwrap')) closeSheet(); }
    }, 1300));
  }
  /* test helper: taps every baboon shortly after it pops up */
  function baboonAuto() {
    if (!PZ || PZ.kind !== 'baboon') return false;
    PZ.auto = true;
    if (PZ.phase === 'ready') { const b = document.querySelector('#sheet [data-bbgo]'); if (b) b.click(); else bbStart(); }
    else Object.keys(PZ.act).forEach(k => { if (!PZ.act[k].done) bbTap(+k); });
    return true;
  }
  function bbBind() {
    const sh = $('#sheet');
    if (sh && !sh._bbBound) {
      sh._bbBound = true;
      /* pointerdown is quicker than click for a reflex game; click still covers keyboard activation */
      sh.addEventListener('pointerdown', e => {
        if (!PZ || PZ.kind !== 'baboon' || PZ.phase !== 'run') return;
        const b = e.target.closest && e.target.closest('[data-bb]'); if (!b) return;
        e.preventDefault(); b._pd = performance.now(); bbTap(+b.dataset.bb);
      });
      sh.addEventListener('click', e => {
        if (!PZ || PZ.kind !== 'baboon') return;
        const b = e.target.closest && e.target.closest('button'); if (!b || b.disabled) return;
        if (b.hasAttribute('data-bbgo')) bbStart();
        else if (b.dataset.bb != null) { if (b._pd && performance.now() - b._pd < 700) { b._pd = 0; return; } bbTap(+b.dataset.bb); }
      });
    }
    if (!document._bbKey) {
      document._bbKey = true;
      document.addEventListener('keydown', e => {
        if (!PZ || PZ.kind !== 'baboon' || !/^[1-8]$/.test(e.key)) return;
        e.preventDefault(); bbTap(+e.key - 1);
      });
    }
  }
