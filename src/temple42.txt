  function slNbr(G, e) { const r = Math.floor(e / G.c), c = e % G.c, out = []; if (r > 0) out.push(e - G.c); if (r < G.r - 1) out.push(e + G.c); if (c > 0) out.push(e - 1); if (c < G.c - 1) out.push(e + 1); return out; }
  /* A* over the board (tile in each cell, 0 = gap); falls back to null when it grows too big */
  function slSolve(G, b0) {
    const R = G.r, C = G.c, N = R * C, goal = b0.map((_, i) => (i + 1) % N).join(',');
    const h = b => { let s = 0; for (let i = 0; i < N; i++) { const t = b[i]; if (!t) continue; const g = t - 1; s += Math.abs(Math.floor(g / C) - Math.floor(i / C)) + Math.abs(g % C - i % C); } return s; };
    const start = b0.slice(), key0 = start.join(','); if (key0 === goal) return [];
    const heap = [], seen = new Map(); let n = 0;
    const push = x => { heap.push(x); let i = heap.length - 1; while (i > 0) { const p = (i - 1) >> 1; if (heap[p].f <= heap[i].f) break; [heap[p], heap[i]] = [heap[i], heap[p]]; i = p; } };
    const pop = () => { const top = heap[0], last = heap.pop(); if (heap.length) { heap[0] = last; let i = 0; for (;;) { const l = i * 2 + 1, r = l + 1; let m = i; if (l < heap.length && heap[l].f < heap[m].f) m = l; if (r < heap.length && heap[r].f < heap[m].f) m = r; if (m === i) break; [heap[m], heap[i]] = [heap[i], heap[m]]; i = m; } } return top; };
    push({ b:start, e:start.indexOf(0), g:0, f:h(start), path:null, mv:-1 }); seen.set(key0, 0);
    while (heap.length && n++ < 160000) {
      const s = pop(), k = s.b.join(',');
      if (k === goal) { const out = []; let p = s; while (p && p.mv >= 0) { out.push(p.mv); p = p.path; } return out.reverse(); }
      for (const q of slNbr(G, s.e)) {
        const b = s.b.slice(); b[s.e] = b[q]; b[q] = 0;
        const kk = b.join(','), g = s.g + 1;
        if (seen.has(kk) && seen.get(kk) <= g) continue;
        seen.set(kk, g); push({ b, e:q, g, f:g + h(b), path:s, mv:b[s.e] });
      }
    }
    return null;
  }

  const stStar = (r, big, c) => '<path d="M0 ' + (-r) + 'L' + (r * 0.29) + ' ' + (-r * 0.36) + 'L' + r + ' ' + (-r * 0.31) + 'L' + (r * 0.47) + ' ' + (r * 0.15) + 'L' + (r * 0.59) + ' ' + (r * 0.81) + 'L0 ' + (r * 0.45) + 'L' + (-r * 0.59) + ' ' + (r * 0.81) + 'L' + (-r * 0.47) + ' ' + (r * 0.15) + 'L' + (-r) + ' ' + (-r * 0.31) + 'L' + (-r * 0.29) + ' ' + (-r * 0.36) + 'Z" fill="' + (c || (big ? '#FFD24A' : '#C9B37A')) + '"' + (big ? ' stroke="' + INK + '" stroke-width="1.4"' : ' opacity=".65"') + '/>';
  /* tapping ring i turns these rings one step */
  function stLinks(G, i) { if (G.link === 0) return [i]; if (G.link === 1) return i < G.k - 1 ? [i, i + 1] : [i]; return [i, (i + 1) % G.k]; }
  function stApply(G, o, i, back) { const n = o.slice(); stLinks(G, i).forEach(j => { n[j] = (n[j] + (back ? ST_P - 1 : 1)) % ST_P; }); return n; }
  function stBFS(G, o0) {
    const key = o => o.join(','), seen = new Map([[key(o0), null]]), q = [o0];
    if (o0.every(x => !x)) return [];
    while (q.length) {
      const o = q.shift();
      for (let i = 0; i < G.k; i++) {
        const n = stApply(G, o, i), kk = key(n); if (seen.has(kk)) continue;
        seen.set(kk, [key(o), i]);
        if (n.every(x => !x)) { const out = []; let c = kk; while (seen.get(c)) { const [p, t] = seen.get(c); out.push(t); c = p; } return out.reverse(); }
        q.push(n);
      }
    }
    return null;
  }
  TPK.stars = {
    build(level, rnd) {
      const L = ST_LV[level], k = L.k, con = ST_CON[tpRI(rnd, ST_CON.length)], band = (ST_R1 - ST_R0) / k;
      /* each star goes to a ring by how far it is from the centre (sorted, split evenly) */
      const pol = con.s.map(([x, y], i) => ({ i, a:Math.atan2(y, x), r:Math.hypot(x, y) })).sort((p, q) => p.r - q.r);
      const stars = [];
      pol.forEach((p, n) => { const ring = Math.min(k - 1, Math.floor(n * k / pol.length)); stars[p.i] = { ring, a:p.a, r:ST_R0 + band * (ring + 0.3 + 0.4 * ((p.r * 7) % 1)) }; });
      const dec = []; for (let r = 0; r < k; r++) for (let j = 0; j < 3; j++) dec.push({ ring:r, a:rnd() * Math.PI * 2, r:ST_R0 + band * (r + 0.25 + rnd() * 0.5), sz:2.4 + rnd() * 2 });
      const G = { k, link:L.link, con, stars, dec, band, o:Array(k).fill(0), ang:Array(k).fill(0), par:0 };
      for (let tries = 0; tries < 200; tries++) {
        let o = Array(k).fill(0);
        /* scrambled backwards, so turning forward undoes it in about d taps */
        for (let t = 0; t < L.d; t++) o = stApply(G, o, tpRI(rnd, k), true);
        if (o.some(x => !x) && tries < 150) continue;
        const sol = stBFS(G, o);
        if (sol && sol.length >= L.d - 1) { G.o = o; G.ang = o.map(x => x * 45); G.par = sol.length; return G; }
      }
      G.o = Array(k).fill(0).map((_, i) => i === 0 ? 3 : 0); G.ang = G.o.map(x => x * 45); G.par = (stBFS(G, G.o) || []).length;
      return G;
    },
    goal:G => 'Halkalara dokun, bir adım dönsünler. Parlak yıldızlar <b>altın halkaların</b> içine otursun: ' + esc(G.con.n) + '.' + (G.link ? ' <b>Zincirli</b> halkalar birlikte döner.' : ''),
    html(G) {
      const u = PZ.uid;
      let s = '<svg class="stsvg" viewBox="0 0 340 340" role="img" aria-label="Yıldız tavanı: ' + esc(G.con.n) + '"><defs><radialGradient id="stSky' + u + '"><stop offset="0" stop-color="#2E4C92"/><stop offset="1" stop-color="#0F1D45"/></radialGradient></defs>';
      s += '<rect width="340" height="340" rx="16" fill="#C9A46A"/><rect x="5" y="5" width="330" height="330" rx="13" fill="none" stroke="#8E6A3C" stroke-width="2.5"/>';
      s += '<circle cx="' + ST_C + '" cy="' + ST_C + '" r="' + (ST_R1 + 6) + '" fill="#E9C46A" stroke="' + INK + '" stroke-width="3"/>';
      for (let i = G.k - 1; i >= 0; i--) {
        const r0 = ST_R0 + G.band * i, r1 = r0 + G.band, c = i % 2 ? '#1B2C63' : '#22397A';
        s += '<g class="stg" id="stR' + i + '" data-tpr="' + i + '" role="button" tabindex="0" aria-label="' + (i + 1) + '. halka: döndür" style="transform:rotate(' + G.ang[i] + 'deg)">';
        s += '<path d="M' + (ST_C + r1) + ' ' + ST_C + 'A' + r1 + ' ' + r1 + ' 0 1 0 ' + (ST_C - r1) + ' ' + ST_C + 'A' + r1 + ' ' + r1 + ' 0 1 0 ' + (ST_C + r1) + ' ' + ST_C + 'ZM' + (ST_C + r0) + ' ' + ST_C + 'A' + r0 + ' ' + r0 + ' 0 1 1 ' + (ST_C - r0) + ' ' + ST_C + 'A' + r0 + ' ' + r0 + ' 0 1 1 ' + (ST_C + r0) + ' ' + ST_C + 'Z" fill="' + c + '" fill-rule="evenodd" stroke="#D9A43A" stroke-width="2"/>';
        /* little notches so the turning shows */
        for (let j = 0; j < ST_P; j++) { const a = j * Math.PI / 4 + Math.PI / 8; s += '<circle cx="' + (ST_C + Math.cos(a) * (r1 - 4)).toFixed(1) + '" cy="' + (ST_C + Math.sin(a) * (r1 - 4)).toFixed(1) + '" r="1.6" fill="#D9A43A" opacity=".7"/>'; }
        G.dec.filter(d => d.ring === i).forEach(d => { s += '<g transform="translate(' + (ST_C + Math.cos(d.a) * d.r).toFixed(1) + ' ' + (ST_C + Math.sin(d.a) * d.r).toFixed(1) + ')">' + stStar(d.sz, false) + '</g>'; });
        G.stars.forEach(st => { if (st.ring === i) s += '<g class="stbig" transform="translate(' + (ST_C + Math.cos(st.a) * st.r).toFixed(1) + ' ' + (ST_C + Math.sin(st.a) * st.r).toFixed(1) + ')"><circle r="10" fill="#FFE9A0" opacity=".25"/>' + stStar(7.5, true) + '</g>'; });
        s += '</g>';
      }
      /* the hub: Nut's sun */
      s += '<circle cx="' + ST_C + '" cy="' + ST_C + '" r="' + (ST_R0 - 2) + '" fill="#F2A81E" stroke="' + INK + '" stroke-width="2.4"/><circle cx="' + ST_C + '" cy="' + ST_C + '" r="' + (ST_R0 - 12) + '" fill="#FFD24A"/>';
      /* the target: gold rings where the bright stars must sit, and the constellation lines */
      const P = st => [(ST_C + Math.cos(st.a) * st.r).toFixed(1), (ST_C + Math.sin(st.a) * st.r).toFixed(1)];
      s += '<g class="stgoal" pointer-events="none">' + G.con.l.map(([a, b]) => { const p = P(G.stars[a]), q = P(G.stars[b]); return '<path d="M' + p.join(' ') + 'L' + q.join(' ') + '"/>'; }).join('');
      s += G.stars.map(st => { const p = P(st); return '<circle cx="' + p[0] + '" cy="' + p[1] + '" r="11"/>'; }).join('') + '</g>';
      /* chains between rings that turn together */
      if (G.link) for (let i = 0; i < G.k; i++) { const ls = stLinks(G, i); if (ls.length < 2) continue; const j = ls[1], ra = ST_R0 + G.band * (i + 0.5), rb = ST_R0 + G.band * (j + 0.5), a = Math.PI / 2 + (j < i ? 0.32 : 0) + i * 0.05; s += '<g class="stchain" pointer-events="none">' + tpDbl('M' + (ST_C + Math.cos(a) * ra).toFixed(1) + ' ' + (ST_C + Math.sin(a) * ra).toFixed(1) + 'L' + (ST_C + Math.cos(a) * rb).toFixed(1) + ' ' + (ST_C + Math.sin(a) * rb).toFixed(1), '#E9C46A', 3) + '<circle cx="' + (ST_C + Math.cos(a) * ra).toFixed(1) + '" cy="' + (ST_C + Math.sin(a) * ra).toFixed(1) + '" r="4" fill="#E9C46A" stroke="' + INK + '" stroke-width="1.6"/><circle cx="' + (ST_C + Math.cos(a) * rb).toFixed(1) + '" cy="' + (ST_C + Math.sin(a) * rb).toFixed(1) + '" r="4" fill="#E9C46A" stroke="' + INK + '" stroke-width="1.6"/></g>'; }
      return s + '</svg>';
    },
    draw(G) {
      for (let i = 0; i < G.k; i++) { const el = $('#stR' + i); if (el) { el.style.transform = 'rotate(' + G.ang[i] + 'deg)'; el.classList.toggle('ok', G.o[i] === 0); } }
    },
    perfect:G => PZ.moves <= G.par + 3,
    tap(G, t) {
      const i = +t.dataset.tpr; if (!(i >= 0)) return;
      stLinks(G, i).forEach(j => { G.o[j] = (G.o[j] + 1) % ST_P; G.ang[j] += 45; });
      PZ.moves++; play('tick');
      TPK.stars.draw(G); tpStatus();
      if (G.o.every(x => !x)) tpWin(G.con.n + ' parlıyor!');
    },
    hint(G) {
      const sol = stBFS(G, G.o); if (!sol || !sol.length) return '';
      const el = $('#stR' + sol[0]); tpGlow(el); TPK.stars.tap(G, el || { dataset:{ tpr:sol[0] } });
      return (sol[0] + 1) + '. halkayı çevirdim.';
    },
    solve(G) { return (stBFS(G, G.o) || []).map(i => '[data-tpr="' + i + '"]'); }
  };
