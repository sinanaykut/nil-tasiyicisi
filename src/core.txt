function haggle(S, q, bo) {
  if (!canHaggle(S)) return null;
  const wins = S.stats.haggles || 0, b = q === 'miss' ? 0 : bo != null ? bo : q === 'perfect' ? Math.max(0.08, 0.12 - wins * 0.01) : 0.05;
  S.haggle = { city:S.city, day:S.day, b };
  if (q === 'miss') S.trust[S.city] = Math.max(0, S.trust[S.city] - 60);
  S.stats.haggles = (S.stats.haggles || 0) + (b > 0 ? 1 : 0);
  S.log.push({ d:S.day, t: b ? MERCHANT_NAMES[S.city] + ' ile pazarlıkta %' + Math.round(b * 100) + ' koparttın.' : MERCHANT_NAMES[S.city] + ' pazarlığa kızdı.' });
  return b;
}

function stockMax(S, i, gid) {
  const m = CITIES[i].mult[gid]; if (m == null) return 0;
  const base = S.stockBase ? S.stockBase[i][gid] : 0;
  let k = 1; (S.mods || []).forEach(md => { if (md.city === i && md.stock > 1 && (!md.g || md.g === gid) && S.day >= md.from && S.day <= md.until) k = Math.max(k, md.stock); });
  return Math.round((base + ROOM[gid] * stockWant(m)) * k);
}

function stockDay(S) {
  if (!S.stockBase) { genStock(S); return; }

function stockMods(S) {
  (S.mods || []).forEach(m => {
    if (m.stock == null || S.day < m.from || S.day > m.until) return;
    (m.g ? [m.g] : GOODS.map(g => g.id)).filter(g => deals(m.city, g)).forEach(g => {
      const t = Math.round(S.stockBase[m.city][g] * m.stock), cur = S.stock[m.city][g];
      if (m.stock < 1 && cur > t) S.stock[m.city][g] = Math.max(1, t);
      if (m.stock > 1 && cur < t) S.stock[m.city][g] = Math.min(stockMax(S, m.city, g), t);
    });
  });
}

function boughtHere(S, g) { const t = S.today; return t && t.d === S.day && t.c === S.city && t.g[g] ? t.g[g] : 0; }
function noteBuy(S, g, n) { if (!S.today || S.today.d !== S.day || S.today.c !== S.city) S.today = { d:S.day, c:S.city, g:{} }; S.today.g[g] = (S.today.g[g] || 0) + n; }
function sameDayCap(S, g) { const it = S.inv[g]; return boughtHere(S, g) && it.n ? Math.max(1, Math.floor(it.cost / it.n)) : 0; }
function quoteBuy(S, g, q) {
  if (!deals(S.city, g) || gById(g).craft) return { n:0, cost:0, reason:'market', p:S.prices[S.city][g] };
  const room = freeCap(S);
  let n = 0, cost = 0, reason = '', p = S.prices[S.city][g];
  const k = 1 - buyDisc(S);
  while (n < q) {
    if (n >= room) { reason = 'cap'; break; }
    if (S.stock && n >= S.stock[S.city][g]) { reason = 'stock'; break; }
    const u = Math.max(1, Math.round(p * k));
    if (S.cash - cost < u) { reason = 'cash'; break; }
    cost += u; n++; p *= 1 + IMPACT;
  }
  return { n, cost, reason, p };
}
/* force: the end-of-run sale (the treasury takes everything, the storeroom does not matter) */
function quoteSell(S, g, q, force) {
  if (!deals(S.city, g)) return { n:0, rev:0, p:S.prices[S.city][g], reason:'market' };
  const held = S.inv[g].n, room = force ? Infinity : stockRoom(S, g);
  let reason = !held ? 'none' : '';
  q = Math.min(q, held);
  if (q > room) { q = room; reason = 'full'; }
  let n = 0, rev = 0, p = S.prices[S.city][g];
  const k = 1 + sellBonus(S), cap = sameDayCap(S, g);
  while (n < q) { rev += cap ? Math.min(cap, Math.round(p * k)) : Math.round(p * k); n++; p *= 1 - IMPACT; }
  return { n, rev, p, flip:!!cap, reason, room };
}
function buy(S, g, q) {
  const r = quoteBuy(S, g, q);
  if (r.n > 0) { S.cash -= r.cost; S.inv[g].n += r.n; S.inv[g].cost += r.cost; S.prices[S.city][g] = r.p; if (S.stock) S.stock[S.city][g] -= r.n; noteBuy(S, g, r.n); r.lvUp = addTrust(S, S.city, r.cost / 70); }
  return r;
}
function sell(S, g, q, force) {
  const it = S.inv[g], avg = it.n ? it.cost / it.n : 0;
  const r = quoteSell(S, g, q, force);
  if (r.n > 0) {
    it.cost = Math.max(0, it.cost - avg * r.n); it.n -= r.n;
    if (S.stock && !force) S.stock[S.city][g] += r.n;
    if (it.n === 0) it.cost = 0;
    S.cash += r.rev; S.prices[S.city][g] = r.p;
    r.profit = Math.round(r.rev - avg * r.n);
    if (!S.over && (!S.stats.best || r.profit > S.stats.best.p)) S.stats.best = { p:r.profit, g, city:S.city, d:S.day };
    if (!S.over && !r.flip) r.lvUp = addTrust(S, S.city, r.rev / 35);
    else if (r.flip) S.trust[S.city] = Math.max(0, S.trust[S.city] - r.rev / 70);
  }
  return r;
}

function dealBuy(S, g, n, unit) {
  if (!deals(S.city, g) || gById(g).craft || S.over) return { n:0, cost:0, reason:'market' };
  /* rounded against the player, so a 0% deal never beats the normal one */
  unit = Math.max(1, Math.ceil(unit - 1e-6));
  let k = Math.min(n | 0, Math.max(0, freeCap(S)), S.stock ? S.stock[S.city][g] : Infinity, Math.floor(S.cash / unit));
  const reason = k < n ? (freeCap(S) < n ? 'cap' : S.stock && S.stock[S.city][g] < n ? 'stock' : 'cash') : '';
  k = Math.max(0, k); if (!k) return { n:0, cost:0, reason };
  const cost = k * unit;
  S.cash -= cost; S.inv[g].n += k; S.inv[g].cost += cost; if (S.stock) S.stock[S.city][g] -= k;
  S.prices[S.city][g] *= Math.pow(1 + IMPACT, k);

}
function dealSell(S, g, n, unit) {
  if (!deals(S.city, g) || S.over) return { n:0, rev:0, reason:'market' };
  const it = S.inv[g], avg = it.n ? it.cost / it.n : 0, room = stockRoom(S, g), k = Math.max(0, Math.min(n | 0, it.n, room)); unit = Math.max(1, Math.floor(unit + 1e-6));
  const reason = k < Math.min(n | 0, it.n) ? 'full' : '';
  if (!k) return { n:0, rev:0, reason };
  const cap = sameDayCap(S, g); if (cap) unit = Math.min(unit, cap);
  const rev = k * unit;
  it.cost = Math.max(0, it.cost - avg * k); it.n -= k; if (!it.n) it.cost = 0;
  S.cash += rev; S.prices[S.city][g] *= Math.pow(1 - IMPACT, k); if (S.stock) S.stock[S.city][g] += k;
  const r = { n:k, rev, unit, reason, profit:Math.round(rev - avg * k) };
  if (!S.stats.best || r.profit > S.stats.best.p) S.stats.best = { p:r.profit, g, city:S.city, d:S.day };
  if (!cap) r.lvUp = addTrust(S, S.city, rev / 35);

  return r;
}

  }
  if (S.quest && S.quest.st === 'active' && S.quest.id === 'kedi' && S.quest.cargo && rng() < 0.75) chaser = 'djedi';
  const sp = S.rival.plan, rival = !desert && !!sp && dest === sp.to && sp.arrive >= S.day && Math.abs(S.day + days - sp.arrive) <= 1, bless = S.bless; S.bless = null;
  if (bless === 'bastet') chaser = null;
  if (S.contra && rng() < 0.85) chaser = 'medjay';
  if (desert && chaser && chaser !== 'bedevi') chaser = 'bedevi';
  return { days, days0:days, chaser, dest, from:S.city, speed:desert ? 0 : Math.max(0, Math.min(3, sailOf(S))), rowers:desert ? 0 : S.up.kurek + boatOf(S).row, extra:(S.perk.tayfa || 0) + (S.hero === 'bakenre' ? 1 : 0), hero:S.hero, first:!S.stats.trips, hp:desert ? 3 : hullNow(S) + (bless === 'hapi' ? 1 : 0), hpMax:desert ? 3 : maxHp(S) + (bless === 'hapi' ? 1 : 0), dmg:desert ? 0 : dmgStage(S), boat:S.boat, dev:Object.assign({}, S.dev), terrain:desert ? 'desert' : 'river', donkeys:S.caravan ? S.caravan.donkeys : 0, val, contra:!!S.contra, rival, rapid, bless, season:seasonIdx(S), seed:Math.floor(rng() * 1e9) };
}
/* quick crossing for players who skip the river run: on average worse than playing */
function quickSea(plan) {
  /* the very first crossing of a newcomer is gentle: no wreck, no ambush */
  if (plan.first) return { hits:0, wreck:false, caught:false, escaped:!!plan.chaser, coins:Math.round(10 + rng() * 30), crates:0, beatRival:false, quick:true };
  const hits = Math.min(plan.hp, Math.floor(rng() * (2 + plan.days + (plan.rapid ? 2 : 0))));
  const caught = plan.chaser ? rng() < 0.6 - 0.1 * plan.rowers - 0.08 * plan.speed : false;
  return { hits, wreck: hits >= plan.hp, caught, escaped: !!plan.chaser && !caught, coins: Math.round(rng() * 40), crates:0, beatRival: plan.rival ? rng() < 0.3 : false, quick:true };
}
function applySea(S, plan, res) {
  const out = [], dz = plan.terrain === 'desert';
  S.stats.trips++; S.stats.hits += res.hits; if (plan.rapid) S.stats.rapids++;
  S.stats.vlog = (S.stats.vlog || '') + (res.wreck ? 'x' : res.caught ? 'c' : res.hits ? 'h' : 'p');

  if (res.wreck) {
    const lost = loseFrac(S, 0.4);
    out.push(dz ? { kind:'bad', title:'Kervan dağıldı', text:'Eşekler kumda dağıldı, toparlaman saatler sürdü. Bulunamayan yük: ' + (lost.join(', ') || 'yok') + '.' } : { kind:'bad', title:'Gemi karaya oturdu', text:'Teknen ağır hasar aldı. Kurtarılamayan yük: ' + (lost.join(', ') || 'yok') + '.' });
  } else if (res.hits > 0) {
    const lostBy = {};
    for (let k = 0; k < res.hits; k++) {

      if (!owned.length) break;
      const g = pick(owned), l = loseGoods(S, g.id, Math.max(1, Math.round(S.inv[g.id].n * 0.15)), true);
      lostBy[g.id] = (lostBy[g.id] || 0) + l;
    }

    out.push({ kind:'bad', title:res.hits + ' darbe aldın', text: lost.length ? (dz ? 'Eşekler tökezledi, kuma düşen yük: ' : 'Sarsıntıda suya düşen yük: ') + lost.join(', ') + '.' : 'Neyse ki kaybedecek yükün yoktu.' });
  } else { S.stats.perfect++; out.push({ kind:'good', title:'Kusursuz yolculuk', text:'Tek bir çizik bile almadan vardın. Tayfa seninle gurur duyuyor.' }); }
  if (plan.chaser) {
    if (res.caught) {
      S.stats.caught++;
      if (plan.chaser === 'bedevi') { const lost = loseFrac(S, 0.3); if (S.caravan) S.caravan.water = Math.max(0, S.caravan.water - 1); out.push({ kind:'bad', title:'Çöl akıncıları yetişti', text:'Yüklerini ve bir tulum suyunu aldılar: ' + (lost.join(', ') || 'boş heybelere güldüler') + '.' }); }
      else if (plan.chaser === 'haydut') { const lost = loseFrac(S, 0.3); out.push({ kind:'bad', title:'Haydutlar yetişti', text:'Kervanını yağmaladılar: ' + (lost.join(', ') || 'boş ambara küfrettiler') + '.' }); }
      else if (plan.chaser === 'medjay') {
        if (S.contra) { const f = Math.min(S.cash, Math.round(S.cash * 0.25 + 300)); S.cash -= f; S.contra = null; out.push({ kind:'bad', title:'Medjay kaçak malı buldu', text:'Hapu\'nun sandıklarına el koydular ve ' + fmtN(f) + ' deben ceza kestiler. Hapu da çok kızacak.' }); S.debt += 800; S.log.push({ d:S.day, t:'Kaçak sandıklar yakalandı. Hapu borcuna 800 deben "zarar" ekledi.' }); }
        else { const f = Math.min(S.cash, Math.round(S.cash * 0.15 + 50)); S.cash -= f; out.push({ kind:'bad', title:'Medjay seni durdurdu', text:'Kaçakçılık şüphesiyle ' + fmtN(f) + ' deben ceza kestiler.' }); }
      }
      else if (plan.chaser === 'djedi') {
        if (S.quest && S.quest.cargo) { S.quest.cargo = 0; S.quest.f.stolen = true; out.push({ kind:'bad', title:'Djedi\'nin adamları heykeli kaptı', text:'Altın kedi gitti! Kâtip Merenre\'nin casuslarına göre Djedi onu Vaset\'te satmaya çalışacak.' }); }
        else { const lost = loseFrac(S, 0.2); out.push({ kind:'bad', title:'Djedi\'nin adamları', text:'Heykeli bulamayınca öfkelerini ambardan çıkardılar: ' + (lost.join(', ') || 'boş ambar') + '.' }); }
      }
      else { const take = Math.min(S.cash, Math.round(S.debt * 0.3)); S.cash -= take; S.debt -= take; out.push({ kind:'bad', title:'Tefecinin adamları', text:'Hapu\'nun adamları seni yakaladı ve zorla ' + fmtN(take) + ' deben tahsil etti. Borcundan düştüler, en azından.' }); }
    } else {
      S.stats.escapes++;
      const who = { haydut:'Haydutları', medjay:'Medjay devriyesini', tefeci:'Tefecinin adamlarını', djedi:'Djedi\'nin adamlarını', bedevi:'Çöl akıncılarını' }[plan.chaser];
      out.push({ kind:'good', title:'Kaçtın!', text: who + (dz ? ' kum fırtınasının içinde kaybettin.' : ' Nil\'in tozunda bıraktın.') });
    }
  }
  if (res.spotted) {
    if (S.contra) { const f = Math.min(S.cash, Math.round(S.cash * 0.25 + 300)); S.cash -= f; S.contra = null; S.debt += 800; out.push({ kind:'bad', title:'Devriye kaçak malı buldu', text:'Gece geçişinde Medjay seni gördü, sandıklara el koydu ve ' + fmtN(f) + ' deben ceza kesti. Hapu da borcuna 800 ekledi.' }); }
    else { const f = Math.min(S.cash, Math.round(S.cash * 0.08 + 40)); S.cash -= f; out.push({ kind:'bad', title:'Devriye fark etti', text:'Gece geçişinde Medjay devriyesi seni gördü: ' + fmtN(f) + ' deben "geçiş bahşişi".' }); }
  }
  if (res.overboard) {
    const lostBy = {};
    for (let k = 0; k < res.overboard; k++) { const owned = GOODS.filter(g => S.inv[g.id].n > 0); if (!owned.length) break; const g = pick(owned), l = loseGoods(S, g.id, Math.max(1, Math.round(S.inv[g.id].n * 0.12)), true); lostBy[g.id] = (lostBy[g.id] || 0) + l; }
    const lost = Object.keys(lostBy).map(id => lostBy[id] + ' ' + gById(id).short.toLowerCase());
    out.push({ kind:'bad', title:'Fırtınada sandık düştü', text:res.overboard + ' sandık güverteden kaydı. ' + (lost.length ? 'Denize giden: ' + lost.join(', ') + '.' : 'Neyse ki boştular.') });
  }
  if (res.notes) res.notes.forEach(n => out.push(n));
  if (res.coins) { S.cash += res.coins; } S.stats.river = (S.stats.river || 0) + (res.coins || 0) + (res.nearCoins || 0);
  /* crates are loaded when they are picked up (res.got); an older result only carries a count and is loaded here */
  let left = res.left || 0;
  if (res.crates || res.got) {
    let gotBy = res.got;
    if (!gotBy) { gotBy = {}; for (let k = 0; k < res.crates; k++) { const t = crateTake(S); if (t) gotBy[t.g] = (gotBy[t.g] || 0) + t.q; else left++; } }
    const got = goodsList(gotBy);
    S.stats.crates += res.crates || 0;
    if (got.length || res.coins) out.push({ kind:'good', title:dz ? 'Yolda bulduğun' : 'Nehirden topladıkların', text:(got.length ? got.join(', ') : '') + (res.coins ? (got.length ? ' ve ' : '') + fmtN(res.coins) + ' deben' : '') + '.' });
  } else if (res.coins) out.push({ kind:'good', title:dz ? 'Yolda bulduğun' : 'Nehirden topladıkların', text:fmtN(res.coins) + ' deben.' });
  if (left > 0) out.push({ kind:'bad', title:'Ambar dolu', text:left + ' ' + (dz ? 'yük denki' : 'sandık') + ' geride kaldı: ' + (dz ? 'eşeklerde' : 'ambarda') + ' yer yoktu. Para ve kutsamalar yine de senin.' });
  if (res.water && S.caravan) { S.caravan.water += res.water; out.push({ kind:'good', title:'Su tulumları', text:res.water + ' tulum su buldun: ' + res.water + ' gün daha dayanırsın.' }); }
  if (res.ramUsed) out.push({ kind:'good', title:'Bakır mahmuz', text:'İlk darbeyi bakır burun karşıladı.' });
  return out;
}
function arrive(S, plan, res) {
  S.seen[S.city] = snap(S, S.city);
  const lost0 = S.stats.lostVal || 0, cash0 = S.cash;
  const out = applySea(S, plan, res);
  const river = (res.coins || 0) + (res.nearCoins || 0), ledger = { river, lost:(S.stats.lostVal || 0) - lost0, fines:Math.max(0, river - (S.cash - cash0)) };
  const d = plan.days, seasonBefore = seasonIdx(S);
  if (plan.terrain !== 'desert') { const left = (res.hp != null ? res.hp : plan.hp - (res.hits || 0)) - (plan.bless === 'hapi' ? 1 : 0); S.hull = res.wreck ? 1 : Math.max(1, Math.min(maxHp(S), left)); if (dmgStage(S) < 2) S.repNag = false; }
  caravanDays(S, d, out);
  S.day += d; S.city = plan.dest; S.edge = null;
  if (S.caravan && isRiver(S.city)) { endCaravan(S); out.push({ kind:'good', title:'Kervan döndü', text:'Eşekler sahiplerine, yükler teknene. ' + CITIES[S.city].name + ' iskelesinde teknen seni bekliyordu.' }); }
  else if (S.caravan && /^(siwa|kharga)$/.test(CITIES[S.city].id)) { const need = CITIES[S.city].days + 1; if (S.caravan.water < need) { S.caravan.water = need; out.push({ kind:'good', title:'Vaha kuyuları', text:'Tulumlar vahanın tatlı suyuyla doldu: dönüş için ' + need + ' günlük su.' }); } }
  const before = S.debt;
  S.debt = Math.round(S.debt * Math.pow(1 + S.rate, d)); S.stats.interest = (S.stats.interest || 0) + (S.debt - before);
  const R = S.rival;
  R.cash = Math.round(R.cash * (1 + SETHI_RATE * d * (0.6 + rng() * 0.8)));
  if (plan.rival && R.plan) {
    S.stats.races++;
    const gn = gById(R.plan.g).short.toLowerCase();
    if (res.beatRival) { S.stats.raceWins++; S.edge = S.city; R.cash = Math.round(R.cash * 0.97); R.plan.arrive = Math.max(R.plan.arrive, S.day + 1); out.push({ kind:'good', title:'Sethi\'yi geçtin!', text:'Rakibinden önce vardın: bu ziyarette satışların yüzde 8 fazla getirir. Sethi ' + gn + ' yükünü ancak yarın boşaltabilecek.' }); }
    else { R.plan.arrive = Math.min(R.plan.arrive, S.day); out.push({ kind:'bad', title:'Sethi senden önce vardı', text:'İskeleyi o kaptı ve pazara ' + gn + ' boşalttı.' }); }
  }
  sethiArrive(S, out);
  for (let i = 0; i < d; i++) rollPrices(S);
  if (res.prophecy && gById(res.prophecy)) { S.prices[S.city][res.prophecy] *= 1.45; out.push({ kind:'good', title:'Kehanet gerçek oldu', text:CITIES[S.city].loc + ' herkes ' + gById(res.prophecy).short.toLowerCase() + ' arıyor: fiyatı bugün yüzde 45 yukarıda.' }); }
  if ((res.maxCombo || 0) >= 5) { S.fame = { city:S.city, day:S.day, b:0.05 }; out.push({ kind:'good', title:'Tayfanın şöhreti', text:'x' + res.maxCombo + ' kıl payı kombosuyla geldiğin limanda konuşuluyor. Bugün satışların %5 fazla getirir.' }); }
  if (S.contra) {
    if (S.contra.to === S.city) { S.cash += S.contra.reward; S.stats.contra = (S.stats.contra || 0) + 1; out.push({ kind:'good', title:'Kaçak teslim edildi', text:'Hapu\'nun adamı sandıkları sessizce aldı ve ' + fmtN(S.contra.reward) + ' deben saydı.' }); S.log.push({ d:S.day, t:'Hapu\'nun kaçak işini bitirdin: ' + fmtN(S.contra.reward) + ' deben.' }); S.contra = null; }
    else if (S.contra.due < S.day) { S.debt += 1000; out.push({ kind:'bad', title:'Hapu\'nun işi gecikti', text:'Sandıklar zamanında gitmedi. Hapu sandıkları geri aldırdı ve borcuna 1.000 deben "gecikme" yazdı.' }); S.contra = null; }
  }
  S.orders = S.orders.filter(o => {
    if (o.due >= S.day) return true;
    S.trust[o.from] = Math.max(0, S.trust[o.from] - (o.advance ? 200 : 120));
    if (o.advance) { const back = Math.min(S.cash, o.advance); S.cash -= back; S.debt += o.advance - back; }
    out.push({ kind:'bad', title:'Sipariş gecikti', text:MERCHANT_NAMES[o.from] + '\'in ' + CITIES[o.to].dat + ' siparişini zamanında götüremedin. Güveni azaldı.' + (o.advance ? ' Aldığın ' + fmtN(o.advance) + ' deben avansı geri ödedin.' : '') });
    return false;
  });
  const off = S.offers[S.city];
  if ((!off || off.due < S.day + 1) && !S.orders.some(o => o.from === S.city)) S.offers[S.city] = rng() < (isRiver(S.city) ? 0.75 : 0.9) ? makeOrder(S, S.city) : null;
  if (seasonIdx(S) !== seasonBefore) { const se = SEASONS[seasonIdx(S)]; out.push({ kind:'info', title:se.name + ': ' + se.title, text:se.desc }); }
  S.seen[S.city] = snap(S, S.city);
  if (S.hero === 'sitre') RIVER.forEach(i => { if (Math.abs(i - S.city) === 1 && isRiver(S.city)) S.seen[i] = snap(S, i); });
  if (!S.visited.includes(S.city)) S.visited.push(S.city);
  if (S.debt > before) out.push({ kind:'info', title:'Faiz işledi', text:'Borcun ' + fmtN(before) + ' debenden ' + fmtN(S.debt) + ' debene çıktı.' });
  if (S.city === 1 && S.debt > 0 && S.cash > 0 && isRiver(plan.from)) out.push({ kind:'info', title:'Hapu\'nun evi burada', text:'Borç yalnızca Mennefer\'de ödenir. Üstteki Borç rakamına dokunup ' + fmtN(Math.min(S.cash, S.debt)) + ' debene kadar ödeyebilirsin.' });
  ledger.interest = S.debt - before;
  S.last = { from:plan.from, days:d, out, news:S.news.slice(), path:res.path || null, quick:!!res.quick, chaser:plan.chaser || null, rival:!!plan.rival, ledger };
  S.pendingEvent = (isRiver(S.city) || CITIES[S.city].id === 'saww') && rng() < 0.45 ? pickEvent(S) : null;
  const cr = has(S, 'crisis') && (S.crises || []).find(c => c.city === S.city && !c.done && S.day >= c.from && S.day <= c.until && CRISIS_EV[c.k]);
  if (cr) S.pendingEvent = 'crisis:' + cr.id;
  S.log.push({ d:S.day, t:CITIES[S.city].name + ' şehrine vardın. ' + out.map(o => o.title).join(' · ') + '.' });
  if (S.log.length > 80) S.log.splice(0, S.log.length - 80);
  return S.last;
}

function eventDef(S, id) {
  if (id && String(id).startsWith('crisis:')) { const c = (S.crises || []).find(x => 'crisis:' + x.id === id); if (!c || !CRISIS_EV[c.k]) return null; const e = CRISIS_EV[c.k](S, c); e.id = id; e.crisis = c; e.ok = () => true; return e; }
  return EVENTS.find(x => x.id === id) || null;
}
function pickEvent(S) { const l = EVENTS.filter(e => e.ok(S)); return l.length ? pick(l).id : null; }
function resolveEvent(S, idx, extra) {
  const e = eventDef(S, S.pendingEvent); S.pendingEvent = null;
  if (!e) return null;
  if (e.crisis) e.crisis.done = true;
  const ch = e.choices[idx], r = e.ok(S) && ch && (!ch.ok || ch.ok(S)) ? ch.run(S, extra) : { kind:'info', text:'Fırsat kaçtı: durum değişmişti.' };
  S.log.push({ d:S.day, t:e.title + ': ' + r.text });
  return r;
}
/* a day in port when no harbour can be reached: interest runs, prices move, Sethi moves */
function waitDay(S) {
  if (S.over || S.day >= S.days) return null;
  const out = [], before = S.debt;
  if (!/^(siwa|kharga)$/.test(CITIES[S.city].id)) caravanDays(S, 1, out);
  S.day += 1; S.debt = Math.round(S.debt * (1 + S.rate)); S.stats.interest = (S.stats.interest || 0) + (S.debt - before);
  sethiArrive(S, out); rollPrices(S); S.seen[S.city] = snap(S, S.city);
  /* v42: a day in town can bring a new order to the board (the oases too) */
  const off = S.offers[S.city];
  if ((!off || off.due < S.day + 1) && !S.orders.some(o => o.from === S.city) && rng() < 0.5) { S.offers[S.city] = makeOrder(S, S.city); if (S.offers[S.city]) out.push({ kind:'info', title:'Panoda yeni sipariş', text:MERCHANT_NAMES[S.city] + ' panoya bir sipariş astı: ' + S.offers[S.city].q + ' ' + gById(S.offers[S.city].g).short.toLowerCase() + ' → ' + CITIES[S.offers[S.city].to].name + '.' }); }
  if (S.debt > before) out.push({ kind:'info', title:'Faiz işledi', text:'Borcun ' + fmtN(before) + ' debenden ' + fmtN(S.debt) + ' debene çıktı.' });
  S.log.push({ d:S.day, t:CITIES[S.city].loc + ' bir gün bekledin.' });
  return out;
}
function finish(S) {
  if (S.caravan) {
    const hub = S.caravan.hub;
    if (S.stash) GOODS.forEach(g => { const b = S.stash[g.id]; if (!b || !b.n) return; S.cash += Math.round(b.n * (deals(hub, g.id) ? S.prices[hub][g.id] * 0.9 : g.base * 0.5)); });
    S.stash = null; S.caravan = null;
  }
  S.over = true;
  (S.work || []).forEach(w => { S.cash += Math.round(w.n * gById(w.g).base * 0.5); }); S.work = [];
  GOODS.forEach(g => { if (deals(S.city, g.id)) sell(S, g.id, Infinity, true); else if (S.inv[g.id].n) { S.cash += Math.round(S.inv[g.id].n * g.base * 0.5); S.inv[g.id] = { n:0, cost:0 }; } });

  const paid = Math.min(S.cash, S.debt);
  S.cash -= paid; S.debt -= paid;
  const hull = Math.max(0, boatWorth(S) - 600), score = S.cash - S.debt + hull;
  S.result = { score, hull, rank:rankOf(score), paid, debtLeft:S.debt, stats:S.stats, rival:S.rival.cash, beat:score > S.rival.cash, start:S.startWorth == null ? START_CASH - START_DEBT : S.startWorth };
  S.result.ach = ACH.filter(a => a.ok(S, S.result)).map(a => a.id);
  S.result.rep = S.mode === 'daily' ? 0 : Math.max(1, Math.floor(Math.max(0, score) / 1200)) + (S.result.beat ? 2 : 0) + (S.repBonus || 0);
  S.log.push({ d:S.day, t:'Sefer bitti. Net servet: ' + fmtN(score) + ' deben.' });
  return S;
}
