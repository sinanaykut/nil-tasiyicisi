  /* ================= sound =================
     Three pieces are composed and rendered once, offline, into looping buffers: the port theme (90 bpm),
     the sailing theme (120 bpm) and a chase layer that sits exactly on top of the sailing theme.
     Nothing is synthesised live while music plays, so busy frames cannot crackle. Effects stay short and live. */
  const Snd = (() => {
    let ac = null, out, mGain, sBus, mode = 'port', started = false;
    const bufs = {}, srcs = {}, gains = {};
    const pref = { music: lsGet(MUS) !== '0', sfx: lsGet(SFX) !== '0' };
    const SR = 32000, TAIL = 4.5, MUSIC_VER = 'm2';
    const midi = n => 440 * Math.pow(2, (n - 69) / 12);
    function impulse(c, sec, decay) {
      const len = Math.floor(c.sampleRate * sec), b = c.createBuffer(2, len, c.sampleRate);
      for (let ch = 0; ch < 2; ch++) { const d = b.getChannelData(ch); for (let i = 0; i < len; i++) d[i] = (Math.random() * 2 - 1) * Math.pow(1 - i / len, decay); }
      return b;
    }
    function panned(c, node, dest, p) {
      if (c.createStereoPanner) { const s = c.createStereoPanner(); s.pan.value = p; node.connect(s); s.connect(dest); } else node.connect(dest);
    }
    /* ---------- a shared instrument kit for one offline render ---------- */
    function kit(c, BEAT) {
      const master = c.createGain(); master.gain.value = 0.9; master.connect(c.destination);
      const verb = c.createConvolver(); verb.buffer = impulse(c, 3.2, 2.8);
      const vOut = c.createGain(); vOut.gain.value = 0.5; verb.connect(vOut); vOut.connect(master);
      const dIn = c.createGain(), dl = c.createDelay(2), dr = c.createDelay(2), fb = c.createGain(), lp = c.createBiquadFilter(), mer = c.createChannelMerger(2), dOut = c.createGain();
      dl.delayTime.value = BEAT * 0.75; dr.delayTime.value = BEAT * 0.75; fb.gain.value = 0.3; lp.type = 'lowpass'; lp.frequency.value = 2600; dOut.gain.value = 0.28;
      dIn.connect(dl); dl.connect(lp); lp.connect(dr); dr.connect(fb); fb.connect(dl); dl.connect(mer, 0, 0); dr.connect(mer, 0, 1); mer.connect(dOut); dOut.connect(master);
      const bus = (dry, rv, dly) => {
        const g = c.createGain(); g.gain.value = dry; g.connect(master);
        const r = c.createGain(); r.gain.value = rv; g.connect(r); r.connect(verb);
        if (dly) { const d = c.createGain(); d.gain.value = dly; g.connect(d); d.connect(dIn); }
        return g;
      };
      const B = { pad:bus(.5, .7, 0), bass:bus(.95, .04, 0), arp:bus(.5, .35, .4), lead:bus(.6, .5, .2), bell:bus(.4, .7, .35), drum:bus(.75, .14, 0), tex:bus(.5, .6, 0), stab:bus(.55, .25, .15) };
      const filt = (type, f, q, dest, pan) => { const n = c.createBiquadFilter(); n.type = type; n.frequency.value = f; n.Q.value = q; panned(c, n, dest, pan || 0); return n; };
      const F = {
        arpL: filt('lowpass', 2400, 1, B.arp, -0.35), arpR: filt('lowpass', 2400, 1, B.arp, 0.35),
        bass: filt('lowpass', 560, 0.7, B.bass), tek: filt('bandpass', 2600, 2, B.drum, 0.15), shk: filt('highpass', 6500, 0.8, B.drum, -0.2),
        clap: filt('bandpass', 1600, 0.8, B.drum), thud: filt('lowpass', 500, 0.7, B.drum), bellL: filt('lowpass', 9000, 0.5, B.bell, -0.3), bellR: filt('lowpass', 9000, 0.5, B.bell, 0.3),
        stab: filt('lowpass', 1100, 2.2, B.stab, 0)
      };
      const nb = c.createBuffer(1, c.sampleRate, c.sampleRate), nd = nb.getChannelData(0);
      for (let i = 0; i < nd.length; i++) nd[i] = Math.random() * 2 - 1;
      const vca = (t, v, a, dec, dest) => { const g = c.createGain(); g.gain.setValueAtTime(0.0001, t); g.gain.exponentialRampToValueAtTime(v, t + a); g.gain.exponentialRampToValueAtTime(0.0001, t + dec); g.connect(dest); return g; };
      function osc(type, f, t, end, dest, detune) { const o = c.createOscillator(); o.type = type; o.frequency.value = f; if (detune) o.detune.value = detune; o.connect(dest); o.start(t); o.stop(end); return o; }
      function pad(notes, t, dur, v) {
        const f = c.createBiquadFilter(); f.type = 'lowpass'; f.Q.value = 0.5;
        f.frequency.setValueAtTime(380, t); f.frequency.linearRampToValueAtTime(1300, t + dur * 0.55); f.frequency.linearRampToValueAtTime(520, t + dur + 2);
        const g = c.createGain(); g.gain.setValueAtTime(0, t); g.gain.linearRampToValueAtTime(v, t + Math.min(1.8, dur * 0.4)); g.gain.setValueAtTime(v, t + dur); g.gain.linearRampToValueAtTime(0, t + dur + 2.4);
        f.connect(g); g.connect(B.pad);
        notes.forEach(n => { osc('sawtooth', midi(n), t, t + dur + 2.5, f, -8); osc('sawtooth', midi(n), t, t + dur + 2.5, f, 8); });
      }
      function bass(n, t, len, v) {
        const g = c.createGain(); g.connect(F.bass);
        g.gain.setValueAtTime(0.0001, t); g.gain.exponentialRampToValueAtTime(v, t + 0.02); g.gain.setTargetAtTime(v * 0.6, t + 0.08, 0.25); g.gain.setTargetAtTime(0.0001, t + Math.max(0.05, len - 0.05), 0.05);
        osc('sine', midi(n), t, t + len + 0.35, g);
        const h = c.createGain(); h.gain.value = 0.24; h.connect(g); osc('triangle', midi(n) * 2, t, t + len + 0.35, h);
      }
      let side = 0;
      const pluck = (n, t, v, dec) => osc('triangle', midi(n), t, t + dec + 0.05, vca(t, v, 0.006, dec, (side++ % 2) ? F.arpL : F.arpR));
      function bell(n, t, v, dec) { const d = (side++ % 2) ? F.bellL : F.bellR; osc('sine', midi(n), t, t + dec + 0.05, vca(t, v, 0.005, dec, d)); osc('sine', midi(n) * 2.76, t, t + dec / 2.76 + 0.05, vca(t, v * 0.28, 0.005, dec / 2.76, d)); }
      function hit(t, len, v, dest, a) { const s = c.createBufferSource(); s.buffer = nb; s.connect(vca(t, v, a || 0.004, len, dest)); s.start(t, (t * 7.31) % 0.8, len + 0.05); }
      function kick(t, v) { const o = osc('sine', 130, t, t + 0.5, vca(t, v, 0.006, 0.45, B.drum)); o.frequency.setValueAtTime(140, t); o.frequency.exponentialRampToValueAtTime(46, t + 0.25); }
      function doum(t, v) { const o = osc('sine', 96, t, t + 0.6, vca(t, v, 0.01, 0.55, B.drum)); o.frequency.setValueAtTime(96, t); o.frequency.exponentialRampToValueAtTime(64, t + 0.3); hit(t, 0.12, v * 0.25, F.thud); }
      const tek = (t, v) => hit(t, 0.08, v, F.tek);
      const shaker = (t, v) => hit(t, 0.05, v, F.shk, 0.01);
      const clap = (t, v) => { hit(t, 0.16, v, F.clap); hit(t + 0.013, 0.12, v * 0.6, F.clap); };
      function stab(n, t, v, len) { const g = vca(t, v, 0.01, len, F.stab); osc('sawtooth', midi(n), t, t + len + 0.05, g, -7); osc('sawtooth', midi(n), t, t + len + 0.05, g, 7); osc('square', midi(n - 12), t, t + len + 0.05, g); }
      function phrase(notes, t, v, opt) {
        opt = opt || {};
        const o1 = c.createOscillator(), o2 = c.createOscillator(), m = c.createGain(), f = c.createBiquadFilter(), g = c.createGain(), lfo = c.createOscillator(), lg = c.createGain();
        const bf = c.createBiquadFilter(); bf.type = 'bandpass'; bf.frequency.value = opt.breath || 1400; bf.Q.value = 1.4; bf.connect(B.lead);
        o1.type = opt.ney ? 'sine' : 'sawtooth'; o2.type = 'triangle'; m.gain.value = opt.ney ? 0.6 : opt.bright ? 0.5 : 0.32;
        f.type = 'lowpass'; f.frequency.value = opt.cut || (opt.ney ? 2400 : 1750); f.Q.value = opt.ney ? 0.7 : 1.2;
        if (opt.ney) bf.frequency.value = 1900;
        lfo.frequency.value = opt.vib || (opt.ney ? 4.6 : 5.1); lg.gain.value = 0; lfo.connect(lg); lg.connect(o1.detune); lg.connect(o2.detune);
        o1.connect(m); m.connect(f); o2.connect(f); f.connect(g); panned(c, g, B.lead, 0.08);
        g.gain.setValueAtTime(0, t);
        let tt = t;
        notes.forEach(([n, beats], i) => {
          const d = beats * BEAT, fr = midi(n);
          if (i === 0) { o1.frequency.setValueAtTime(fr, tt); o2.frequency.setValueAtTime(fr, tt); }
          else { o1.frequency.setTargetAtTime(fr, tt, 0.025); o2.frequency.setTargetAtTime(fr, tt, 0.025); }
          g.gain.setTargetAtTime(v, tt, 0.04);
          g.gain.setTargetAtTime(v * 0.5, tt + d - Math.min(0.12, d * 0.3), 0.03);
          lg.gain.setValueAtTime(0, tt); lg.gain.linearRampToValueAtTime(11, tt + Math.min(0.5, d * 0.7));
          hit(tt, Math.min(opt.ney ? 0.5 : 0.3, d), v * (opt.ney ? 0.42 : 0.16), bf, opt.ney ? 0.06 : 0.04);
          tt += d;
        });
        g.gain.setTargetAtTime(0, tt - 0.1, 0.16);
        [o1, o2, lfo].forEach(o => { o.start(t); o.stop(tt + 1.5); });
      }
      /* oud: a plucked lute, bright attack that darkens quickly, with a tiny pitch fall */
      function oud(n, t, v, dec) {
        const fr = midi(n), f = c.createBiquadFilter(); f.type = 'lowpass'; f.Q.value = 2.5;
        f.frequency.setValueAtTime(3400, t); f.frequency.exponentialRampToValueAtTime(520, t + dec * 0.7);
        panned(c, f, B.arp, (side++ % 2) ? -0.25 : 0.25);
        const g = vca(t, v, 0.003, dec, f);
        const o = osc('sawtooth', fr, t, t + dec + 0.05, g); o.frequency.setValueAtTime(fr * 1.012, t); o.frequency.exponentialRampToValueAtTime(fr, t + 0.05);
        const g2 = c.createGain(); g2.gain.value = 0.5; g2.connect(g); osc('triangle', fr / 2, t, t + dec + 0.05, g2);
      }
      return { master, B, F, nb, osc, vca, pad, bass, pluck, bell, hit, kick, doum, tek, shaker, clap, stab, phrase, oud };
    }

    /* ---------- port theme: calm, D minor with a Hijaz colour ---------- */
    function composePort(c, BEAT, BAR, LOOP) {
      const K = kit(c, BEAT);
      const CH = [
        { b:38, pad:[50,57,64,65], arp:[62,69,64,69,74,69,65,69] },
        { b:34, pad:[46,53,57,62], arp:[58,65,62,65,70,65,69,65] },
        { b:31, pad:[43,50,57,58], arp:[55,62,57,62,67,62,58,62] },
        { b:33, pad:[45,52,55,61], arp:[57,64,61,64,69,64,67,64] }
      ];
      const MEL_A = [[69,2],[74,1],[76,1], [77,2],[76,1],[74,1], [72,2],[74,1],[70,1], [69,4], [67,2],[70,1],[74,1], [72,2],[70,1],[69,1], [73,2],[76,2], [76,3],[73,1]];
      const MEL_B = [[74,2],[77,1],[81,1], [79,2],[77,1],[76,1], [77,2],[74,2], [72,3],[69,1], [70,2],[74,1],[79,1], [77,2],[76,1],[74,1], [76,2],[73,1],[76,1], [74,4]];
      const ws = c.createBufferSource(); ws.buffer = K.nb; ws.loop = true;
      const wf = c.createBiquadFilter(); wf.type = 'bandpass'; wf.Q.value = 0.8;
      const wg = c.createGain(); wg.gain.value = 0.02;
      wf.frequency.setValueAtTime(500, 0);
      for (let k = 1; k <= 16; k++) wf.frequency.linearRampToValueAtTime(k % 2 ? 900 : 380, k * LOOP / 16);
      ws.connect(wf); wf.connect(wg); wg.connect(K.B.tex); ws.start(0); ws.stop(LOOP + TAIL);
      return function bar(bar) {
        const t0 = bar * BAR, ch = CH[Math.floor(bar / 2) % 4];
        const sec = bar < 4 ? 'I' : bar < 12 ? 'B' : bar < 28 ? 'T' : 'O';
        if (bar % 2 === 0) K.pad(ch.pad, t0, BAR * 2, sec === 'T' ? 0.05 : 0.06);
        if (sec === 'I' || sec === 'O') { if (bar % 2 === 0) K.bass(ch.b, t0, BAR * 2 - 0.15, 0.42); }
        else if (sec === 'B') K.bass(ch.b, t0, BAR - 0.1, 0.46);
        else { K.bass(ch.b, t0, BEAT * 1.5, 0.5); K.bass(ch.b, t0 + BEAT * 1.5, BEAT * 0.5, 0.32); K.bass(ch.b, t0 + BEAT * 2, BEAT * 1.5, 0.46); K.bass(ch.b + 12, t0 + BEAT * 3.5, BEAT * 0.5, 0.26); }
        if (sec === 'B' || sec === 'T' || (sec === 'O' && bar < 30)) {
          for (let k = 0; k < 8; k++) {
            if (sec === 'O' && k % 2) continue;
            const n = ch.arp[k] + (sec === 'T' && bar % 4 === 3 && k >= 4 ? 12 : 0);
            K.pluck(n, t0 + k * BEAT / 2 + (k % 2 ? 0.014 : 0), sec === 'B' ? 0.05 + (bar - 4) * 0.003 : 0.07, 0.45);
          }
        }
        if (sec === 'B') {
          K.doum(t0, 0.45); K.doum(t0 + BEAT * 1.5, 0.28);
          if (bar >= 8) { K.tek(t0 + BEAT * 3, 0.16); for (let k = 0; k < 8; k++) K.shaker(t0 + k * BEAT / 2 + (k % 2 ? 0.02 : 0), k % 2 ? 0.045 : 0.025); }
        }
        if (sec === 'T') {
          K.kick(t0, 0.75); K.kick(t0 + BEAT * 2, 0.62); K.doum(t0 + BEAT * 1.5, 0.3);
          K.tek(t0 + BEAT, 0.14); K.tek(t0 + BEAT * 3, 0.18); if (bar % 2) K.tek(t0 + BEAT * 3.5, 0.1);
          K.clap(t0 + BEAT, 0.09); K.clap(t0 + BEAT * 3, 0.09);
          for (let k = 0; k < 8; k++) K.shaker(t0 + k * BEAT / 2 + (k % 2 ? 0.02 : 0), k % 2 ? 0.05 : 0.028);
          if (bar === 27) for (let k = 0; k < 4; k++) K.doum(t0 + BEAT * 2 + k * BEAT / 2, 0.24 + k * 0.05);
        }
        if (sec === 'I' && bar % 2 === 1) K.bell(ch.arp[4] + 12, t0 + BEAT * 2, 0.11, 2.6);
        if (sec === 'T' && bar % 2 === 1) K.bell(ch.arp[2] + 24, t0 + BEAT * 3.5, 0.06, 2);
        if (bar === 12) K.phrase(MEL_A, t0, 0.1);
        if (bar === 20) K.phrase(MEL_B, t0, 0.11);
        if (bar >= 28) [[81, 28, 0], [86, 28, 2], [88, 28, 3], [89, 29, 0], [88, 30, 0], [86, 30, 2], [81, 31, 0]].forEach(([n, b, k]) => { if (b === bar) K.bell(n, b * BAR + k * BEAT, 0.09, 3.2); });
      };
    }

    /* ---------- sailing theme: driving, D Hijaz (D Eb F# G A Bb C) ---------- */
    const SAIL_CH = [
      { b:38, pad:[50,54,57,62], arp:[62,66,69,66,74,69,66,63] },
      { b:39, pad:[51,55,58,63], arp:[63,67,70,67,75,70,67,66] },
      { b:38, pad:[50,54,57,62], arp:[62,66,69,66,74,69,67,66] },
      { b:36, pad:[48,51,55,60], arp:[60,63,67,63,72,67,63,62] },
      { b:31, pad:[43,50,55,58], arp:[55,58,62,58,67,62,58,57] },
      { b:39, pad:[51,55,58,63], arp:[63,67,70,67,75,70,67,66] },
      { b:38, pad:[50,54,57,62], arp:[62,66,69,66,74,69,66,63] },
      { b:33, pad:[45,49,52,57], arp:[57,61,64,61,69,64,61,58] }
    ];
    function composeSail(c, BEAT, BAR) {
      const K = kit(c, BEAT);
      const MEL = [[74,1],[75,.5],[74,.5],[72,1],[70,1], [69,2],[70,1],[72,1], [74,1.5],[75,.5],[78,1],[75,1], [74,4], [79,1],[78,.5],[75,.5],[74,1],[72,1], [70,1.5],[72,.5],[74,2], [72,1],[70,1],[69,1],[67,1], [69,4]];
      return function bar(bar) {
        const t0 = bar * BAR, ch = SAIL_CH[Math.floor(bar / 2) % 8], E = BEAT / 2;
        if (bar % 2 === 0) K.pad(ch.pad, t0, BAR * 2, 0.042);
        const bp = [0, 0, 12, 0, 0, 0, 12, 10];
        for (let k = 0; k < 8; k++) K.bass(ch.b + bp[k], t0 + k * E, E * 0.85, k % 4 === 0 ? 0.5 : 0.34);
        if (bar >= 2) for (let k = 0; k < 8; k++) K.pluck(ch.arp[k] + (bar >= 8 && bar % 2 ? 12 : 0), t0 + k * E + (k % 2 ? 0.012 : 0), k % 2 ? 0.045 : 0.06, 0.32);
        K.doum(t0, 0.5); K.tek(t0 + E, 0.13); K.tek(t0 + E * 3, 0.15); K.doum(t0 + E * 4, 0.36); K.tek(t0 + E * 6, 0.16);
        if (bar % 2) K.tek(t0 + E * 7, 0.1);
        K.kick(t0, 0.5); K.kick(t0 + BEAT * 2, 0.42);
        for (let k = 0; k < 16; k++) K.shaker(t0 + k * BEAT / 4, k % 4 === 2 ? 0.05 : 0.022);
        if (bar >= 4) { K.clap(t0 + BEAT, 0.06); K.clap(t0 + BEAT * 3, 0.07); }
        if (bar === 8) K.phrase(MEL, t0, 0.1, { bright:true, cut:2200, vib:6 });
        if (bar === 7 || bar === 15) for (let k = 0; k < 4; k++) K.doum(t0 + BEAT * 2 + k * E, 0.22 + k * 0.06);
        if (bar % 4 === 0 && bar < 8) K.bell(ch.arp[4] + 12, t0 + BEAT * 3, 0.06, 1.8);
      };
    }
    /* ---------- chase layer: lies on top of the sailing theme, faded in when someone is behind you ---------- */
    function composeChase(c, BEAT, BAR) {
      const K = kit(c, BEAT);
      return function bar(bar) {
        const t0 = bar * BAR, ch = SAIL_CH[Math.floor(bar / 2) % 8], S16 = BEAT / 4;
        for (let k = 0; k < 4; k++) K.kick(t0 + k * BEAT, 0.6);
        for (let k = 0; k < 16; k++) if (k % 4) K.tek(t0 + k * S16, k % 2 ? 0.07 : 0.11);
        K.clap(t0 + BEAT, 0.1); K.clap(t0 + BEAT * 3, 0.11);
        [0.5, 1.5, 2.5, 3.5].forEach((b, i) => K.stab(ch.b + 24 + (i === 3 && bar % 2 ? 1 : 0), t0 + b * BEAT, 0.07, 0.16));
        if (bar % 2 === 1) for (let k = 0; k < 4; k++) K.stab(ch.b + 36 + [0, 1, 4, 7][k], t0 + BEAT * 2 + k * S16 * 2, 0.05, 0.12);
      };
    }
    /* ---------- main menu: "Nil'in Şafağı" — slow and wide, E Hijaz (E F G# A B C D), ney over harp and frame drum ---------- */
    function composeMenu(c, BEAT, BAR) {
      const K = kit(c, BEAT);
      const E = { b:40, pad:[52,56,59,64], arp:[64,68,71,76,71,68] }, Fc = { b:41, pad:[53,57,60,65], arp:[65,69,72,77,72,69] }, Dm = { b:38, pad:[50,53,57,62], arp:[62,65,69,74,69,65] };
      const CH = [E, Fc, E, Dm, E, Fc, Dm, E, Fc, E];
      const M1 = [[71,2],[72,1],[71,1],[69,2],[68,2], [69,1],[68,1],[65,1],[68,1],[64,4], [64,1],[65,1],[68,2],[69,2],[71,2], [72,1.5],[71,0.5],[69,1],[68,1],[69,4]];

      return function bar(k) {

        if (k % 2 === 0) K.pad(ch.pad, t0, BAR * 2, k < 4 || k >= 18 ? 0.045 : 0.055);
        if (k % 4 === 0) K.bass(40, t0, BAR * 4 - 0.1, 0.3);
        if (k % 2 === 0 && k >= 4) K.bass(ch.b, t0, BAR * 2 - 0.2, 0.22);
        if (k >= 2 && k < 18) for (let j = 0; j < 6; j++) K.pluck(ch.arp[j] + (full ? 12 : 0), t0 + j * BEAT * 2 / 3, j === 0 ? 0.06 : 0.04, 1.6);
        if (k >= 4 && k < 18) { K.doum(t0, 0.42); K.tek(t0 + BEAT * 1.5, 0.09); K.doum(t0 + BEAT * 2, 0.24); K.tek(t0 + BEAT * 3, 0.12); if (k % 2) K.tek(t0 + BEAT * 3.5, 0.07); }
        if (full) { for (let j = 0; j < 8; j++) { K.shaker(t0 + j * BEAT / 2, j % 2 ? 0.04 : 0.022); K.stab(ch.pad[j % 2 ? 2 : 0], t0 + j * BEAT / 2, 0.025, 0.22); } }
        if (k === 4) K.phrase(M1, t0, 0.1, { ney:true });
        if (k === 12) K.phrase(M2, t0, 0.11, { ney:true });
        if (k === 0) K.bell(76, t0 + BEAT * 2, 0.08, 3.4);
        if (k === 11 || k === 17) for (let j = 0; j < 4; j++) K.doum(t0 + BEAT * 2 + j * BEAT / 2, 0.2 + j * 0.05);
        if (k >= 18) for (let j = 0; j < 4; j++) K.pluck(ch.arp[5 - j] + 12, t0 + j * BEAT, 0.045, 2.2);
        if (k === 19) K.bell(64, t0 + BEAT * 2, 0.07, 3.5);
      };
    }
    /* ---------- port 2: "Çarşı Davulları" — a busy bazaar, G Hijaz (G Ab B C D Eb F), oud riff and darbuka ---------- */
