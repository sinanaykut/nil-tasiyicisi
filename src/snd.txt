  /* ================= sound =================
     Six pieces are composed in code and rendered once by NilSynth (snd-syn.txt) into seamless looping buffers:
     the title theme, three port tunes, the sailing theme and a chase layer that sits exactly on top of it.
     Rendering runs in a Web Worker (built from NilSynth's own source), or, where Workers are not allowed, in
     short slices on the main thread, so input never waits. The first tune is played while it is still being
     rendered (one-second segments, handed over to the finished loop at its end). Finished tunes are kept in
     IndexedDB (16-bit) so later visits start at once; at most three are held decoded in memory.
     Nothing is synthesised live while music plays, so busy frames cannot crackle. Effects stay short and live. */
  /* the render worker's message loop (its source text is sent to the worker; it must stay self-contained) */
  function SndWorkerMain(S) {
    const i16 = f => { const a = new Int16Array(f.length); for (let i = 0; i < f.length; i++) { const v = f[i]; a[i] = v > 1 ? 32767 : v < -1 ? -32767 : Math.round(v * 32767); } return a; };
    const bg = {};
    self.onmessage = e => {
      const m = e.data;
      if (m.op === 'hi') { self.postMessage({ op:'hi' }); return; }
      if (m.op === 'speed') { bg[m.id] = m.bg; return; }
      if (m.op !== 'render') return;
      /* background renders leave the CPU free part of the time (phones with few cores); urgent ones run flat out */
      bg[m.id] = !!m.bg;
      try {
        const t0 = Date.now(), J = S.job(m.name);
        if (m.stream) J.onSeg = (pos, L, R) => self.postMessage({ op:'seg', id:m.id, pos, L, R }, [L.buffer, R.buffer]);
        let last = 0;
        const run = () => {
          try {
            if (!J.step(bg[m.id] ? 40 : 150)) { if (J.prog - last >= 0.04) { last = J.prog; self.postMessage({ op:'prog', id:m.id, p:J.prog }); } setTimeout(run, bg[m.id] ? 32 : 0); return; }
            const L = J.out[0], R = J.out[1], a = i16(L), b = i16(R);
            self.postMessage({ op:'done', id:m.id, L, R, a, b, stats:Object.assign({ ms:Date.now() - t0 }, J.stats) }, [L.buffer, R.buffer, a.buffer, b.buffer]);
          } catch (err) { self.postMessage({ op:'err', id:m.id, msg:String(err && err.message || err) }); }
          delete bg[m.id];
        };
        run();
      } catch (err) { self.postMessage({ op:'err', id:m.id, msg:String(err && err.message || err) }); }
    };
  }
  const Snd = (() => {
    let ac = null, out, mGain, sBus, mode = 'port', started = false;
    const bufs = {}, srcs = {}, gains = {};
    const pref = { music: lsGet(MUS) !== '0', sfx: lsGet(SFX) !== '0' };
    const SR = 32000, MUSIC_VER = 'm45', MAXBUF = 3, SEGMIN = 2;
    const midi = n => 440 * Math.pow(2, (n - 69) / 12);
    /* a short warm room for the effects */
    function impulse(c, sec, decay) {
      const len = Math.floor(c.sampleRate * sec), b = c.createBuffer(2, len, c.sampleRate);
      for (let ch = 0; ch < 2; ch++) { const d = b.getChannelData(ch); let lp = 0; const a = 1 - Math.exp(-2 * Math.PI * 3200 / c.sampleRate); for (let i = 0; i < len; i++) { lp += a * ((Math.random() * 2 - 1) - lp); d[i] = lp * 1.8 * Math.pow(1 - i / len, decay) * Math.min(1, i / (0.01 * c.sampleRate)); } }
      return b;
    }
    const PORTS = ['port1', 'port2', 'port3', 'menu'];
    const TRACKS = {
      menu:  { sec:24 * 4 * 60 / 76,  name:'Nil\'in Şafağı',   desc:'Ana menü teması · ney, kanun ve arp' },
      port1: { sec:32 * 4 * 60 / 100, name:'Mennefer Limanı',  desc:'Sıcak liman çarşısı · ud, ney ve darbuka' },
      port2: { sec:32 * 4 * 60 / 116, name:'Çarşı Davulları',  desc:'Hareketli pazar · kamış, ud ve el çırpma' },
      port3: { sec:20 * 4 * 60 / 66,  name:'Tapınak Gölgesi',  desc:'Gizemli ve dingin · ney, arp, sistrum' },
      sail:  { sec:32 * 4 * 60 / 104, name:'Yelkenler Açık',   desc:'Nehir yolculuğu · arp dalgaları ve ney' },
      chase: { sec:32 * 4 * 60 / 104, name:'Takip',            desc:'' }
    };

    /* ---------- the renderer: a Worker, or slices on the main thread ---------- */
    let wk = null, wkOk = null, syn = null, jobId = 0;
    const jobs = {};
    function failWorker(why) {
      if (wkOk === false) return;
      wkOk = false; lastErr = lastErr || ''; try { if (wk) wk.terminate(); } catch (e) {} wk = null;
      /* whatever the worker was doing is redone here */
      Object.keys(jobs).forEach(id => { const j = jobs[id]; if (!j.local) { delete jobs[id]; j.stream = j.stream && !streams[j.name]; runLocal(j); } });
      void why;
    }
    function worker() {
      if (wkOk === false) return null;
      if (wk) return wk;
      try {
        const code = '"use strict";\n' + NilSynth.toString() + '\n' + SndWorkerMain.toString() + '\nSndWorkerMain(NilSynth());\n';
        const url = URL.createObjectURL(new Blob([code], { type:'text/javascript' }));
        wk = new Worker(url);
        setTimeout(() => { try { URL.revokeObjectURL(url); } catch (e) {} }, 10000);
        wk.onerror = e => { if (e && e.preventDefault) e.preventDefault(); failWorker('error'); };
        wk.onmessage = e => onMsg(e.data);
        wk.postMessage({ op:'hi' });
        setTimeout(() => { if (wkOk !== true) failWorker('silent'); }, 4000);
      } catch (e) { wkOk = false; wk = null; }
      return wk;
    }
    function onMsg(m) {
      if (m.op === 'hi') { wkOk = true; return; }
      const j = jobs[m.id]; if (!j) return;
      if (m.op === 'prog') { prog[j.name] = m.p; changed(); }
      else if (m.op === 'seg') addSeg(j.name, m.pos, m.L, m.R);
      else if (m.op === 'done') { delete jobs[m.id]; j.res({ L:m.L, R:m.R, a:m.a, b:m.b, stats:m.stats }); }
      else if (m.op === 'err') { delete jobs[m.id]; j.rej(new Error(m.msg)); }
    }
    /* main-thread fallback: urgent work in ~10 ms slices between frames, background work only in idle time */
    let mch = null;
    const soon = fn => { try { if (!mch) { mch = new MessageChannel(); mch.q = []; mch.port1.onmessage = () => { const f = mch.q.shift(); if (f) f(); }; } mch.q.push(fn); mch.port2.postMessage(0); } catch (e) { setTimeout(fn, 0); } };
    const idle = fn => window.requestIdleCallback ? requestIdleCallback(d => fn(d.didTimeout ? 8 : Math.max(6, Math.min(14, d.timeRemaining() - 2))), { timeout:50 }) : setTimeout(() => fn(8), 16);
    function runLocal(j) {
      j.local = true; jobs[j.id] = j;
      if (!syn) syn = NilSynth();
      const t0 = Date.now(), J = syn.job(j.name);
      if (j.stream) J.onSeg = (pos, L, R) => addSeg(j.name, pos, L, R);
      const tick = ms => {
        try {
          if (!J.step(ms || 10)) { prog[j.name] = J.prog; changed(); if (j.bg) idle(tick); else soon(tick); return; }
          delete jobs[j.id]; j.res({ L:J.out[0], R:J.out[1], stats:Object.assign({ ms:Date.now() - t0, local:true }, J.stats) });
        } catch (e) { delete jobs[j.id]; j.rej(e); }
      };
      setTimeout(tick, 0);
    }
    const fewCores = (navigator.hardwareConcurrency || 2) <= 4;
    function synth(name, stream, urgent) {
      return new Promise((res, rej) => {
        const j = { id:++jobId, name, stream, res, rej, bg:!stream && !urgent };
        const w = worker();
        if (w) { j.bg = j.bg && fewCores; jobs[j.id] = j; w.postMessage({ op:'render', id:j.id, name, stream, bg:j.bg }); }
        else runLocal(j);
      });
    }
    /* a tune the player now waits for is rendered at full speed */
    function hurry(name) { Object.keys(jobs).forEach(id => { const j = jobs[id]; if (j.name === name && j.bg) { j.bg = false; if (wk && !j.local) wk.postMessage({ op:'speed', id:j.id, bg:false }); } }); }
    /* typed arrays -> AudioBuffer (before the first tap there is no AudioContext yet) */
    function mkBuf(L) {
      try { return new AudioBuffer({ length:L, numberOfChannels:2, sampleRate:SR }); } catch (e) {}
      if (ac) return ac.createBuffer(2, L, SR);
      const OAC = window.OfflineAudioContext || window.webkitOfflineAudioContext;
      return new OAC(2, 1, SR).createBuffer(2, L, SR);
    }
    function toBuf(L, R) {
      const b = mkBuf(L.length);
      if (b.copyToChannel) { b.copyToChannel(L, 0); b.copyToChannel(R, 1); } else { b.getChannelData(0).set(L); b.getChannelData(1).set(R); }
      return b;
    }
    /* 16-bit <-> float in slices, so a long tune never blocks a frame */
    function sliced(n, fn) { return new Promise(res => { let i = 0; const go = () => { const e = Math.min(n, i + 196608); fn(i, e); i = e; if (i < n) setTimeout(go, 0); else res(); }; go(); }); }
    function f2i(f) { const a = new Int16Array(f.length); return sliced(f.length, (s, e) => { for (let i = s; i < e; i++) { const v = f[i]; a[i] = v > 1 ? 32767 : v < -1 ? -32767 : Math.round(v * 32767); } }).then(() => a); }
    function i2f(a) { const f = new Float32Array(a.length); return sliced(a.length, (s, e) => { for (let i = s; i < e; i++) f[i] = a[i] / 32767; }).then(() => f); }

    /* ---------- IndexedDB cache ---------- */
    let dbp = null;
    function db() {
      if (dbp) return dbp;
      dbp = new Promise(res => {
        setTimeout(() => res(null), 2500);
        try {
          const r = indexedDB.open('nil-kervanci-audio', 1);
          r.onupgradeneeded = () => { try { r.result.createObjectStore('tracks'); } catch (e) {} };
          /* note which tunes are already stored, so they are only decoded when needed */
          r.onsuccess = () => { const d = r.result; try { const q = d.transaction('tracks', 'readonly').objectStore('tracks').getAllKeys(); q.onsuccess = () => { (q.result || []).forEach(k => { const p = String(k).split('@'); if (p[1] === MUSIC_VER && +p[2] === SR) stored[p[0]] = true; }); res(d); }; q.onerror = () => res(d); } catch (e) { res(d); } };
          r.onerror = () => res(null); r.onblocked = () => res(null);
        } catch (e) { res(null); }
      });
      return dbp;
    }
    const ckey = name => name + '@' + MUSIC_VER + '@' + SR;
    function cacheGet(name) {
      return db().then(d => new Promise(res => {
        if (!d) return res(null);
        try { const q = d.transaction('tracks', 'readonly').objectStore('tracks').get(ckey(name)); q.onsuccess = () => res(q.result || null); q.onerror = () => res(null); } catch (e) { res(null); }
      }));
    }
    /* older renders (other MUSIC_VER) are removed the first time the new one is stored */
    function cachePut(name, a, b) {
      db().then(d => {
        /* no IndexedDB (private mode, sandboxed frame): keep a 16-bit copy in memory instead (half the size of a decoded tune) */
        if (!d) { mem16[name] = [a, b]; stored[name] = true; changed(); return; }
        try {
          const tx = d.transaction('tracks', 'readwrite'), st = tx.objectStore('tracks');
          const del = st.openCursor();
          del.onsuccess = () => { const c = del.result; if (c) { if (String(c.key).split('@')[0] === name && c.key !== ckey(name)) c.delete(); c.continue(); } };
          st.put({ L:a.length, ch:[a.buffer, b.buffer] }, ckey(name));
          tx.oncomplete = () => { stored[name] = true; changed(); };
        } catch (e) {}
      });
    }
    function load(name) {
      if (mem16[name]) return Promise.all(mem16[name].map(i2f)).then(([L, R]) => toBuf(L, R));
      return cacheGet(name).then(rec => {
        if (rec && rec.L && rec.ch) return Promise.all(rec.ch.map(x => i2f(new Int16Array(x)))).then(([L, R]) => { stored[name] = true; fromDisk[name] = true; return toBuf(L, R); });
        /* stream only when nothing is sounding yet (the player is waiting for this one); sail and chase must stay in lockstep */
        const stream = name !== 'sail' && name !== 'chase' && !Object.keys(srcs).length;
        return synth(name, stream, urgentNow[name]).then(r => {
          stats[name] = r.stats;
          const b = toBuf(r.L, r.R);
          (r.a ? Promise.resolve([r.a, r.b]) : Promise.all([f2i(r.L), f2i(r.R)])).then(([a, c]) => cachePut(name, a, c));
          return b;
        });
      });
    }
    const prog = {}, fromDisk = {}, mem16 = {}, stored = {}, stats = {}, streams = {}, lru = [];
    let chT = 0;
    function changed() { if (chT) return; chT = setTimeout(() => { chT = 0; if (onChange) onChange(); }, 120); }

    /* ---------- streaming: play the first tune while its later segments are still being made ---------- */
    function addSeg(name, pos, L, R) {
      const st = streams[name] || (streams[name] = { segs:[], players:[] });
      const sg = { pos, L, R, buf:null }; st.segs.push(sg);
      st.players.forEach(P => schedSeg(P, sg));
      if (st.segs.length === SEGMIN && started) apply();
    }
    function segBuf(sg) { if (!sg.buf) { sg.buf = toBuf(sg.L, sg.R); sg.L = sg.R = null; } return sg.buf; }
    function schedSeg(P, sg) {
      if (P.dead) return;
      const at = P.t0 + sg.pos / SR, b = segBuf(sg), nw = ac.currentTime + 0.02;
      if (at + b.duration <= nw) return;
      const s = ac.createBufferSource(); s.buffer = b; s.connect(gainOf(P.k));
      if (at >= nw) s.start(at); else s.start(nw, nw - at);
      P.nodes.push(s); s.onended = () => { const i = P.nodes.indexOf(s); if (i >= 0) P.nodes.splice(i, 1); try { s.disconnect(); } catch (e) {} };
    }
    function loopSrc(k, when, offset) {
      const s = ac.createBufferSource(); s.buffer = bufs[k]; s.loop = true; s.connect(gainOf(k)); s.start(when, offset || 0); return s;
    }
    /* a player is what srcs[k] holds: a looping buffer, or a chain of streamed segments handing over to the loop */
    function player(k, when) {
      const P = { k, t0:when, nodes:[], dead:false, stop() { P.dead = true; P.nodes.forEach(n => { try { n.stop(); } catch (e) {} try { n.disconnect(); } catch (e) {} }); P.nodes = []; } };
      if (bufs[k]) { P.nodes.push(loopSrc(k, when)); return P; }
      const st = streams[k]; if (!st || st.segs.length < SEGMIN) return null;
      st.players.push(P); st.segs.forEach(sg => schedSeg(P, sg));
      return P;
    }
    function streamDone(k) {
      const st = streams[k]; if (!st) return;
      delete streams[k];
      st.players.forEach(P => {
        if (P.dead) return;
        const end = P.t0 + bufs[k].duration, nw = ac.currentTime + 0.03;
        P.nodes.push(end >= nw ? loopSrc(k, end) : loopSrc(k, nw, (nw - P.t0) % bufs[k].duration));
      });
    }

    /* ---------- playback: tracks are prepared one at a time in the background and crossfade by mode ---------- */
    let menuTrack = 'menu', curPort = PORTS.includes(lsGet(MUS + '-t')) ? lsGet(MUS + '-t') : 'port1', auto = lsGet(MUS + '-a') !== '0', rotT = null, onChange = null, portVisits = 0;
    const pending = {}, queue = [], stopT = {};
    let lastErr = '', busy = 0;
    const urgentNow = {};
    function ensure(name, urgent) {
      if (bufs[name]) { touch(name); return Promise.resolve(bufs[name]); }
      if (urgent) { urgentNow[name] = true; hurry(name); }
      if (!pending[name]) pending[name] = new Promise(res => { queue[urgent ? 'unshift' : 'push']({ name, res }); pump(); });
      else if (urgent) { const i = queue.findIndex(q => q.name === name); if (i > 0) queue.unshift(queue.splice(i, 1)[0]); }
      return pending[name];
    }
    function pump() {
      while (busy < 1 && queue.length) {
        busy++; const job = queue.shift();
        load(job.name).then(b => {
          bufs[job.name] = b; prog[job.name] = 1; busy--; touch(job.name);
          if (streams[job.name] && ac) streamDone(job.name); else delete streams[job.name];
          job.res(b); evict(); changed(); if (started) apply(); pump();
        }).catch(e => { busy--; delete pending[job.name]; lastErr = (e && e.message) || 'render'; changed(); pump(); setTimeout(() => { if (!bufs[job.name]) ensure(job.name); }, 3000); });
      }
    }
    function touch(k) { const i = lru.indexOf(k); if (i >= 0) lru.splice(i, 1); lru.push(k); }
    /* keep at most MAXBUF tunes decoded; a dropped one comes back from IndexedDB in a moment */
    function evict() {
      const d = desired(), keep = new Set(Object.keys(d).concat(Object.keys(srcs)));
      if (mode === 'chase' || mode === 'sea') { keep.add('sail'); keep.add('chase'); }
      let held = Object.keys(bufs).length;
      for (let i = 0; i < lru.length && held > MAXBUF; i++) {
        const k = lru[i]; if (keep.has(k) || !stored[k] || !bufs[k]) continue;
        delete bufs[k]; delete pending[k]; prog[k] = 0; held--; lru.splice(i, 1); i--;
      }
    }
    /* the tunes needed soon are decoded; the rest are only rendered once (if not stored yet) */
    function prefetch() {
      const soon = [menuTrack, curPort, 'sail', 'chase'];
      soon.forEach(k => ensure(k));
      db().then(() => ['port1', 'port2', 'port3', 'menu'].forEach(k => { if (!stored[k]) ensure(k); }));
    }
    function gainOf(k) { if (!gains[k]) { gains[k] = ac.createGain(); gains[k].gain.value = 0; gains[k].connect(mGain); } return gains[k]; }
    function desired() {
      if (mode === 'menu') return { [menuTrack]:0.6 };
      if (mode === 'port') return { [curPort]:0.62 };
      return { sail: mode === 'chase' ? 0.5 : 0.6, chase: mode === 'chase' ? 0.75 : 0 };
    }
    function stopSrc(k) { if (srcs[k]) { srcs[k].stop(); delete srcs[k]; } }
    function apply(tc) {
      if (!ac || !started) return;
      const d = desired(), names = Object.keys(d);
      const live = k => bufs[k] || (names.length === 1 && streams[k] && streams[k].segs.length >= SEGMIN);
      const missing = names.filter(k => !live(k) && !srcs[k]);
      if (missing.length) { missing.forEach(k => ensure(k, true).then(() => apply())); return; }
      const t = ac.currentTime, ts = t + 0.06;
      const fresh = names.filter(k => !srcs[k] && live(k));
      /* a new source must start silent, but the "start at 0" event has to sit BEFORE the fade-in on the timeline, or it cancels the fade */
      fresh.forEach(k => { const g = gainOf(k).gain; g.cancelScheduledValues(t); g.setValueAtTime(0, t); const P = player(k, ts); if (P) srcs[k] = P; clearTimeout(stopT[k]); stopT[k] = 0; });
      Object.keys(TRACKS).forEach(k => {
        if (!gains[k]) return;
        const g = gains[k].gain, inD = k in d;
        if (!fresh.includes(k)) { g.cancelScheduledValues(t); g.setValueAtTime(g.value, t); }
        /* fading out starts a little later than fading in, so the change of tune has no dip */
        g.setTargetAtTime(d[k] || 0, inD ? t : t + Math.min(0.5, (tc || 0.7) * 0.4), tc || 0.7);
        /* a faded-out tune is stopped once its fade is over (the timer is not re-armed by later changes) */
        if (!inD && srcs[k]) { if (!stopT[k]) stopT[k] = setTimeout(() => { stopT[k] = 0; if (!(k in desired()) && srcs[k]) { stopSrc(k); evict(); } }, 4000 + (tc || 0.7) * 1500); }
        else if (inD) { clearTimeout(stopT[k]); stopT[k] = 0; }
      });
      if (mode === 'port') armRotation(); else clearTimeout(rotT);
      if (onChange) onChange();
    }
    /* in automatic mode the port tune changes after it has played through twice */
    function armRotation() {
      clearTimeout(rotT);
      if (!auto) return;
      rotT = setTimeout(() => { if (mode === 'port' && auto) next(true); }, TRACKS[curPort].sec * 2 * 1000);
    }
    function next(quiet) {
      const i = PORTS.indexOf(curPort); curPort = PORTS[(i + 1) % PORTS.length]; if (mode === 'menu') menuTrack = curPort;
      if (!quiet) lsSet(MUS + '-t', curPort);
      if (mode === 'port' || mode === 'menu') apply(1.4); else if (onChange) onChange();
      return curPort;
    }
    function pick(name) { if (!PORTS.includes(name)) return; curPort = name; menuTrack = name; lsSet(MUS + '-t', name); if (mode === 'port' || mode === 'menu') apply(1.2); else if (onChange) onChange(); }
    function setAuto(on) { auto = !!on; lsSet(MUS + '-a', on ? '1' : '0'); if (mode === 'port') armRotation(); if (onChange) onChange(); }
    function init() {
      if (ac) { wake(); return ac; }
      try { ac = new (window.AudioContext || window.webkitAudioContext)(); } catch (e) { return null; }
      ac.onstatechange = () => { if (ac.state === 'running' && started) apply(0.4); changed(); };
      const lim = ac.createDynamicsCompressor();
      lim.threshold.value = -3; lim.knee.value = 0; lim.ratio.value = 20; lim.attack.value = 0.002; lim.release.value = 0.15;
      out = ac.createGain(); out.gain.value = 0.85; out.connect(lim); lim.connect(ac.destination);
      mGain = ac.createGain(); mGain.gain.value = 0; mGain.connect(out);
      /* effects: soften the top end (phone speakers), then a short warm room */
      sBus = ac.createGain(); sBus.gain.value = pref.sfx ? 0.7 : 0;
      const sHs = ac.createBiquadFilter(); sHs.type = 'highshelf'; sHs.frequency.value = 4500; sHs.gain.value = -5;
      const sLp = ac.createBiquadFilter(); sLp.type = 'lowpass'; sLp.frequency.value = 9000; sLp.Q.value = 0.5;
      sBus.connect(sHs); sHs.connect(sLp); sLp.connect(out);
      const sv = ac.createConvolver(); sv.buffer = impulse(ac, 1.1, 3);
      const svg = ac.createGain(); svg.gain.value = 0.2; sLp.connect(sv); sv.connect(svg); svg.connect(out);
      document.addEventListener('visibilitychange', () => { if (!ac) return; if (document.hidden) ac.suspend(); else ac.resume(); });
      wake();
      if (pref.music) startMusic();
      setInterval(watchdog, 2000);
      return ac;
    }
    /* every user gesture tries to wake the audio engine; WebKit also needs a sound started inside the gesture */
    function wake() {
      if (!ac || ac.state === 'running' || document.hidden) return;
      try { const b = ac.createBuffer(1, 1, ac.sampleRate), s = ac.createBufferSource(); s.buffer = b; s.connect(ac.destination); s.start(0); } catch (e) {}
      try { const pr = ac.resume(); if (pr && pr.then) pr.then(() => { if (started) apply(0.4); changed(); }).catch(() => {}); } catch (e) {}
    }
    /* if a wanted track is ready but nothing is sounding, start it */
    function watchdog() {
      if (!ac || !started || !pref.music || ac.state !== 'running') return;
      const d = desired();
      if (Object.keys(d).some(k => d[k] > 0 && (bufs[k] || streams[k]) && (!srcs[k] || gains[k].gain.value < 0.02))) apply(0.5);
    }
    function startMusic() {
      if (!ac || !pref.music) return;
