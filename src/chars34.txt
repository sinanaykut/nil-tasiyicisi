      if (!h.F || !h.acts) return;
      const k = kind === 'tap' ? ['laugh', 'wave', 'cheer', 'shrug'][(h.taps = (h.taps || 0) + 1) % 4] : kind;
      if (k === 'laugh') { h.F.laugh = 1.6; playOnce(h, 'cheer'); return; }
      if (k === 'angry') { h.F.flash = 'angry'; h.F.flashT = 1.6; fade(h, 'angry'); clearTimeout(h.rt); h.rt = setTimeout(() => { if (!h.dead) { h.F.flash = null; apply(h, true); } }, 1600); return; }
      if (h.acts[k]) { if (ONCE[k]) playOnce(h, k); else fade(h, k); }
    }
    function face(h, dt) {
      const F = h.F, U = h.M.U;
      let md = MOODS[F.flash || (F.laugh > 0 ? 'happy' : F.mood)] || MOODS.neutral;
      if (F.laugh > 0) F.laugh -= dt;
      /* blink */
      F.blinkT -= dt;
      if (F.blinkT <= 0) { F.blink = 0.16; F.blinkT = 2.2 + Math.random() * 3.2; if (Math.random() < 0.18) F.blinkT = 0.25; }
      let eye = md.eye;
      if (F.laugh > 0) eye = 'happy';
      if (F.blink > 0) { F.blink -= dt; const b = F.blink; eye = b > 0.11 || b < 0.04 ? (eye === 'happy' ? 'happy' : 'half') : (eye === 'happy' ? 'happy' : 'closed'); }
      slot(U.uEyeL.value, EYE[eye]); slot(U.uEyeRt.value, EYE[eye]);
      slot(U.uBrL.value, BROW[F.brow1 ? 'raised' : md.bl]); slot(U.uBrRt.value, BROW[md.br]);
      /* mouth: visemes while the text types */
      let mo = F.laugh > 0 ? 'grin' : md.mouth;
      if (F.talk) {
        F.visT -= dt;
        if (F.visT <= 0 || !F.vis) { const L = md.talk; let v = L[(Math.random() * L.length) | 0]; if (v === F.vis) v = L[(L.indexOf(v) + 1) % L.length]; F.vis = v; F.visT = 0.07 + Math.random() * 0.08; }
        mo = F.vis;
      } else F.vis = null;
      slot(U.uMo.value, MOUTH[mo] != null ? MOUTH[mo] : MOUTH.smile);
      U.uBlush.value += ((F.laugh > 0 ? 0.8 : md.blush) - U.uBlush.value) * Math.min(1, dt * 4);

      const tx = lk ? lk[0] : F.lookT.x, ty = lk ? lk[1] : F.lookT.y;
      F.look.x += (tx - F.look.x) * Math.min(1, dt * 9); F.look.y += (ty - F.look.y) * Math.min(1, dt * 9);
      U.uLook.value.set(Math.max(-1, Math.min(1, F.look.x)), Math.max(-1, Math.min(1, F.look.y)));
    }

    function headTurn(h, dt) {
      const F = h.F, B = h.I.bones, T0 = h.T0;
      F.hy += ((F.look.x * 0.38) - F.hy) * Math.min(1, dt * 5); F.hp += ((-F.look.y * 0.22 + 0.04) - F.hp) * Math.min(1, dt * 5);
      if (h.base === 'think' || h.once === 'wave') { F.hy *= 0.9; F.hp *= 0.9; }
      h.I.root.updateMatrixWorld(true);
      [['neck_01', 0.35], ['Head', 0.65]].forEach(([n, k]) => {
        const b = B[n]; if (!b || !b.parent) return;
        b.parent.getWorldQuaternion(h.q0);
        b.getWorldQuaternion(h.q1);
        h.e0.set(F.hp * k, F.hy * k, 0, 'YXZ'); h.qa.setFromEuler(h.e0);
        h.q1.premultiply(h.qa);
        b.quaternion.copy(h.q0.invert().multiply(h.q1));
        b.updateMatrixWorld(true);
      });
    }
    function frame(h, dt) {
      const el = h.el, r = h.r, cam = h.cam, T0 = h.T0;

    function fit(h) {
      const fr = h.fr, cam = h.cam;
      const top = fr.top, bot = fr.bot, mid = (top + bot) / 2, half = (top - bot) / 2;
      const fov = fr.fov, need = half / Math.tan(fov * Math.PI / 360);
      const needW = (fr.w || 0.3) / Math.tan(fov * Math.PI / 360) / Math.max(0.3, cam.aspect);
      const d = Math.max(need, needW);
      cam.fov = fov; cam.position.set(Math.sin(fr.yaw) * d, mid + 0.02, Math.cos(fr.yaw) * d); cam.lookAt(0, mid, 0); cam.near = d * 0.3; cam.far = d * 3; cam.updateProjectionMatrix();
    }
