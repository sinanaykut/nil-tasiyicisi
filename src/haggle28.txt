  const HG_OFF = [0.05, 0.12, 0.20], HG_MAX = 4;
  function hgRng(seed) {
    let a = typeof seed === 'number' ? seed >>> 0 : String(seed == null ? Math.random() : seed).split('').reduce((h, c) => Math.imul(h ^ c.charCodeAt(0), 16777619) >>> 0, 2166136261);
    if (!a) a = 0x9E3779B9;
    return () => { a = (a + 0x6D2B79F5) >>> 0; let t = a; t = Math.imul(t ^ (t >>> 15), t | 1); t ^= t + Math.imul(t ^ (t >>> 7), t | 61); return ((t ^ (t >>> 14)) >>> 0) / 4294967296; };
  }
  /* what the player can see: shelf 0 almost empty, 1 half, 2 full, 3 overflowing and dusty; crowd 0 nobody … 3 everyone wants it */
  const hgShelf = f => f > 1.15 ? 3 : f > 0.62 ? 2 : f > 0.28 ? 1 : 0;
  const hgCrowd = w => w > 1.25 ? 3 : w > 1.05 ? 2 : w > 0.85 ? 1 : 0;
  function hgRoom(o, rnd) {
    const f = Math.max(0, Math.min(2, (o.stock || 0) / Math.max(1, o.stockFull || 1))), w = o.want || 1, buy = o.side !== 'sell';
    let r = buy ? 0.02 + 0.11 * Math.min(f, 1.6) + (1 - w) * 0.1 : 0.03 + (w - 1) * 0.3 + (0.6 - Math.min(f, 1.2)) * 0.06;
    r += (o.trust || 0) * 0.012 + (o.mood || 0) * 0.03 + Math.min(0.025, Math.log2(1 + (o.qty || 1)) / 200) + (rnd() - 0.5) * 0.05;
    return Math.max(0.02, Math.min(0.24, r));
  }
  function hgNew(o) {
    o = Object.assign({ ci:0, g:'bugday', side:'buy', qty:1, unit:10, stock:10, stockFull:10, want:1, trust:0, mood:0 }, o);
    const ci = Math.max(0, Math.min(HG_M.length - 1, o.ci | 0)), M = HG_M[ci], rnd = o.rng || hgRng(o.seed);
    const f = Math.max(0, (o.stock || 0) / Math.max(1, o.stockFull || 1));
    const st = { o, ci, M, rnd, buy:o.side !== 'sell', R:hgRoom(o, rnd), cur:0, round:0, praised:false, walked:false, fin:false, res:null,
      pat:Math.max(2, 3 + M.pat + ((o.trust || 0) >= 2 ? 1 : 0) + ((o.mood || 0) > 0.4 ? 1 : 0) - ((o.mood || 0) < -0.4 ? 1 : 0)),
      shelf:hgShelf(f), f, crowd:hgCrowd(o.want || 1), tell:null, bluff:false, mood:(o.mood || 0) > 0.3 ? 'happy' : (o.mood || 0) < -0.3 ? 'sad' : 'neutral', log:[] };
    hgTell(st);
    return st;
  }
  /* body language shows how much room is left; ~15 % of the time he bluffs */
  function hgTell(st) {
    const has = st.R - st.cur >= 0.03; st.bluff = st.rnd() < 0.15;
    st.tell = (has !== st.bluff) ? st.M.room : st.M.lim;
  }
  const hgR = x => Math.round(x * 1000) / 1000;
  /* one player action → { ev, tell, bluff, learn } ; acts: offer (k = 0..2), praise, walk, accept, leave.
     A tell is learned only when the answer showed its meaning: he gave ground after a "room" tell, or held firm after a "limit" tell. */
  function hgStep(st, act, k) {
    const before = st.cur, out = hgStep0(st, act, k);
    if (out && (act === 'offer' || act === 'walk')) {
      let l = out.ev === 'back' || out.ev === 'yes' || (out.ev === 'counter' && st.cur - before >= 0.015) ? 'room' : out.ev === 'firm' || out.ev === 'bye' || out.ev === 'angry' || out.ev === 'counter' ? 'limit' : null;
      if (l && (l === 'room') !== (out.tell === st.M.room)) l = null;
      out.learn = l;
    }
    return out;
  }
  function hgStep0(st, act, k) {
    if (st.fin) return null;
    const M = st.M, prevTell = st.tell, prevBluff = st.bluff, out = { act, ev:'', tell:prevTell, bluff:prevBluff, learn:null };
    const end = (done, angry, ev) => { st.fin = true; out.ev = ev; st.res = { done, mult:done ? hgR(st.buy ? 1 - st.cur : 1 + st.cur) : 1, angry:!!angry }; if (!done) delete st.res.mult; };
    if (act === 'praise') {
      if (st.praised) return null;
      st.praised = true; out.ev = 'praise'; out.praise = M.praise;
      if (M.praise > 0) { st.R = Math.min(0.26, st.R + 0.025); st.pat++; st.mood = 'happy'; }
      else if (M.praise < 0) { st.R = Math.max(0.02, st.R - 0.01); st.pat--; st.mood = 'angry'; if (st.pat <= 0) { end(false, true, 'angry'); return out; } }
      else { st.R += 0.01; st.mood = 'thinking'; }
      hgTell(st); return out;
    }
    if (act === 'accept') { end(true, false, 'deal'); return out; }
    if (act === 'leave') { end(false, false, 'leave'); return out; }
    if (st.round >= HG_MAX) return null;
    if (act === 'walk') {
      if (st.walked) return null;
      st.walked = true; st.round++;
      if (st.R - st.cur >= 0.03) { st.cur = hgR(st.cur + (st.R - st.cur) * 0.7); out.ev = 'back'; st.mood = 'surprised'; hgTell(st); }
      else end(false, false, 'bye');
      return out;
    }
    if (act === 'offer') {
      const a = HG_OFF[k];
      if (a == null || a <= st.cur + 1e-9) return null;
      st.round++;
      if (a <= st.R) {
        /* easy for him → yes; close to his limit → he haggles back a little first, then gives in if you insist */
        if (a <= st.R - 0.035 || st.asked === a || st.round >= HG_MAX) { st.cur = a; out.ev = 'yes'; st.mood = 'happy'; end(true, false, 'yes'); return out; }
        st.asked = a; st.cur = hgR(Math.max(st.cur, a - 0.02 - st.rnd() * 0.015)); out.ev = 'counter'; st.mood = 'thinking'; hgTell(st);
        if (st.round >= HG_MAX) out.last = true;
        return out;
      }
      st.asked = a;
      const gap = a - st.R;
      st.pat -= gap >= 0.1 ? 2 : gap >= 0.03 ? 1 : 0;
      if (st.pat <= 0) { end(false, true, 'angry'); st.mood = 'angry'; return out; }
      const before = st.cur;
      st.cur = hgR(Math.min(st.R, st.cur + (st.R - st.cur) * (gap < 0.03 ? 0.85 : 0.5)));
      out.ev = st.cur - before < 0.005 ? 'firm' : 'counter';
      st.mood = st.pat <= 1 ? 'angry' : out.ev === 'firm' ? 'sly' : 'thinking';
      hgTell(st);
      if (st.round >= HG_MAX) out.last = true;
      return out;
    }
    return null;
  }
  /* what a player would do: 'read' = reads shelf, crowd and every tell; 'cues' = shelf and crowd only (no notebook yet); 'random'; 'greedy'; 'timid' */
  function hgGuess(st) {
    const s = st.shelf, c = st.crowd;
    return st.buy ? [0.04, 0.08, 0.13, 0.18][s] + [0.02, 0, -0.015, -0.03][c] : [0.03, 0.05, 0.1, 0.17][c] + [0.02, 0.01, 0, -0.01][s];
  }
  function hgPick(st, strat, known) {
    const offs = HG_OFF.map((a, k) => a > st.cur + 1e-9 ? k : -1).filter(k => k >= 0), canOff = st.round < HG_MAX;
    if (strat === 'random') {
      const A = []; if (canOff) offs.forEach(k => A.push(['offer', k])); if (!st.praised) A.push(['praise']); if (canOff && !st.walked) A.push(['walk']); A.push(['accept']);
      return A[Math.floor(st.rnd() * A.length)];
    }
    if (strat === 'greedy') return canOff && offs.length ? ['offer', offs[offs.length - 1]] : ['accept'];
    if (strat === 'timid') return canOff && st.cur === 0 ? ['offer', 0] : ['accept'];
    /* the reader: knows each merchant's tells (from the notebook) and what the shelf and the crowd mean */
    known = known || {};
    const M = st.M, cues = strat === 'cues', mean = known[st.tell] || (cues ? null : st.tell === M.room ? 'room' : st.tell === M.lim ? 'limit' : null);
    if (!st.praised && (known.praise != null ? known.praise : cues ? 0 : M.praise) > 0) return ['praise'];
    if (st.est == null) st.est = hgGuess(st);
    if (st.lastCur != null && st.cur > st.lastCur) st.est = mean === 'room' ? Math.max(st.est, st.cur + 0.05) : Math.min(st.est, st.cur + 0.01);
    st.lastCur = st.cur;
    if (!canOff || st.mood === 'angry' || (mean === 'limit' && st.cur > 0)) return ['accept'];
    const want = st.est + (mean === 'room' ? 0.035 : -0.01);
    let pickK = -1; offs.forEach(k => { if (HG_OFF[k] <= want) pickK = k; });
    if (mean === 'room' && !st.walked && st.round < HG_MAX - 1 && (pickK < 0 || st.est - HG_OFF[pickK] > 0.035)) return ['walk'];
    if (pickK >= 0) return ['offer', pickK];
    return st.cur > 0 || !offs.length ? ['accept'] : ['offer', offs[0]];
  }
