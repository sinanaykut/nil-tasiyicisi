      const Ln = o.L, B = o.B, H = o.H, sh = o.sheer || 0, N = o.N || 9, col = o.col, cin = o.cin || '#5E3A1F', band = o.band || col, keel = o.keel == null ? H * 0.55 : o.keel;

      const st = [];
      for (let i = 0; i <= N; i++) {
        const t = i / N, z = -Ln / 2 + Ln * t, sn = Math.sin(Math.PI * t), w = Math.max(0.35, B / 2 * Math.pow(sn, o.pf || 0.65)), top = H + sh * Math.pow(2 * t - 1, 2), kd = -keel * Math.pow(sn, 0.5);

      }
      for (let i = 0; i < N; i++) for (let j = 0; j < prof.length - 1; j++) {
        const a = st[i][j], b = st[i][j + 1], c = st[i + 1][j + 1], d = st[i + 1][j], cc = (j === 0 || j === prof.length - 2) ? band : col;
        k.quad(a, b, c, d, cc);

        k.quad(inn(a), inn(d), inn(c), inn(b), cin);
      }
      /* end caps where the hull narrows to a point: small posts */
      for (let i = 0; i < N; i++) { L(k, st[i][0], st[i + 1][0], 0.7); L(k, st[i][prof.length - 1], st[i + 1][prof.length - 1], 0.7); }

    }
