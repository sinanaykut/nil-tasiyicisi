     3×10 board snaking 1→10, 11←20, 21→30; three pieces each; four throw sticks (light sides = 1..4, none = 5).
     Land on a lone enemy → swap places; two enemies side by side protect each other; 26 is safe;
     27 (water) sends the piece back to 15 (ankh); bear off by reaching exactly 31; 1, 4 and 5 throw again.
     v28: o.tutorial → Paneb coaches the first game step by step (snTut*); "Kurallar" opens a picture card in every game. */
  const SN_START = { p:[5, 7, 9], a:[6, 8, 10] };
  function snLegal(me, op, v) {
    const out = [];
    me.forEach((f, i) => {
      if (f >= 31) return;
      const t = f + v;
      if (t > 31) return;
      if (t === 31) { out.push({ i, f, t, kind:'off', fin:31, v }); return; }
      if (me.includes(t)) return;
      const j = op.indexOf(t);
      if (j >= 0) { if (t === 26 || op.includes(t - 1) || op.includes(t + 1)) return; out.push({ i, f, t, kind:'cap', j, fin:t, v }); return; }
      if (t === 27) { let r = 15; while (r > 1 && (me.includes(r) || op.includes(r))) r--; out.push({ i, f, t, kind:'water', fin:r, v }); return; }
      out.push({ i, f, t, kind:'move', fin:t, v });
    });
    return out;
  }
  /* a plain heuristic: bear off, capture advanced pieces, reach the safe house, keep pairs, avoid the water */
  function snScore(me, op, m, noise) {
    let s = (m.fin - m.f) * 0.8;
    if (m.kind === 'off') s += 100;
    if (m.kind === 'cap') s += 30 + op[m.j] * 0.6;
    if (m.kind === 'water') s -= 50;
    if (m.fin === 26) s += 22;
    const after = me.map((p, k) => k === m.i ? m.fin : p);
    if (m.fin < 31 && m.fin !== 26) {
      if (after.includes(m.fin - 1) || after.includes(m.fin + 1)) s += 10;
      else s -= op.filter(p => p < m.fin && m.fin - p <= 5).length * 6;
    }
    if (me.includes(m.f - 1) || me.includes(m.f + 1)) s -= 5;
    return s + (noise ? Math.random() * 3 : 0);
  }
  function snBest(side, ms, noise) {
    const me = PZ.pos[side], op = PZ.pos[side === 'p' ? 'a' : 'p'];
    let b = 0, bs = -1e9; ms.forEach((m, k) => { const s = snScore(me, op, m, noise); if (s > bs) { bs = s; b = k; } });
    return b;
  }

  function senetAuto(opt) {
    if (!PZ || PZ.kind !== 'senet' || PZ.fin) return false;
    PZ.auto = true; if (opt && opt.fast) PZ.speed = 0.12; else if (opt && opt.speed) PZ.speed = opt.speed;
    if (PZ.tut && PZ.tut.wait) { snTutOk(); return true; }
    if (PZ.turn === 'p' && PZ.phase === 'throw') snThrow();
    else if (PZ.turn === 'p' && PZ.phase === 'move') snMove(snBest('p', PZ.ms, true));
    return true;
  }
  function snBind() {
    const sh = $('#sheet');
    if (sh && !sh._snBound) {
      sh._snBound = true;
      sh.addEventListener('click', e => {
        if (!PZ || PZ.kind !== 'senet') return;
        const b = e.target.closest && e.target.closest('button');
        if (b) {
          if (b.disabled) return;
          if (b.hasAttribute('data-snact')) snAction();
          else if (b.hasAttribute('data-snok')) snTutOk();
          else if (b.hasAttribute('data-snrules')) snRules(true);
          else if (b.hasAttribute('data-snrx')) snRules(false);
          else if (b.hasAttribute('data-sngive') && !PZ.fin && PZ.phase !== 'over') {
            if (PZ.giveUp && Date.now() - PZ.giveUp < 3500) { PZ.phase = 'over'; play('bad'); snFin(false); }
            else { PZ.giveUp = Date.now(); b.textContent = 'Emin misin?'; b.classList.add('red'); const tok = PZ.tok; PZ.timers.push(setTimeout(() => { if (PZ && PZ.tok === tok) { b.textContent = 'Pes et'; b.classList.remove('red'); } }, 3500)); }
          }
          return;
        }
        const g = e.target.closest && e.target.closest('[data-snsq]');
        if (g) snTapSq(+g.getAttribute('data-snsq'));
      });
    }
    if (!document._snKey) {
      document._snKey = true;
      document.addEventListener('keydown', e => {
        if (!PZ || PZ.kind !== 'senet' || PZ.turn !== 'p' || PZ.fin) return;
        const onBtn = e.target && e.target.closest && e.target.closest('button');
        if ((e.key === ' ' || e.key === 'Enter') && !onBtn) { e.preventDefault(); snAction(); }
        else if (PZ.phase === 'move' && /^Arrow/.test(e.key) && PZ.ms.length > 1) {
          e.preventDefault(); const d = e.key === 'ArrowLeft' || e.key === 'ArrowUp' ? -1 : 1;
          PZ.sel = (PZ.sel + d + PZ.ms.length) % PZ.ms.length; play('click'); snHl();
        }
      });
    }
